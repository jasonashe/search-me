༈ ཨོཾ་སྭ་སྟི། རབ་གནས་ཤིན་ཏུ་བསྡུས་པའི་སྡོམ་ཚིག་རྗེ་བདག་ཉིད་ཆེན་པོའི་ཕྱག་བཞེས་འདུལ་འཛིན་གྲགས་པ་རྒྱལ་མཚན་གྱི་ཟིན་བྲིས་སུ་བཀོད་པ་ལས། སྔོན་འགྲོ་གཏོར་མ་མཆོད་རྫས་རྣམས། །བསངས་སྦྱངས་བསྐྱེད་ནས་བྱིན་གྱིས་བརླབས། །བདག་བསྐྱེད་རྟེན་བསྐྱེད་སྦྱང་བ་བྱ། །སྐུ་སོགས་བྱིན་བརླབ་ཡེ་ཤེས་དྲངས། །བསྐྲད་གཞུག་དབང་ལྷ་སྤྱན་དྲངས་ཏེ། །དབང་བསྐུར་རིགས་བདག་གིས་རྒྱས་གདབ། །སྤྱན་དབྱེ་མཆོད་བསྟོད་༄༅། །གཟུགས་བསྒྱུར་བྱ། །ཤིས་པ་བརྗོད་དང་བརྟན་བཞུགས་བྱ། །སླར་ཡང་ཤིས་པ་བརྗོད་པའོ། །ཞེས་གསུངས་པ་ལྟར། འཇིགས་བྱེད་བཅུ་གསུམ་མ་ལ་བརྟེན་ནས་བྱེད་ན། སྔོན་འགྲོ་གཏོར་མ་ཞེས་པ་ནི། བདག་བསྐྱེད་ཀྱི་ཐོག་མ་སྔོན་འགྲོ་གཏོར་མ་ནས་ཡིན་པར་ཟ་མ་ཏོག་ནས་འབྱུང་བ་ལྟར་ཏེ་འོན་ཀྱང་འཕྲིན་འཆོལ་གྱི་ཚེ་རབ་གནས་ཀྱི་བྱ་བ་ལ་བར་གཅོད་བཟློག་པར་གསོལ་བ་གདབ་དགོས་པའི་དོན་ཏོ། །མཆོད་རྫས་རྣམས། བསངས་སྦྱངས་བསྐྱེད་ནས་བྱིན་གྱིས་བརླབ། ཅེས་པ་ནི། བདག་བསྐྱེད་ཀྱི་མཆོད་པ་དང་ལྷན་ཅིག་ཏུ་མདུན་བསྐྱེད་ཀྱི་མཆོད་པ་བྱིན་རླབས་བྱ་བ་དང། བདག་བསྐྱེད་ཅེས་པ་ནི། ཚོགས་ཞིང་མཆོད་པ་ནས་ཉེར་བསྡུའི་?་བར་རྒྱས་བསྡུས་སྐབས་དང་སྦྱར་བའོ། །སྡོམ་ཚིག་ཏུ་མི་གསལ་ཡང་དེའི་དོན་འགྲོལ་བའི་ཚེ་སྐབས་འདིར་བུམ་སྒྲུབ་གསུངས་པ་བཞིན་བྱ་དགོས་ཏེ་འོག་ཏུ་རྟེན་ལ་ཁྲུས་གསོལ་བྱ་དགོས་པས་སོ། །རྟེན་བསྐྱེད་ཅེས་པ་ནི། རྟན་རྣམས་སོ་སོའི་དམ་ཚིག་པ་བསྐྱེད་པ་སྟེ། དེ་ཡང་བསངས་སྦྱངས་བྱས་ཏེ་པད་ཉི་གཉིས་དང་པད་ཟླའི་སྟེང་དུ་ཆོ་ག་གསུམ་གྱིས་སྐུ་རྟེན་རྣམས་གཙོ་བོ་ཡབ་ཡུམ་དང། གསུང་རྟེན་རྣམས་འདོད་ཆགས་གཤིན་རྗེའི་གཤེད་ཡབ་ཡུམ་དང་ཐུགས་རྟེན་བཞུགས་ཁང་གཏི་མུག་གཤིན་རྗེའི་གཤེད་ཡབ་ཡུམ་དུ་བསྐྱེད་པའོ། །སྦྱང་བ་བྱ། ཞེས་པ་ནི། གཏོར་མ་བྱིན་ནས་བགེགས་སྦྱང་བ་དང། ཁྲུས་གསོལ་བྱས་ནས་དྲི་མ་སྦྱང་བའོ། །སྐུ་སོགས་བྱིན་བརླབ་ཅེས་པ་ནི། དམ་ཚིག་པར་བསྐྱེད་པ་རྣམས་ཀྱི་སྐྱེ་མཆེད་དང་སྐུ་གསུང་ཐུགས་བྱིན་རླབས་བྱ་བ་དང། ཡེ་ཤེས་དྲངས། ཞེས་པ་ནི། ཡེ་ཤེས་པ་སྤྱན་འདྲེན་དང། བསྐྲད། ཅེས་པ་ནི། དེའི་ཕྱི་འབྲང་གི་འགེགས་བསྐྲད་པ་དང་གཞུག །ཅེས་པ་ནི། དམ་ཚིག་པ་ལ་གཞུག་པ་དང། དབང་ལྷ་སྤྱན་དྲངས་ཏེ། དབང་བསྐུར་རིགས་བདག་གིས་རྒྱས་གདབ། ཅེས་པ་ནི། དབང་བསྐུར་རྒྱས་གདབ་བོ། སྤྱན་དབྱེ་མཆོད་བསྟོད་གཟུགས་བསྒྱུར་བྱ། ཞེས་པ་ནི། སྤྱན་དབྱེ་དང། མཆོད་པ་དང། བསྟོད་པ་དང། གཟུགས་བསྒྱུར་རྣམས་ཏེ་གོ་སླའོ། །ཤིས་པ་བརྗོད་དང་བརྟན་བཞུགས་བྱ། ཞེས་པ་ལ། གཞུང་ཁ་ཅིག་ཏུ། མངའ་དབུལ་ཤིས་བརྗོད་བརྟན་བཞུགས་བྱ། ཞེས་འབྱུང་བ་ལྟར་ན། མངའ་དབུལ། ཞེས་པ་ནི། སྤྱིའི་མངའ་དབུལ་ཙམ་མམ་དེ་ཡང་མ་ལྕོགས་ན་མཎྜལ་འབུལ་བ་དང། ཤིས་བརྗོད། ཅེས་པ་ནི། ཡེརྷར་དང་མཆོག་གསུམ་གྱི་བཀྲ་ཤིས་གང་རུང་དང། བརྟན་བཞུགས། ཞེས་པ་ནི། བཅད་ལྷུག་གཉིས་ཀའམ་བསྡུ་ན་ལྷུག་རྐྱང་ཙམ་དང། སླར་ཡང་ཤིས་པ་བརྗོད་པའོ། །ཞེས་པ་ནི། ཡི་དམ་གང་བྱེད་ཀྱི་བཀར་ཤིས་བྱེད་པ་སྟེ། མདོར་ན་སྡོམ་ཚིག་འདི་རྒྱས་བསྡུས་དང་གོ་རིམ་མི་འཕྱུག་པའི་སྲོག་ཏུ་བྱས་ལ། དགེ་ལེགས་རྒྱ་མཚོའི་ཆར་འབེབས་ལྟ་བུའི་རབ་གནས་ཀྱི་གཞུང་རྒྱས་པ་ཞིག་དང། རང་གི་ཡི་དམ་གང་ལ་བརྟེན་ནས་བྱེད་པ་དེའི་གཞུང་ཚིག་གཉིས་གུང་བསྒྲིགས་ཏེ། མེད་མི་རུང་གི་ཆོ་ག་གཙོ་བོ་རྣམས་མ་ཆད་ཅིང། བྱས་ཀྱང་ཕུན་སུམ་ཚོགས་༄༅། །ལམ་བྱས་ཀྱང་ཆད་སྐྱོན་མི་འབྱུང་བ་རྣམས། སྐབས་འདིར་རབ་གནས་རྒྱས་འབྲིང་བསྡུས་གསུམ་གྱི་ནང་ནས་བསྡུས་པ་ཡིན་པའི་ཕྱིར་བཞག་ནས་མཛད་དགོས་པར་སེམས་པས་དེ་བཞིན་མཛད་པར་ཞུ༑

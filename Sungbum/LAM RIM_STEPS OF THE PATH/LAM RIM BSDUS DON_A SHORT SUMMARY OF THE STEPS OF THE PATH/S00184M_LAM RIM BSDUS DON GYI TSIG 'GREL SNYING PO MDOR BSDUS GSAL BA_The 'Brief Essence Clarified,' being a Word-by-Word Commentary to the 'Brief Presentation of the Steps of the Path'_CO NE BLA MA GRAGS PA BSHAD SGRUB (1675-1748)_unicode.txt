༄༅། །ལམ་རིམ་བསྡུས་དོན་གྱི་ཚིག་འགྲེལ་སྙིང་པོ་མདོར་བསྡུས་གསལ་བ་ཞེས་བྱ་བ་བཞུགས་སོ། ། ༄༅༅། །ན་མོ་སུ་མ་ཏི་ཀཱི་ཏི་ཡེ། ཐུགས་བསྐྱེད་མགྱོགས་འགྲོའི་ཤུགས་ཀྱིས་ཉེར་དྲངས་པའི། །གཞན་ཕན་འཕྲིན་ལས་འགྲོ་བའི་དཔལ་དུ་ཤར། །སྐལ་ལྡན་པདམོ་ཡོངས་ཀྱི་གཉེན་གཅིག་པོ། །ཐུབ་དབང་སྨྲ་བའི་ཉི་མ་དེ་ལ་འདུད། །བློ་ཆེན་གཞན་གྱིས་ཤིན་ཏུ་དཔག་དཀའ་བའི། །ཟབ་དོན་ལེགས་པར་འགྲེལ་བའི་རྣམ་དཔྱོད་ཀྱིས། །བཟང་པོའི་མཛད་མཆོག་གྲགས་པའི་སྣང་བ་ཅན། །རྗེ་བཙུན་བླ་མའི་ཞབས་ལ་གུས་ཕྱག་འཚལ། །གང་དེའི་ལེགས་པར་བཤད་པའི་མཐར་ཐུག་ནི། །བྱང་ཆུབ་ལམ་གྱི་རིམ་པ་ཞེས་གྲགས་པ། །དེ་ཡི་གནད་ཀུན་གཅིག་ཏུ་རབ་བསྒྲིལ་པ། །ལམ་རིམ་བསྡུས་པའི་དགོངས་དོན་གསལ་བར་བྱ། །དེ་ལ་འདིར་བདག་ཅག་གི་བླ་མ་ཆོས་ཀྱི་རྒྱལ་པོ་བཙོང་ཁ་པ་ཆེན་པོས་མཛད་པའི་ཉམས་མགུར་རམ། ལམ་རིམ་བསྡུས་པའི་དོན་ནི་གང་བཤད་པར་བྱ་བའི་ཆོས་སོ། །འདི་ལ་གསུམ། སྔོན་འགྲོ་མཆོད་བརྗོད་སོགས་དང། དངོས་གཞི་གཞུང་དོན་དང། མཇུག་གི་བཤད་པའི་ཁྱད་པར་རོ། །དང་པོ་ལ་གཉིས། མཆོད་པར་བརྗོད་པ་དང། བཤད་བྱ་ངོས་བཟུང་བའོ། །དང་པོ་ལའང་གཉིས་ཏེ། དང་པོ་ནི། ལ་ལར་ན་མོ་གུ་རུ་མཉྫུ་གྷོ་ཥཱ་ཡ། ཞེས་འབྱུང་བ་ནི་སླའོ། །གཉིས་པ་ལ་ལྔ་ལས། དང་པོ་ཐུབ་དབང་ལ་ཕྱག་འཚལ་བ་ནི། ཕུན་ཚོགས་དགེ་ལེགས་བྱེ་བས་བསྐྲུན་པའི་སྐུ། །ཞེས་སོགས་རྐང་པ་བཞི་སྟེ། འདི་ལ་མཁྱེན་བརྩེ་ནུས་གསུམ་མམ། སྤངས་རྟོགས་ཀྱི་ཡོན་ཏན་སྤྱི་དང། བྱེ་བྲག་སྐུའི་ཡོན་ཏན་གྱི་དབང་དུ་བྱས་ན། མཚན་དང་དཔེ་བྱད་སོགས་༄༅། །ཡོན་ཏན་དུ་མས་ཁྱད་པར་དུ་བྱས་པ་དེ་ནི། རྒྱུ་གང་ལས་འཁྲུངས་པ་ཡོད་དེ། བསོད་ནམས་དང་ཡེ་ཤེས་ཀྱིས་བསྡུས་པ་ཕུན་སུམ་ཚོགས་པའི་དགེ་ལེགས་རེ་རེ་ཙམ་མ་ཡིན་པར་བྱེ་བས་བསྐྲུན་པ་སྟེ་བསྐྱེད་པ་ཡིན་པའི་ཕྱིར་རོ། །འདིར་བྱེ་བ་ཞེས་པ་ནི། མང་ཚིག་ཡིན་གྱི། གྲངས་ཀྱི་བྱེ་བ་ཙམ་ལ་མི་བཟུང་ངོ། །མཚན་དཔེ་རེ་རེ་ལ་རྒྱུ་ཇི་ཙམ་དགོས་པ་ནི། རིན་ཆེན་འཕྲེང་བ་ལས་གསལ་ལ། རྒྱུ་སོ་སོ་བ་རེ་རེའི་དབང་དུ་བྱས་པ་ནི། ཡུམ་གྱི་མདོ་སོགས་དང། མངོན་རྟོགས་རྒྱན་ནའང་ཡོད་དོ། །གསུང་གི་ཡོན་ཏན་ལ་ངོ་བོའི་དབང་དུ་བྱས་ནའང་རུང་ལ་འབྲས་བུའི་དབང་དུ་བྱས་ན། དེ་བཞིན་གཤེགས་པའི་གསུང་གི་ཡོན་ཏན་ནམ་བྱེད་ལས་ཡོད་དེ། གྲངས་མཐའ་ཡས་པའི་འགྲོ་བའི་གནས་སྐབས་དང་མཐར་ཐུག་གི་རེ་བ་སྐོང་བར་བྱེད་པའི་ཕྱིར་རོ། །རེ་བ་སྐོང་ཚུལ་ཡང། ཆོས་བསྟན་ནས་མངོན་མཐོ་དང་ངེས་ལེགས་ལ་བཀོད་པའི་སྒོ་ནས་ཕན་འདོགས་པ་དང། དེ་དང་དེའི་སྐབས་ཀྱི་ཐེ་ཚོམ་གཅོད་པ་ལ་བྱའོ། །ཐུགས་ཀྱི་ཡོན་ཏན་གྱི་ཁྱད་པར་ཡོད་དེ། ངོ་བོའི་དབང་དུ་བྱས་ན་མ་ལུས་ཇི་ལྟ་ཇི་སྙེད་ཀྱི་ཤེས་བྱ་ཐམས་ཅད་ཇི་བཞིན་མངོན་སུམ་དུ་གཟིགས་པའི་ཕྱིར་རོ། །དེ་ལྟ་བུའི་སྐུ་གསུང་ཐུགས་ཀྱི་ཡོན་ཏན་དང་ལྡན་པ། ཤཱཀྱའི་རིགས་ཀྱི་གཙོ་བོ་ཤཱཀྱ་ཐུབ་པ་དེ་ལ་ཇི་ལྟར་ཕྱག་འཚལ་བ་ཡོད་དེ། ཡན་ལག་གི་དམ་པ་མགོ་བོས་ཕྱག་འཚལ་བ་སྟེ་འདུད་པར་བྱེད་པའི་ཕྱིར་རོ། །གཉིས་པ་ནི། ཟླ་མེད་སྟོན་པ་དེ་ཡི། ཞེས་སོགས་བཞི་སྟེ། ཕྱག་འཚལ་ལོ། །གང་ལ་ན། འགྲན་ཟླ་མེད་པའི་སྟོན་པ་དེ་ཡི་ཆོས་ཀྱི་སྲས་བྱང་སེམས་རྣམས་ཀྱི་ནང་ནས་མཆོག་ཏུ་གྱུར་ཅིང་རྒྱལ་བ་དེའི་སྐུ་གསུང་ཐུགས་ཀྱི་མཛད་པ་ཀུན་གྱི་ཁུར་བསྣམས་པ་སྟེ་བཞེས་ནས་འཇིག་རྟེན་གྱི་ཁམས་འདིར་མ་ཟད་གྲངས་མེད་པའི་སངས་རྒྱས་ཀྱི་ཞིང་དུ་བྱང་སེམས་སོགས་སྤྲུལ་པ་དུ་མའི་གར་གྱིས་རྣམ་པར་རོལ་བ་ཅན་མི་ཕམ་བྱམས་པ་དང་འཇམ་པའི་དབྱངས་གཉིས་ལའོ། །མཛད་པའི་ཁུར་བཞེས་ཚུལ་ནི། བྱམས་འཇམ་དབྱངས་གཉིས་ཀྱིས་ཐུབ་པའི་དམ་པའི་ཆོས་སྤྱི་དང། ཁྱད་པར་རྒྱ་ཆེན་སྤྱོད་པ་དང་ཟབ་མོའི་ལྟ་བའི་བརྒྱུད་རིམ་བཟུང་སྟེ་སྤེལ་བར་མཛད་པ་འདིའོ། །གསུམ་པ་ནི། ཤིན་ཏུ་དཔག་པར་དཀའ་བ་སོགས་བཞི་སྟེ། ཕྱག་འཚལ་ལོ། །གང་ལ་ན། རྒྱལ་བའི་གསུང་རབ་སྤྱི་དང་ཁྱད་པར། ཤིན་ཏུ་དཔག་པར་ཏེ་རྟོགས་པར་དཀའ་བ་རྒྱལ་བའི་ཡུམ་རྒྱས་འབྲིང་བསྡུས་པ་གསུམ་གྱི་དོན་ཇི་ལྟ་བ་བཞིན་དུ་དགོངས་པ་འགྲེལ་པར་མཛད་པའི་སྒོ་ནས་འཛམ་གླིང་གི་རྒྱན་དུ་གྱུར་པ་མཚོན་ལ་མགོན་པོ་ཀླུ་སྒྲུབ་དང་འཕགས་པ་ཐོགས་མེད་ཅེས་ནི་ས་འོག་ས་སྟེང་ས་བླ་སྟེ་ས་གསུམ་ན་ཡོངས་སུ་གྲགས་པའི་ཞབས་ལའོ། །གང་གིས་ན། དེ་?་བློ་བཟང་གྲགས་པའི་དཔལ་གྱིས་སོ། །བཞི་པ་ནི། ཤིང་རྟ་ཆེན་པོ་གཉིས་ལས་སོགས་བཞི་སྟེ། འདུད་དོ། །གང་ལ་ན། ཤིང་རྟ་ཆེན་པོ་ཀླུ་སྒྲུབ་དང་ཐོགས་མེད་རྣམ་པ་གཉིས་ལས་ལེགས་པར་བརྒྱུད་པའི་ཟབ་མོ་ལྟ་བ་དང། རྒྱ་ཆེན་སྤྱོད་པའི་ལམ་གྱི་རིམ་པ་ཚང་ལ་མ་ནོར་བར་ཡོངས་སུ་རྫོགས་པའི་གནད་བསྡུས་པའི་གདམས་པའི་མཛོད་འཛིན་པ་དཔལ་མར་མེ་མཛད་ལའོ། །དེ་ཡང་ཇོ་བོ་ཆེན་པོས་བླ་མ་གསེར་གླིང་པ་བསྟེན་ནས་བྱམས་པ་དང་ཐོགས་མེད་ནས་བརྒྱུད་པའི་གདམས་པ་གསན་ལ། རིགས་པའི་ཁུ་བྱུག་བསྟེན་ནས་འཇམ་དབྱངས་དང་ཀླུ་སྒྲུབ་ནས་བརྒྱུད་པའི་གང་ས་དུད་?་མ་ལུས་པ་གསན་ནས་གདམས་པའི་ཆུ་བོ་གཉིས་ཟུང་དུ་འབྲེལ་བར་མཛད་དོ། །ལྔ་པ་ནི། རབ་འབྱམས་གསུང་རབ་ཀུན་ལ་སོགས་བཞི་སྟེ། ཕྱག་འཚལ་ལོ། །གང་ལ་ན། ཕྱོགས་རེ་ཙམ་མ་ཡིན་པར་རབ་འབྱམས་གསུང་རབ་ཀུན་ལ་ལྟ་བའི་མིག་ཏུ་གྱུར་ཅིང། སྐལ་བ་བཟང་པོར་གྱུར་བའི་གདུལ་བྱ་ཐར་པར་བགྲོད་པའི་འཇུག་ངོགས་ཏེ་ལམ་༄༅། །མཆོག་ཏུ་གྱུར་པའི་སྐྱེས་བུ་གསུམ་གྱི་ལམ་གྱི་རིམ་པ་འདི་ཉིད། བརྩེ་བའི་ཐུགས་ཀྱིས་བསྐྱོད་པ་སྟེ་དྲངས་པའི་ཐབས་མཁས་ཀྱི་མཛད་པ་ཡིས་གདུལ་བྱ་ལ་གསལ་བར་མཛད་པའི་བཤེས་གཉེན་རྣམས་ལའོ། །ཚུལ་ཇི་ལྟར་ན། ཚིག་ཙམ་མ་ཡིན་པར་ཡིད་གུས་པའི་སྒོ་ནས་སོ། འདིར་ལྟ་བའི་མིག་ཅེས་པ་ནི། ཁ་ཅིག་བཤེས་གཉེན་ལ་སྦྱོར་བར་སྣང་ཡང། ལམ་རིམ་ཉིད་ལ་བྱེད་པར་རྗེའི་དགོངས་པ་ཡིན་པར་ལམ་རིམ་ཆེན་མོའི་མཇུག་ཏུ་གསལ་ལོ། །གཉིས་པ་བཤད་བྱ་ངོས་བཟུང་བ་ལ་གསུམ། བརྒྱུད་པའི་ཁྱད་པར། ཁྱད་ཆོས། དགོས་པའོ། །དང་པོ་ནི། འཛམ་གླིང་མཁས་པ་ཡོངས་ཀྱི་སོགས་བཞི་སྟེ། འཛམ་གླིང་འདིར་བྱོན་པའི་མཁས་པ་ཡོངས་ཀྱི་གཙུག་གི་རྒྱན་དུ་གྱུར་ཅིང། མཁས་པའི་སྙན་པ་ཕྱོགས་ཀུན་དུ་གྲགས་པའི་བ་དན་འགྲོ་བའི་གནས་འདི་ན་ཉི་མ་ལྟར་ལྷང་ངེར་གསལ་བ་ཀླུ་སྒྲུབ་དང་ཐོགས་མེད་གཉིས་ལས་རིམ་པ་བཞིན་དུ་ལེགས་པར་བརྒྱུད་པའི་བྱང་ཆུབ་ལམ་གྱི་རིམ་པའི་གཞུང་ཉིད་འཆད་པར་བྱའོ། །ཞེས་རྩོམ་པ་དམ་བཅའ་དོན་གྱིས་མཛད་དོ། །གཉིས་པ་ལ་གཉིས། དཔེ་དོན་གྱི་ཁྱད་པར་དང། ཁྱད་ཆོས་དངོས་སོ། །དང་པོ་ནི། སྐྱེ་རྒུའི་འདོད་དོན་མ་ལུས་སྐོང་བས་ན། །སོགས་བཞི་སྟེ། དེ་ལྟར་བརྒྱུད་པའི་བྱང་ཆུབ་ལམ་གྱི་རིམ་པ་འདི་ནི། དཔེ་དོན་གྱི་ཁྱད་པར་ཇི་ལྟ་བུ་དང་ལྡན་ཞེ་ན། ཡོད་དེ། ཐར་འདོད་ཀྱི་སྐྱེ་རྒུའི་འདོད་དོན་མ་ལུས་པ་སྐོང་བས་ན། གདམས་པ་རིན་ཆེན་དབང་གི་རྒྱལ་པོ་ལྟ་བུ་སྟེ། ཡིད་བཞིན་གྱི་ནོར་བུ་རིན་པོ་ཆེ་དང་ཆོས་མཚུངས་ཤིང། ཐེག་པ་ཆེ་ཆུང་གི་གཞུང་ལུགས་བཟང་པོ་སྟོང་གི་ལེགས་བཤད་ཀྱི་ཆུ་བོ་མ་ལུས་པ་འདུ་བའི་ཕྱིར་ན། དཔལ་ལྡན་ཏེ་ཕུན་སུམ་ཚོགས་པ་ལེགས་པར་བཤད་པའི་རྒྱ་མཚོ་ལྟ་བུའང་ཡིན་ནོ། །འདོད་པའི་དོན་སྐང་ཚུལ་ནི། འདིའི་དོན་ཉམས་སུ་བླངས་པ་ལས་མངོན་མཐོ་དང་ངེས་ལེགས་འགྲུབ་པ་དང། གཞུང་ལུགས་སྟོང་སྟེ་མང་?་ཚིག་སྨོས་པ་ཡིན་པས་གྲངས་སྟོང་ཁོ་ནར་མི་བཟུང་ཞིང། དེ་རྣམས་འདུ་ཚུལ་ཡང་རྗོད་བྱེད་ཀྱི་ཚིག་ཐམས་ཅད་འདུ་བའི་དོན་མིན་གྱི་བརྗོད་བྱའི་དོན་མཐའ་དག་འདུ་བའོ། །གཉིས་པ་ནི། བསྟན་པ་ཐམས་ཅད་འགལ་མེད་སོགས་བཞི་སྟེ། དེ་ལྟ་བུའི་གཞུང་འདི་ནི། ཁྱད་ཆོས་སམ་ཆེ་བ་བཞི་དང་ལྡན་ཏེ། གདམས་པ་འདི་ལ་བརྟེན་ནས་རྒྱལ་བས་ཆོས་ཚུལ་ཇི་སྙད་ཅིག་གསུངས་པ་ཐམས་ཅད་ལས། ཁ་ཅིག་ནི་ལམ་གྱི་གཙོ་བོ། ཁ་ཅིག་ནི་དེའི་ཡན་ལག་སྟོན་བྱེད་དུ་ཤེས་ནས་དངོས་དང་བརྒྱུད་བ་ཅི་རིགས་ཀྱི་སྒོ་ནས་འཚང་རྒྱ་བའི་ཆ་རྐྱེན་དུ་གོ་བར་འགྱུར་བས་ན། བསྟན་པ་ཐམས་ཅད་འགལ་མེད་དུ་རྟོགས་པའི་ཆེ་བ་དང། མདོ་སྔགས་ཀྱི་གསུང་རབ་རྣམས་དང་དེའི་དགོངས་འགྲེལ་གྱི་བསྟན་བཅོས་རྣམས་བཤད་ཆོས་ཙམ་དུ་བཟུང་ནས་ཉམས་སུ་ལེན་རྒྱུ་དེ་དག་གི་དོན་ལས་གཞན་ཞིག་ན་ཡོད་པར་འཛིན་པའི་ལོག་རྟོག་རྣམས་ལྡོག་སྟེ། གསུང་རབ་དགོངས་འགྲེལ་དང་བཅས་པའི་བརྗོད་བྱ་མཐའ་དག་ཉམས་སུ་ལེན་གྱི་གོ་རིམ་དང་མཐུན་པར་བཤེས་གཉེན་བསྟེན་ཚུལ་ནས་ཞི་ལྷག་གི་བར་བསྡུས་ནས་དཔྱད་སྒོམ་དང་འཛེག་སྒོམ་ཅི་རིགས་ཀྱི་སྒོ་ནས་ཉམས་སུ་ལེན་ཚུལ་ཤེས་པར་འགྱུར་བས་ན། གསུང་རབ་ཐམས་ཅད་གདམས་པར་འཆར་བའི་ཆེ་བ་དང། དེ་ལྟར་གསུང་རབ་ཐམས་ཅད་གདམས་ངག་གི་མཆོག་ཡིན་ཀྱང་བློ་མ་སྦྱངས་པའི་ལས་དང་པོ་བས་གཞུང་ཆེ་བ་དེ་དག་གི་དགོངས་དོན་རང་སྟོབས་ཀྱིས་རྟོགས་ནས་ཉམས་སུ་ལན་མི་ཤེས་པས། དེ་དག་གི་དགོངས་དོན་རྣམས་འདུས་པའི་བླ་མའི་མན་ངག་བྱང་ཆུབ་ལམ་གྱི་རིམ་པ་འདི་ལྟ་བུ་ལ་བརྟེན་ནས་གཞུང་ཆེ་བ་དེ་རྣམས་ཀྱི་དགོངས་དོན་ལེགས་པར་རྟོགས་པར་འགྱུར་བས་ན། རྒྱལ་བའི་དགོངས་པ་བདེ་བླག་ཏུ་རྟོགས་པའི་ཆེ་བ་དང། དེ་ལྟར་རྙེད་པའི་ཚེ་སངས་རྒྱས་ཀྱི་བཀའ་ཐམས་ཅད་དངོས་དང་བརྒྱུད་བ་ཅི་རིགས་ཀྱི་སྒོ་ནས་འཚང་རྒྱ་བའི་ཐབས་སུ་གོ་ནས་འཚང་རྒྱ་བའི་ཐབས་མི་སྟོན་པའི་སངས་རྒྱས་ཀྱི་བཀའ་མེད་པར་ཤེས་ཏེ། གསུང་རབ་དེ་དག་ལ་གདུལ་བྱའི་དབང་གིས་བརྗོད་བྱ་ཅུང་ཟད་རེ་མི་མཐུན་པ་དང་མཆོག་དམན་ཡོད་ཀྱང། ཐམས་ཅད་འཚང་རྒྱ་བའི་ཐབས་སྟོན་པ་ཙམ་ལ་ཁྱད་པར་མེད་པར་ངེས་ནས་ཆོས་སྤོང་བའི་ལས་མི་འབྱུང་བ་སྟེ། ཉེས་སྤྱོད་ཆེན་པོའི་གཡང་ས་ལས་ཀྱང་སྲུང་བའི་ཆེ་བ་བཞི་༄༅། །དང་ལྡན་པའི་ཕྱིར་རོ། །གསུམ་པ་ལ་གཉིས། ཆེ་བ་དེ་ལྡན་གྱི་ཆོས་འདི་ཉམས་སུ་ལེན་རིགས་པ་དང། ཉམས་སུ་བླངས་པའི་ཕན་ཡོན་ནོ། །དང་པོ་ནི། དེ་ཕྱིར་རྒྱ་བོད་མཁས་པའི་སོགས་བཞི་སྟེ། སྔར་བཤད་པའི་ཆེ་བ་དེ་ལྟར་ཡོད་པའི་རྒྱུ་མཚན་དེའི་ཕྱིར་ན། རྒྱ་གར་དང་བོད་ཀྱི་མཁས་པའི་སྐྱེ་བོ་ནི་སྐལ་བ་དང་ལྡན་པ་དུ་མས་ཉམས་ལེན་དུ་བསྟེན་པའི་གདམས་པ་མཆོག་ཏུ་གྱུར་པ་སྐྱེས་བུ་ཆུང་འབྲིང་ཆེ་གསུམ་གྱི་ལམ་གྱི་རིམ་པ་ཡིས་ནི་ཡིད་རབ་ཏུ་མི་འཕྲོག་པའི་རྣམ་དཔྱོད་དང་ལྡན་པ་སུ་ཞིག་ཡོད་དེ་མེད་དོ། །གཉིས་པ་ནི། གསུང་རབ་ཀུན་གྱི་སྙིང་པོ་སོགས་བཞི་སྟེ། ཐེག་པ་ཆེ་ཆུང་གི་གསུང་རབ་ཀུན་གྱི་༆༆༆སྙིང་པོ་བསྡུ་བསྡུ་བའི་གཞུང་གི་ཚུལ་འདི་ནི། ལན་མང་དུ་ལྟ་ཅི་ཐུན་རེ་སྟོན་པ་དང་ལན་རེ་ཉན་པས་ཀྱང་གཞུང་ཆེ་བ་ནས་དམ་ཆོས་འཆད་པ་དང་ཐོས་པའི་ཕན་ཡོན་གྱི་ཚོགས་རླབས་ཆེན་འབྱུང་བར་གསུངས་པ་དེ་རྣམས་སྡུད་པར་ངེས་པས་སོ། །རྒྱུ་མཚན་དེས་ན། དེའི་དོན་ཡིད་ལ་བསམ་པར་བྱའོ། །ཞེས་པ་ནི་དོན་བསྡུ་བའི་སྒོ་ནས་གདམས་པའོ། །བསྡུ་བསྡུ་བ་ནི་གསུང་རབ་ཐམས་ཅད་ཀྱི་གནད་བསྡུས་པ་དང་དེ་ཡང་མདོར་བསྡུས་ཏེ་བསྟན་པའི་དོན་ནོ། །གཉིས་པ་གཞུང་གི་དོན་བཤད་པ་ལ། མཆོད་བརྗོད་སོགས་ནི། གཞུང་གི་ཡན་ལག་ཡིན་ལ། འདིར་གཞུང་དོན་དངོས་ཀྱི་དབང་དུ་བྱས་ཏེ་འཆད་པ་ལ་གཉིས། ཐུན་མོང་བ་དང། ཐུན་མོང་མ་ཡིན་པའི་ལམ་ལ་བསླབ་ཚུལ་ལོ། །དང་པོ་ལ་གཉིས། སྔོན་འགྲོ་དང། དངོས་གཞིའོ། །དང་པོ་ནི། དེ་ནས་འདི་ཕྱིའི་ལེགས་ཚོགས་སོགས་བརྒྱད་དེ། ལམ་རིམ་ཡོངས་རྫོགས་དང་དེའི་ཆ་ཤས་ཙམ་ལས་སུ་བླངས་པ་ལས་ཀྱང་ཕན་ཡོན་ཆེན་པོ་ཡོད་ཚུལ་བསམ་པ་སོགས་སྔོན་དུ་བཏང་སྟེ་དེ་ནས་འདི་ལྟར་ཉམས་སུ་བླང་བར་བྱའོ། །ཞེས་མཚམས་སྦྱར་རོ། །ཕན་ཡོན་དེ་ལྟར་མཐོང་བ་དེས། འདི་དང་ཕྱི་མའི་ལེགས་ཚོགས་ཇི་སྙེད་མཆིས་པའི་རྟེན་འབྲེལ་ལེགས་པར་འགྲིག་པར་འགྱུར་བའི་རྩ་བ་ནི། རང་ལ་ལམ་སྟོན་པའི་དགེ་བའི་བཤེས་གཉེན་དམ་པར་ཤེས་པར་བྱས་ཏེ། འབད་པ་དྲག་པོ་ཡིས། བླ་མ་ལ་སྐྱོན་རྟོག་གི་བློ་བཀག་ནས་དད་པ་སྐྱེ་ཅི་ཐུབ་བྱེད་པའི་སྒོ་ནས་བསམ་པས་བསྟེན་པ་དང། བླ་མ་མཉེས་པ་སྒོ་མཐའ་དག་ནས་བསྒྲུབ་ཅིང་མ་མཉེས་པ་ནམ་ཡང་མི་བསྒྲུབ་པ་དང། ལུས་ཀྱིས་བཀུར་ཞིང་ངག་གིས་བསྟོད་པ་སོགས་དང། ཅི་གསུང་བཀའ་བཞིན་དུ་སྒྲུབ་པའི་སྒོ་ནས་སྦྱོར་བས་ཚུལ་བཞིན་དུ་བསྟེན་པ་རུ་སྟེ་བསྟེན་པ་ལ་རག་ཡས་པར་མཐོང་ནས་རྐྱེན་ཅུང་ཟད་ཙམ་དུ་མ་ཟད། སྲོག་གི་ཕྱིར་ཡང་མི་གཏོང་བར་བཀའ་བཞིན་དུ་སྒྲུབ་པའི་མཆོད་པས་མཉེས་པར་བྱེད་དགོས་སོ། །དེའི་མཇུག་ཏུ། རྗེའི་གསུང་འབུམ་ཐོར་བུ་ལས། རྣལ་འབྱོར་ངས་ཀྱང་ཉམས་ལེན་དེ་ལྟར་བྱས། །ཐར་འདོད་ཁྱེད་ཀྱང་དེ་བཞིན་བསྐྱང་འཚལ་ལོ། ། ཞེས་འབྱུང་བ་ནི། རྗེའི་གསུང་དངོས་ཡིན་ལ། ཆོས་སྤྱོད་ལས། རྗེ་བཙུན་བླ་མས་ཉམས་ལེན་དེ་ལྟར་མཛད། །ཐར་འདོད་བདག་ཀྱང་དེ་བཞིན་བསྐྱང་འཚལ་ལོ། །ཞེས་འབྱུང་བ་ནི། ཕྱིས་གཞན་གྱིས་བསྒྱུར་བ་སྟེ། དོན་གོ་སླ་ཞིང་ཕྱི་མ་གཞན་ལའང་སྦྱོར། བླ་མའི་མཚན་ཉིད། བསྟེན་ཚུལ་སོགས་ལམ་རིམ་སོགས་ནས་ཤེས་པར་བྱའོ། །གཉིས་པ་ལ་གཉིས། དལ་འབྱོར་ལ་སྙིང་པོ་ལེན་པར་བསྐུལ་བ་དང། སྙིང་པོ་ཇི་ལྟར་ལེན་ཚུལ་ལོ། །དང་པོ་ནི། དལ་བའི་རྟེན་འདི་ཡིད་བཞིན་ནོར་སོགས་ཏེ། དལ་བརྒྱད་འབྱོར་བཅུ་ལྡན་པའི་སྒོ་ནས་ཆོས་སྒྲུབ་པ་ལ་ཁོས་པས་ན་དལ་བའི་རྟེན་ཁྱད་པར་ཅན་འདི་ནི། ཡིད་བཞིན་གྱི་ནོར་བུ་རིན་པོ་ཆེ་ལས་ཀྱང་ལྷག་པ་འདི་འདྲ་བ་ནི་སྔར་ཡང་ཡང་རྙེད་པ་མ་ཡིན་གྱི་རྙེད་པ་དུས་ད་རེས་ཙམ་ཞིག་ཡིན་ནོ། །དེས་རྙེད་ན་དོན་ཆེ་བར་མ་ཟད། རྙེད་པར་དཀའ་བར་བསྟན་ཏོ། །ཡིད་བཞིན་གྱི་ནོར་བུ་ལས་ལྷག་ཚུལ་ནི་ནོར་བུ་ལས་དགོས་འདོད་འབྱུང་ཡང་ཚེ་འདིའི་གནས་སྐབས་ཀྱི་འདོད་དོན་ཡིན་ལ། རྟེན་འདི་ལ་བརྟེན་ནས་མངོན་མཐོ་ཙམ་དུ་མ་ཟད། ངེས་ལེགས་ཐར་པ་དང་ཐམས་ཅད་མཁྱེན་པ་བསྒྲུབས་ནའང་ཐོབ་ནུས་པར་གསུངས་པའི་ཕྱིར་ན་ལྷག་གོ །དེ་ལྟར་རྙེད་དཀའ་ཞིང་རྙེད་ན་དོན་ཆེ་བའི་དལ་འབྱོར་གྱི་རྟེན་འདི་དོན་མེད་དུ་མི་བཏང་བར་འདི་ལ་རང་གིས་ཅི་ནུས་ཀྱི་སྒོ་ནས་ཆོས་བྱ་བའི་དམ་བཅའ་བརྟན་པོ་བྱའོ། །གཉིས་པ་སྙིང་པོ་ཇི་ལྟར་ལེན་ཚུལ་ལ་༄༅། །གཉིས། ལམ་གྱི་དབང་དུ་བྱས་པའི་གསུང་རབ་རྣམས་འདུ་ཚུལ་དང། ཉམས་སུ་ལེན་ཚུལ་དངོས་སོ། །དང་པོ་ནི། སངས་རྒྱས་བཅོམ་ལྡན་འདས་ཀྱིས་ཆོས་ཀྱི་ཕུང་པོ་ཇི་སྙེད་ཅིག་གསུངས་པ་ནི་སེམས་ཅན་གྱི་དོན་དུ་ཡིན་ལ། དེ་ཡང་མངོན་མཐོ་ངེས་ལེགས་གཉིས་ལས། མངོན་མཐོ་ལས་བརྩམས་ཏེ་གཙོ་བོར་གསུངས་པ་རྣམས་སྐྱེས་བུ་ཆུང་ངུ་དངོས་སམ་དེ་དང་ཐུན་མོང་བའི་ཆོས་སྐོར་དུ་འདུ་ཞིང། ངེས་ལེགས་ལ་གཉིས་ལས་ཐར་པ་ལ་བརྩམས་ཏེ་གཙོ་བོར་གསུངས་པ་རྣམས་སྐྱེས་བུ་འབྲིང་དངོས་སམ་དེ་དང་ཐུན་མོང་བའི་ཆོས་སྐོར་དུ་འདུ་ལ། ཐམས་ཅད་མཁྱེན་པ་སྒྲུབ་པ་ལས་བརྩམས་ཏེ་གསུངས་པ་ཐམས་ཅད་སྐྱེས་བུ་ཆེན་པོའི་ཆོས་སྐོར་དུ་འདུ་བ་ཡིན་ནོ། །འདིར་སྐྱེས་བུ་ཆུང་འབྲིང་དང་ཐུན་མོང་བའི་ལམ་འགའ་ཞིག་སྐྱེས་བུ་ཆེན་པོའི་ལམ་གྱི་ཡན་ལག་ཏུ་དགོས་པ་དེར་འཁྲིད་པ་ཡིན་གྱི། སྐྱེས་བུ་ཆུང་འབྲིང་དངོས་ཀྱི་ལམ་ལ་འཁྲིད་པ་ནི་མ་ཡིན་ཏེ། མངོན་མཐོ་ཙམ་ཞིག་དོན་གཉེར་གྱི་བློ་དེ་སྐྱེས་བུ་ཆུང་ངུའི་ལམ་དངོས་དང། རང་དོན་དུ་ཐར་པ་ཙམ་ཞིག་དོན་གཉེར་གྱི་བློ་དེ་སྐྱེས་བུ་འབྲིང་གི་ལམ་དངོས་ཡིན་པས། དེར་འཁྲིད་ན་ནི་སྐྱེས་བུ་ཆེན་པོའི་ལམ་གྱི་གོ་སར་འཁྲིད་པའི་སྐྱོན་དུ་འགྱུར་བའི་ཕྱིར་རོ། །འོ་ན། དེ་གཉིས་དང་ཐུན་མོང་བའི་ལམ་ཇི་ལྟ་བུ་ཞེ་ན། སྡིག་སྤོང་དགེ་བ་སྒྲུབ་པའི་སྒོ་ནས་མངོན་མཐོ་དོན་གཉེར་གྱི་བློ་དེ། སྐྱེས་བུ་ཆུང་ངུ་དང་ཐུན་མོང་བའི་ལམ་དང། འཁོར་བ་ལས་གྲོལ་འདོད་ཀྱི་སྒོ་ནས་ཐར་པ་དོན་གཉེར་གྱི་བློ་དེ་སྐྱེས་བུ་འབྲིང་དང་ཐུན་མོང་བའི་ལམ་ཡིན་ཏེ། དེ་གཉིས་ཀའང་བྱང་སེམས་ཀྱི་རྒྱུད་ལ་ཡོད་ཅིང། དེས་ཉམས་སུ་ལེན་དགོས་པའི་ཕྱིར་རོ། །གལ་ཏེ་སྐྱེས་བུ་ཆུང་འབྲིང་གི་ལམ་གཉིས་སྐྱེས་བུ་ཆེན་པོའི་ལམ་གྱི་གོལ་ས་?་ཡིན་ན། དེའི་གེགས་སམ་མི་མཐུན་ཕྱོགས་ཡིན་པའི་ཕྱིར་འཚང་རྒྱ་བའི་ཐབས་ཀྱང་མ་ཡིན་པར་འགྱུར་རོ་ཞེ་ན། སྐྱོན་མེད་དེ། ལམ་དེ་གཉིས་སྐྱེས་བུ་ཆེན་པོ་མྱུར་དུ་འཚང་རྒྱ་བའི་ལམ་གྱི་གོལ་ས་ཡིན་པས་དེ་འཚང་རྒྱ་བའི་ཐབས་མ་ཡིན་ཀྱང། ཉན་རང་རྣམས་རིམ་གྱིས་རྒྱུད་སྨིན་ཏེ་འཚང་རྒྱ་བའི་ཐབས་སུ་འགྱུར་བ་ལ་འགལ་བ་ཅི་ཡང་མེད་པའི་ཕྱིར་རོ། །དེ་ཡང་རང་ཉིད་གཅིག་པུ་ཞི་བདེ་དོན་གཉེར་གྱི་བློ་ལྟ་བུ་ལ་གཙོ་བོར་དགོངས་པའོ། །གཉིས་པ་ལ་གསུམ། སྐྱེས་བུ་ཆུང་ངུ་དང་ཐུན་མོང་བའི་ལམ། འབྲིང་དང་ཐུན་མོང་བའི་ལམ། ཆེན་པོའི་ལམ་དངོས་སོ། །དང་པོ་ལ་གཉིས། ཕྱི་མ་དོན་གཉེར་གྱི་བློ་བསྐྱེད་པ། ཕྱི་མའི་བདེ་ཐབས་བསྟེན་པའོ། །དང་པོ་ལ་གཉིས། འཆི་བ་མི་རྟག་པ་བསམ་པ་དང། ངན་འགྲོའི་སྡུག་བསྔལ་བསམ་པའོ། །དང་པོ་ནི། རྙེད་དཀའ་འཇིག་སླ་སོགས་བཞི་སྟེ། རྒྱུ། ངོ་བོ། དཔེའི་སྒོ་ནས་རྙེད་པར་དཀའ་ཞིང། འཆི་རྐྱེན་མང་བས་འཇིག་པར་སླ་བ་ནི། དཔེར་ན། ནམ་མཁའི་གློག་མྱུར་དུ་འཇིག་པ་དང་འདྲ་བའི་ཚུལ་འདི་བསམས་ནས་དགྲ་འདུལ་གཉེན་སྐྱོང་སོགས་འཇིག་རྟེན་གྱི་བྱ་བ་ཀུན་དཔེར་ན། སྙིང་པོ་མེད་པའི་ཤུན་པའི་སྤུན་པ་འཕྱར་བ་བཞིན་དུ་དོན་མེད་པར་རྟོགས་པར་གྱུར་ནས་ཉིན་མཚན་ཀུན་ཏུ་རྟེན་འདི་ལ་སྙིང་པོ་ལེན་པ་དགོས་སོ། །གཉིས་པ་ངན་འགྲོའི་སྡུག་བསྔལ་བསམ་པ་ནི། ཤི་ནས་ངན་འགྲོར་མི་སྐྱེ་ཞེས་པ་སྟེ། དེ་ལྟར་འཆི་བ་མི་རྟག་པར་མ་ཟད། ཤི་ནས་ལས་དཀར་ནག་གཉིས་ཀྱིས་ཇི་ལྟར་འཕངས་པ་བཞིན་སྐྱེ་བ་ལེན་དགོས་ལ། དེ་ནི་དགེ་བའི་ལས་ཀྱིས་བདེ་འགྲོར་སྐྱེ་ཞིང་བདེ་བ་མྱོང་ལ། མི་དགེ་བའི་ལས་ཀྱིས་ངན་འགྲོར་སྐྱེ་ཞིང་སྡུག་བསྔལ་མྱོང་ངོ། །བདག་གིས་ནི་བདེ་སྐྱོ་འགྲོར་སྐྱེ་བའི་རྒྱུ་ཆེར་མ་བསྒྲུབས་ལ། ངན་འགྲོར་སྐྱེ་བའི་ནི་མང་དུ་བསགས་སོ། །དེའི་ཕྱིར་ན་ཕྱི་མར་དམྱལ་བ་ལ་སོགས་པ་ངན་འགྲོར་མི་སྐྱེ་བའི་གདེངས་ཐོབ་པ་བདག་ལ་མེད་ཅིང། མེད་པར་མ་ཟད། དེར་སྐྱེ་བཤུགས་ཆེའོ་སྙམ་ནས་ངན་འགྲོའི་སྡུག་བསྔལ་ཡང་ཡང་བསམས་ཏེ་དེ་ལས་ཐར་འདོད་ཤུགས་ཅན་བསྐྱེད་དོ། །གཉིས་པ་ཕྱི་མའི་བདེ་ཐབས་ལ་གསུམ། སྐྱབས་འགྲོ་བསླབ་བྱ་དང་བཅས་པ་དང། དེ་ལ་ལས་གཉིས་ཀྱི་བླང་དོར་ལ་འཇུག་དགོས་པ་དང། རྣམ་མཁྱེན་སྒྲུབ་པའི་རྟེན་བཤད་པའོ། །དང་པོ་ནི། དེ་ཡི་འཇིགས་སྐྱོབ་དཀོན་མཆོག་གསུམ་སོགས་གསུམ་སྟེ། དེ་ལྟར་ལས་ཀྱི་དབང་གིས་ངན་འགྲོར་ལྟུང་བར་འགྱུར་བས་ངན་འགྲོ་དེ་ཡི་འཇིགས་པ་ལས་སྐྱོབ་པའི་སྐྱབས་ནི་དཀོན་མཆོག་གསུམ་དུ་ངེས་༄༅། །སོ། །དེ་ཡང་ཇི་སྐད་དུ། སངས་རྒྱས་ཆོས་དང་དགེ་འདུན་ཏེ། །ཐར་པ་འདོད་པ་རྣམས་ཀྱི་སྐྱབས། །ཞེས་འབྱུང་བ་ལྟར། དཀོན་མཆོག་གསུམ་སོ་སོའི་ཁྱད་པར་དང་ཡོན་ཏན་ཤེས་པའི་སྒོ་ནས་སྐྱབས་སུ་འགྲོ་དགོས་སོ། །འགྲོ་ཚུལ་ཡང་སྙིང་ཐག་པ་ནས་དཀོན་མཆོག་གསུམ་ལ་འཁོར་བ་དང་ངན་སོང་གི་སྡུག་བསྔལ་ལས་སྐྱོབ་པའི་ནུས་པ་ཡོད་པར་ཡིད་ཆེས་པའི་སྒོ་ནས་རེ་ལྟོས་འཆའ་བའི་བློ་བསྐྱེད་པའོ། །དེ་ལྟར་འཇིགས་པ་དེ་ལས་སྐྱོབ་ནུས་པར་ཤེས་ནས་སྐྱོབ་པ་དེའི་ཕྱིར་དུ་དེ་གསུམ་ལ་སྐྱབས་སུ་སོང་བའི་སྐྱབས་འགྲོ་ཤིན་ཏུ་བསྟེན་པ་དང་སྐྱབས་འགྲོ་དེ་ཡི་བསླབ་བྱ་ཐུན་མོང་བ་དང་ཐུན་མོང་མ་ཡིན་པ་ལམ་རིམ་སོགས་ནས་བཤད་པ་ལ་ཉམས་པ་མེད་པར་བྱའོ། །གཉིས་པ་ནི། དེ་ཡང་དཀར་ནག་ལས་འབྲས་སོགས་ཏེ། དེ་ཡང་དད་པ་ལ་སྤྱིར་དང་བའི་དད་པ། ཡིད་ཆེས་པའི་དད་པ། འདོད་པའི་དད་པ་དང་གསུམ་ཡོད་ཀྱང་འདིར་ནི་གཙོ་བོ་ཡིད་ཆེས་པའི་དད་པ་སྟེ། དཀོན་མཆོག་གིས་སྐྱོབ་པ་དེ་ཡང་ལས་འབྲས་ལ་བླང་དོར་ཕྱིན་ཅི་ལོག་ཏུ་བསྒྲུབས་ན་མི་འབྱུང་བས། དཀར་པོ་དགེ་བ་དང་ནག་པོ་མི་དགེ་བའི་ལས་འབྲས་བུ་དང་བཅས་པ་ཇི་ལྟར་ཡིན་ལེགས་པར་བསམས་ནས་དགེ་བ་འབྲས་བུ་དང་བཅས་པ་ལ་བླང་བྱ་དང། མི་དགེ་འབྲས་བུ་དང་བཅས་ལ་དོར་བྱར་བྱེད་པའི་སྒོ་ནས་ཚུལ་བཞིན་དུ་སྒྲུབ་པ་ལ་རག་ལས་སོ། །མདོར་ན་སངས་རྒྱས་ཀྱིས་དགེ་མི་དགེ་སོགས་སོགས་ལས་དང་འབྲས་བུ་ཇི་ལྟར་གསུངས་པ་ལ་ཡིད་ཆེས་པའི་སྒོ་ནས་བླང་དོར་བྱེད་དགོས་པའི་དོན་ནོ། །ལྷག་མ་རྗེ་བཙུན་བླ་མས་སོགས་སླའོ། །གསུམ་པ་ལ་གཉིས། དངོས་དང། སྟོབས་བཞི་བསྟེན་ཚུལ་ལོ། །དང་པོ་ནི། ལམ་མཆོག་སྒྲུབ་ལ་སོགས་ཏེ། ཁྱད་པར་དུ་ལམ་མཆོག་རྣམ་མཁྱེན་སྒྲུབ་པ་ལ་ཚེ་རིང་བ་ཁ་དོག་བཟང་བ་སོགས་རྣམ་སྨིན་གྱི་རྒྱུ་བརྒྱད་སོགས་མཚན་ཉིད་ཚང་བའི་རྟེན་ཅི་རིགས་པ་མ་རྙིད་ཀྱི་བར་དུ་ལམ་སྒྲུབ་ཀྱང་ས་ཆོད་མི་འོང་བས། དེ་འོང་བའི་ཆེད་དུ་རྟེན་མཚན་ཉིད་ཚང་བ་དེ་ཡི་མ་ཚང་མེད་པའི་རྒྱུ་སྒྲུབ་པ་ལ་བསླབ་པར་བྱའོ། །རྒྱུ་ནི་གཞན་གྱི་སྲོག་སྐྱོབ་པ། རྟེན་ལ་མར་མེ་འབུལ་བ་སོགས་མང་ངོ། །དེའི་ཚེ་ལས་འབྲས་ལ་བླང་དོར་ཅི་ཐུབ་བྱས་ཀྱང། བག་མེད་པ་དང་ཉོན་མོངས་མང་བའི་དབང་གིས་ཉེས་པ་འབྱུང་བས་སོ། །གཉིས་པ་ནི། སྒོ་གསུམ་སྡིག་ལྟུང་སོགས་ཏེ། རང་ཅག་ནི། ལུས་ངག་ཡིད་ཀྱི་སྒོ་གསུམ་ལས་ཉོན་སྤྱི་དང་ཁྱད་པར་རང་བཞིན་གྱི་ཁ་ན་མ་ཐོ་བའི་སྡིག་པ་དང། བཅས་པའི་ཁ་ན་མ་ཐོ་བའི་ལྟུང་བའི་དྲི་མས་སྤགས་པ་འདི་ནི། ལས་དང་ཉོན་མོངས་པའི་སྒྲིབ་པ་གཉིས་ལས་ལྷག་པར་ལས་ཀྱི་སྒྲིབ་པ་སྦྱོང་བ་གནད་ཆེ་བས་ན། དུས་རྒྱུན་ཏུ་རྟེན་གྱི་སྟོབས་དང། སུན་འབྱིན་པའི་སྟོབས། སླར་ལྡོག་པའི་སྟོབས། གཉེན་པོ་ཀུན་ཏུ་སྤྱོད་པའི་སྟོབས་ཏེ་སྟོབས་བཞིའི་བཤགས་སྡོམ་ཚང་བར་བསྟེན་པར་གཅེས་སོ། །དྲི་མས་སྤགས་པ་འདི་ཞེས་གཞུང་མང་པོར་འབྱུང་བ་ལྟར་བྱས་ལ། ཆོས་སྤྱོད་དུ་ནི་འདིས་ཞེས་འབྱུང་ངོ། །སྟོབས་བཞིའི་ངོས་འཛིན་དང་བཤགས་ཚུལ་རྒྱས་པར་གཞན་དུ་ཤེས་པར་བྱའོ། །རྗེ་བཙུན་བླ་མས་སོགས་སྔར་བཞིན་ནོ། །གཉིས་པ་སྐྱེས་བུ་འབྲིང་དང་ཐུན་མོང་བའི་ལམ་ཉམས་སུ་ལེན་ཚུལ་གཉིས། སྡུག་ཀུན་གཉིས་ཀྱི་ཉེས་དམིགས་པ་བསམ་དགོས་པའི་རྒྱུ་མཚན་དང། བསམ་ཚུལ་དངོས་སོ། །དང་པོ་ནི། སྡུག་བདེན་གཉེས་དམིགས་སོགས་ཏེ། འཁོར་བ་ལས་ཐར་བར་འདོད་པས་ཐོག་མར་སྡུག་ཀུན་གཉིས་ཀྱི་ཉེས་དམིགས་བསམ་དགོས་ཏེ། ངན་སོང་གསུམ་གྱི་སྡུག་བསྔལ་དང་བདེ་འགྲོའི་སྡུག་བསྔལ་སྤྱི་དང་བྱེ་བྲག་སྐྱེ་རྒན་འཆི་སོགས་སྡུག་བསྔལ་བདེན་པས་བསྡུས་པའི་འཁོར་བའི་ཉེས་དམིགས་བསམ་པ་ལ་འབད་ན་ནི། སྡུག་བསྔལ་ལས་ཐར་འདོད་མེད་པར་ཐར་པ་དོན་གཉེར་གྱི་བློ་ཇི་ལྟ་བ་བཞིན་མི་སྐྱེ་ཞིང། འདོད་ཆགས་སོགས་ཀྱི་ངོ་བོ་སྐྱེ་ཚུལ་ཇི་ལྟར་ཡིན་པའི་ཁྱད་པར་ཤེས་པའི་སྒོ་ནས་ཀུན་འབྱུང་བདེན་པ་འཁོར་བའི་འཇུག་རིམ་ཏུ་འགྲོ་ཚུལ་མ་བསམས་ན། འབེན་མ་མཐོང་བར་མདའ་འཕན་པ་བཞིན་འཁོར་བའི་རྩ་བ་གཅོད་ཚུལ་མི་ཤེས་པས་སོ། །གཉིས་པ་ལ་གཉིས། སྡུག་བསྔལ་འཁོར་བའི་ཉེས་དམིགས་དང། ཀུན་འབྱུང་གི་འཁོར་བའི་འཇུག་རིམ་བསམ་པའོ། །དང་པོ་ནི། སྲིད་ལས་ངེས་འབྱུང་སོགས་ཏེ། འཁོར་བའི་གནས་སུ་སྐྱེ་རྒན་འཆི་བཞི་དང་མྱ་ངན་དང་སྨྲེ་སྔགས་འདོན་པ་ཡིད་མི་བདེ་བ་དང་ཚ་གྲང་འཐབ་རྩོད་༄༅། །སོགས་ཀྱི་སྡུག་བསྔལ་རྣམས་མཐོང་ནས་སྲིད་པ་འཁོར་བ་ལས་ངེས་པར་འབྱུང་འདོད་ཀྱི་སྐྱོ་ཤས་དྲག་པོ་བསྟེན་པ་དང། འཆད་འགྱུར་ལྟར་སྡུག་བསྔལ་དེ་བསྐྱེད་པའི་རྒྱུ་ཡང་ཤེས་པར་བྱའོ། །གཉིས་པ་ནི། འཁོར་བར་གང་གིས་སོགས་ཏེ། འཁོར་བར་འཆིང་བ་བྱེད་གང་གིས་བཙངས་?་བ་ཤེས་པ་གཅེས་པ་སྟེ་གལ་ཆེའོ། །འཆིང་བྱེད་ལ་ལས་ཉོན་གཉིས་ཏེ་ཁྱད་པར་ཕྱེན་མང་ཡང། དོན་བསྒྲིལ་ན་སོ་སྐྱེའི་རྒྱུད་ཀྱི་འདོད་ཆགས་ཁོང་ཁྲོ་ང་རྒྱལ་མ་རིག་པ་ཐེ་ཚོམ་ཉོན་མོངས་ཅན་ཕྲག་དོག་སོགས་ཉོན་མོངས་ཐམས་ཅད་ནི་ཉོན་མོངས་པའི་ཀུན་འབྱུང་བདེན་པ་ཡིན་ལ། ཉོན་མོངས་དེས་ཀུན་ནས་བསླངས་པའི་ལས་དགེ་མི་དགེ་གང་ཡིན་ཀྱང་ལས་ཀྱི་ཀུན་འབྱུང་བདེན་པ་ཡིན་ནོ། །དེ་ཡང་གཙོ་བོ་མ་རིག་པའི་དབང་གིས་ལས་བསགས་ལས་ཀྱི་དབང་གིས་འཁོར་བར་འཇུག་པའི་རྟེན་འབྲེལ་ཡན་ལག་བཅུ་གཉིས་ཀྱི་རིམ་པ་ལེགས་པར་ཤེས་ཤིང། མདོར་ན་ལས་དང་ཉོན་མོངས་གཉིས་ཀྱི་དབང་གིས་འཁོར་བར་འཁོར་ཚུལ་རྣམས་བླ་མའི་མན་ངག་ལ་བརྟེན་ནས་ཤེས་པ་ནི། ཀུན་འབྱུང་འཁོར་བའི་འཇུག་རིམ་ཤེས་པའོ། །འཁོར་བའི་རྩ་བ་གཅོད་ཚུལ་ནི་བདག་མེད་རྟོགས་པའི་ཤེས་རབ་ཀྱིས་མ་རིག་པའི་ཞེན་ཡུལ་ལེགས་པར་སུན་ཕྱུང་བའི་ཚུལ་གྱིས་གོམས་པར་བྱས་པའི་མཐར་མ་རིག་པ་སྤངས་པ་ན་འཁོར་བའི་རྩ་བ་བཅད་ཅིང། ཉོན་མོངས་གཞན་ཐམས་ཅད་ཀྱང་ཞོར་ལ་སྦྱོང་བར་འགྱུར་བ་ཡིན་ནོ། །སྡུག་ཀུན་གཉིས་ཀྱི་ནང་ནས་བུན་འབྱུང་སྤངས་ན་ཤིང་རྩ་བ་བཅད་ན་བསྐམས་འགྲོ་བ་ལྟར་སྡུག་བསྔལ་ཡང་སྤོང་བར་འགྱུར་པ་གཞུང་ཆེན་པོ་རྣམས་ཀྱི་དགོངས་པ་ཡིན་མོད། དེང་སང་གི་ལས་དང་པོ་བས་ནི། ཉོན་མོངས་རྩ་བ་ནས་སྤོང་མི་ནུས་པས། སྟོབས་བཞིའི་བཤགས་པ་ལ་བརྟེན་ནས་ཀུན་འབྱུང་གི་ནང་ཚན་སྡིག་ལྟུང་རྣམས་སྤོང་བ་གནད་ཆེའོ། །ཁྱད་པར་དུ་ཚུལ་ཁྲིམས་ཀྱི་བསླབ་པ་སོགས་གསུམ་ལ་འབད་དགོས་ལ། ཚུལ་ཁྲིམས་ཀྱི་བསླབ་པ་ནི་གཙོ་བོའི་དབང་དུ་བྱས་ན་མི་དགེ་བ་བཅུ་སྤོང་བའི་སྤོང་སེམས་ཡིན་ལ། ཏིང་ངེ་འཛིན་གྱི་བསླབ་པའི་གཙོ་བོ་ནི་ཞི་གནས་ཡིན་མོད། དེར་མ་ངེས་ཏེ་དགེ་བའམ་ཡང་དག་པའི་དམིགས་པ་ལ་རིང་ཐུད་གང་ཡིན་ཡང་རྩེ་གཅིག་ཏུ་གནས་པའི་སེམས་པ་རྣམས་ཏིང་ངེ་འཛིན་གྱི་བསླབ་པས་བསྡུས་སོ། །ཤེས་རབ་ཀྱི་བསླབ་པའི་གཙོ་བོ་ནི་ལྷག་མཐོང་ཡིན་ཀྱང། དེར་མ་ངེས་ཏེ། བདག་མེད་ཕྲ་རགས་ཅི་རིགས་ལ་དཔྱོད་པའི་རིགས་པ་དོན་མཐུན་རྣམས་དང། ཆོས་ཀྱི་ཚིག་དོན་གཞན་ལའང་ལེགས་པར་དཔྱོད་པའི་ཤེས་རབ་རྣམས་ནི་ཤེས་རབ་ཀྱི་བསླབ་པར་འདུས་སོ། །གསུམ་པ་སྐྱེས་བུ་ཆེན་པོའི་ལམ་གྱི་རིམ་པ་ལ་གཉིས། སྤྱོད་པའི་རྟེན་སེམས་བསྐྱེད་དང། སྤྱོད་པ་ལ་བསླབ་ཚུལ་ལོ། །དང་པོ་ལ་གཉིས་ལས། དང་པོ་ཁྱད་ཆོས་སུམ་ཕན་ཡོན་ནི། སེམས་བསྐྱེད་ཐེག་མཆོག་སོགས་བཞི་སྟེ། ཐེག་ཆེན་སེམས་བསྐྱེད་ནི། ཐེག་མཆོག་ལས་ཀྱི་གཞུང་ཤིང་ལྟ་བུ་སྟེ། སྦྱིན་དྲུག་སོགས་རླབས་ཆེན་སྤྱོད་པ་རྣམས་ཀྱི་སྐྱེ་བ་དང་གནས་པ་དང་འཕེལ་བའི་གཞི་དང་རྟེན་ཡིན་ལ། བསོད་ནམས་དང་ཡེ་ཤེས་ཀྱི་ཚོགས་གཉིས་ཀུན་རྫོགས་བྱང་དུ་སྒྱུར་བར་བྱེད་པ་ལ། ལྕགས་ཁམས་གསེར་དུ་སྒྱུར་བར་བྱེད་པའི་གསེར་འགྱུར་གྱི་རྩི་ལྟ་བུ་དང། རབ་འབྱམས་དགེ་ཚོགས་མང་པོ་སྡུད་པའི་བསོད་ནམས་ཀྱི་གཏེར་ཆེན་པོའོ། །གཞུང་འདིས་སེམས་དེ་བསྐྱེད་དགོས་པའི་རྒྱུ་མཚན་ཡང་བསྟན་ལ། གཞི་དང་རྟེན་ཞེས་པས་ཐེག་ཆེན་སེམས་བསྐྱེད་དེ། ཐེག་ཆེན་ལམ་གྱི་འཇུག་སྒོར་ཡང་བསྟན་ཏོ༑ ༑གཉིས་པ་བསྐྱེད་ཚུལ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་གཉིས་ཏེ། ཐེག་ཆེན་སེམས་བསྐྱེད་ཀྱི་ཁྱད་ཆོས་སམ་ཕན་ཡོན་དེ་ལྟར་ཤེས་ནས་རྒྱལ་བའི་སྲས་བྱང་ཆུབ་སེམས་དཔའ་གཞན་དོན་དུ་འཇུག་པར་ནུས་པ་དཔའ་བོ་རྣམས་ཀྱིས་ནི། སེམས་ཅན་ཐམས་ཅད་ཀྱི་དོན་དུ་སངས་རྒྱས་ཐོབ་འདོད་ཀྱི་རིན་ཅེན་སེམས་མཆོག་དེ་ཉམས་ཡོན་གྱི་གཙོ་བོའམ་ཐུགས་དམ་གྱི་མཐིལ་དུ་འཛིན་པར་མཛད་པ་ལྟར་བྱའོ། །རྗེ་བཙུན་སོགས་སྔར་བཞིན་ནོ། །གཉིས་པ་སྤྱོད་པ་ལ་བསླབ་ཚུལ་ལ་དྲུག །སྦྱིན་པ། ཚུལ་ཁྲིམས། བཟོད་པ། བརྩོན་འགྲུས། བསམ་གཏན། ཤེས་རབ་པོ། །དང་པོ་ལ་གཉིས་ལས་དང་པོ་ཁྱད་ཆོས་སམ་ཕན་ཡོན་ནི། སྦྱིན་པ་འགྲོ་བའི་སོགས་བཞི་སྟེ། ༄༅། །ལུས་ལོངས་སྤྱོད་སོགས་གཞན་ལ་གཏོང་བའི་སྦྱིན་པ་ནི། སློང་བ་པོའི་འགྲོ་བའི་རེ་བ་སྐོང་བར་བྱེད་པའི་ཡིད་བཞིན་གྱི་ནོར་བུ་ལྟ་བུ་དང། དངོས་པོ་ལ་ཆགས་ཞེན་གཅོད་པའི་ཕྱིར་ན་སེར་སྣའི་མདུད་པ་གཅོད་པའི་མཚོན་ཆ་མཆོག་ཡིན་ལ། འགྲོ་བའི་དོན་ལ་མ་ཞུམ་ཞིང་སྙིང་སྟོབས་སྐྱེད་པའི་རྒྱལ་སྲས་ཀྱི་སྤྱོད་པ་དང། གཏོང་བ་པོའི་སྙན་པའི་གྲགས་པ་ཕྱོགས་བཅུར་སྒྲོགས་པའི་གཞིའོ། །གཉིས་པ་གཏོང་ཚུལ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས། ཚུལ་དེ་ལྟར་ཤེས་ནས་རང་གི་ལུས་དང་ཟས་ནོར་སོགས་ལོངས་སྤྱོད་དང། བདེར་གེ་?་ཐམས་ཅད་གཞན་ལ་ཡོངས་སུ་གཏོང་བའི་ལམ་བཟང་པོ་མཁས་པས་བསྟེན་པར་བྱའོ། །ལོངས་སྤྱོད་ཞེས་པས་ཟང་ཟིང་གི་སྦྱིན་པ་དང། དགེ་ཞེས་པས་ཀྱང་ཆོས་ཀྱི་སྦྱིན་པའང་བསྡུས་ཤིང། མི་འཇིགས་སྦྱིན་པ་ནི་ཤུགས་ཀྱིས་ཐོབ་པོ། །གཉིས་པ་ལ་སྔར་ལྟར་གཉིས་ལས། དང་པོ་ནི། ཚུལ་ཁྲིམས་ཉེས་སྤྱད་སོགས་བཞི་སྟེ། ཚུལ་ཁྲིམས་ལ་ཉེས་སྤྱོད་སྡོམ་པ། དགེ་བ་ཆོས་སྡུད་སེམས་ཅན་དོན་བྱེད་གསུམ་ལས་གཙོ་བོར་ཉེས་སྤྱོད་སྡོམ་པའི་དབང་དུ་བྱས་ན། ཚུལ་ཁྲིམས་དེ་ནི། སྒོ་གསུམ་གྱི་ཉེས་སྤྱོད་ཀྱི་དྲི་མ་འཁྲུད་པའི་ཆུ་ལྟ་བུ་དང། ཉོན་མོངས་པའི་ཚ་གདུང་སེལ་བར་བྱེད་པའི་ཟླ་བའི་འོད་ལྟ་བུ་དང། སྲུང་བ་པོའི་རྟེན་གྱི་གང་ཟག་དེ་སྐྱེ་རྒུའི་དབུས་ན་བི་རྒྱལ་ལྷུན་པོ་ལྟ་བུར་བརྗིད་ཆེ་ཞིང། དེ་ལྟ་བུའི་སྟོབས་ཀྱིས་གཞན་དག་གིས་རང་ལ་འཚེ་བའི་བསྡིགས་པ་མེད་པར་འགྲོ་བ་ཀུན་འདུད་པར་བྱེད་དོ། །གཉིས་པ་སྲུང་ཚུལ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་ཏེ། ཚུལ་དེ་ལྟར་ཤེས་ནས་ཡང་དག་པར་བླངས་པའི་ཚུལ་ཁྲིམས་ནི་སྔོན་གྱི་དམ་པ་རྣམས་ཀྱིས་མིག་བཞིན་དུ་སྲུང་བར་མཛད་པ་ལྟར་བྱའོ། །གསུམ་པ་ལ་གཉིས་ལས་དང་པོ་ནི། བཟོད་པ་སྟོབས་ལྡན་སོགས་བཞི་སྟེ། བཟོད་པ་ལ་སྡུག་བསྔལ་དང་ཡིན། གནོད་པ་ལ་ཇི་མི་སྙམ་པ། ཆོས་ལ་ངེས་ཐོབ་ཀྱི་བཟོད་པ་གསུམ་ལས། འདིར་དངོས་བསྟན་པར་བའི་དབང་དུ་བྱས་ནས་གནོད་པ་ལ་བཟོད་པའི་བཟོད་པ་ནི། གནོད་པའི་ཁུར་སྐྱེར་ནུས་པའི་བྱང་སེམས་སྟོབས་ལྡན་རྣམས་ལ་མཛེས་བྱེད་དུ་གྱུར་པའི་རྒྱན་གྱི་མཆོག་ཡིན་ལ། ཉོན་མོངས་པའི་མེས་གདུང་བའི་གཉེན་པོ་ལ་དཀའ་ཐུབ་དུ་མ་ཡོད་ཀྱང། དཀའ་ཐུབ་ཀུན་གྱི་ཕུལ་ཏེ་མཆོག་དང་ཞེ་སྡང་ལག་འགྲོ་སྟེ་སྤྲུལ་གྱིས་མཚོན་པར་བྱེད་པ་སླུའི་དགྲ་ལ་མང་དུ་ཡོད་ཀྱང། ནམ་མཁའི་ལྡིང་སྟེ་ཁྱུད་བཞིན་དང། ཕ་རོལ་གྱིས་ཚིག་རྩུབ་ཀྱི་མཚོན་འཕེན་པ་ལ་དེས་མི་མིག་པའི་སྲ་བའི་གོ་ཆ་སྲབ་ལྟ་བུ་ཡིན་ནོ། །གཉིས་པ་དེ་བསྒོམ་པ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་ཏེ། ཚུལ་དེ་ལྟར་ཤེས་ནས་བཟོད་པའི་མཆོག་སྔོན་གྱི་དམ་པས་རང་གི་གོ་ཆ་ལ་ཐབས་རྣམས་སྣ་ཚོགས་ཚུལ་གྱིས་རྒྱུད་ལ་གོམས་པར་མཛད་པ་ལྟར་བྱའོ། །བཞི་པ་ལ་གཉིས་ལས། དང་པོ་ནི། བརྩོན་འགྲུས་ལ་གསུམ་ལས། དང་པོ་ནི། མི་བཟློག་བརྟན་པའི་སོགས་བཞི་སྟེ། གཙོ་བོར་གོ་ཆའི་བརྩོན་འགྲུས་ཀྱི་དབང་དུ་བྱས་ན། ཕྱིར་མི་བཟློག་ཅིང་བརྟན་པའི་བརྩོན་འགྲུས་ཀྱི་གོ་ཆ་བགོས་ན། ལུང་དང་རྟོགས་པའི་ཡོན་ཏན་ཐམས་ཅད་ཡར་དོའི་ཟླ་བ་བཞིན་གོང་དུ་འཕེལ་ལ། སྤྱོད་ལམ་ཐམས་ཅད་ལ་རྩོམ་པ་དོན་དང་ལྡན་པར་འགྱུར་ཞིང། རང་གིས་བྱ་བ་གང་བརྩམས་པའི་ལས་ཀྱི་མཐའ་རྣམས་ཡིད་ལ་བསམས་པ་བཞིན་དུ་འགྲུབ་པར་འགྱུར་རོ། །གཉིས་པ་རྩོམ་ཚུལ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་ཏེ། ཇི་སྐད་བཤད་པ་དེ་ལྟར་ཤེས་ནས་བྱི་བཤོལ་གྱི་ལེ་ལོ་ཀུན་སེལ་བར་བྱེད་པའི་རླབས་ཆེན་གྱི་བརྩོན་འགྲུས་སྔོན་རྒྱལ་སྲས་རྣམས་ཀྱིས་བརྩམས་པར་བྱས་པ་ལྟར་འཇུག་གོ །ལྔ་པ་ལ་གཉིས་ལས། དང་པོ་ནི། བསམ་གཏན་སེམས་ལ་སོགས་བཞི་སྟེ། དམིགས་པ་རྩ་གཅིག་པའི་བསམ་གཏན་ནི། རང་གི་སེམས་ལ་ཇི་ལྟར་འདོད་པ་བཞིན་དབང་བསྒྱུར་བའི་རྒྱལ་པོ་ལྟ་བྷ་སྟེ། དེ་ཡང་དམིགས་པ་ལ་བཞག་ན་ནི། བྱིང་རྒོད་ཀྱིས་གཡོ་བ་མེད་པར་དེ་ཡི་དབང་པོ་བཞིན་དུ་གནས་ལ། བཞག་པ་དེ་བཏང་ན་ཡང་དགེ་བའི་དམིགས་པ་སྟེ་ཡུལ་ཀུན་ལ་འཇུག་ཅིང། རང་གི་ལུས་སེམས་དགེ་བའི་ཕྱོགས་ལ་བགོལ་ཅུ་རུང་བའི་ལས་སུ་རུང་བའི་ཤིན་སྦྱངས་ཀྱི་བདེ་ཆེན་འདྲེན་ནོ། །དམིགས་པ་ལ་རྩེ་གཅིག་ཏུ་བཞག་པའི་སྟོབས་ཀྱིས། ལུས་སེམས་ཤིན་སྦྱངས་ཀྱི་བདེ་བ་ཁྱད་པར་ཅན་འདྲེན་༄༅། །ཚུལ་ནི་རྗེའི་ལམ་རིམ་ལས་ཤེས་པར་བྱའོ། །གཉིས་པ་བསྒོམ་ཚུལ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་ཏེ། ཚུལ་དེ་ལྟར་ཤེས་ནས་སྔོན་བྱོན་པའི་རྣལ་འབྱོར་གྱི་དབང་པོ་རྣམས་ཀྱིས་རྣམ་གཡེང་གི་དགྲ་འཇོམས་ཤིང། དམིགས་པ་ལ་རྩེ་གཅིག་པའི་ཏིང་ངེ་འཛིན་དུས་རྒྱུན་ཏུ་བསྟེན་པ་ལྟར་བསླབ་པོ། །འདི་ནི་གཙོ་བོ་ཞིག་གནས་བསྒོམས་པའི་དབང་དུ་བྱས་ལ། རྗེ་བཙུན་སོགས་སྔར་བཞིན་ནོ། །དྲུག་པ་ལ་གསུམ། ཤེས་རབ་བསྒོམ་ཚུལ། ཞི་ལྷག་ཟུང་དུ་འབྲེལ་ཚུལ། མཉམ་རྗེས་གཉིས་ཀ་སྐྱོང་ཚུལ་ཡན་ལག་དང་བཅས་པའོ། །དང་པོ་ལ་ཁྱད་ཆོས་སམ་ཕན་ཡོན་དང། བསྒོམ་ཚུལ་གཉིས་ལས། དང་པོ་ནི། ཤེས་རབ་ཟབ་མོའི་དེ་ཉིད་སོགས་བཞི་སྟེ། ཤེས་རབ་ལ་ཇི་ལྟ་བ་རྟོགས་པའི་ཤེས་རབ་སོགས་གསུམ་ཡོད་ཀྱང། འདིར་གཙོ་བོ་ནི་དང་པོའི་དབང་དུ་བྱས་ནས། ཟབ་དོན་རྟོགས་པའི་ཤེས་རབ་ནི། ཆོས་རྣམས་ཀྱི་ཟབ་མོའི་དེ་ཁོ་ན་ཉིད་ལྟ་བར་བྱེད་པའི་མིག་ལྟ་བུ་དང། སྲིད་པ་འཁོར་བའི་རྩ་བ་མ་རིག་པ་དྲུང་ནས་འབྱིན་པའི་ལམ་དུ་འགྱུར་ཞིང། ཡུམ་གྱི་མདོ་སོགས་རྒྱལ་བའི་གསུང་རབ་ཀུན་ལས་བསྔགས་པའི་ཡོན་ཏན་གྱི་གཏེར་ཆེན་པོ་དང། རྨོངས་པའི་གཏི་མུག་གི་མུན་པ་སེལ་བར་བྱེད་པའི་སྒྲོན་མེ་མཆོག་ཏུ་གྲགས་པ་ཡིན་ནོ། །གཉིས་པ་ནི། དེ་ལྟར་ཤེས་ནས་སོགས་ཏེ། ཚུལ་དེ་ལྟར་ཤེས་ནས་ཐར་པ་འདོད་པའི་མཁས་པ་ཡིས་ལམ་དེ་ཉིད་འབད་པ་དུ་མས་རང་རྒྱུད་ལ་བསྐྱེད་པར་མཛད་དོ། །གོང་དུ་བརྗོད་པའི་ཕྱིན་དྲུག་སོ་སོའི་ངོ་བོ། དབྱེ་བ་སོགས་རྒྱས་པར་ནི་འདིར་མང་དུ་དོགས་པས་ལམ་རིམ་ཆེན་མོ་སོགས་ལས་ཤེས་པར་བྱ་ཞིང། དོན་བསྡུས་པ་ཞིག་ཁོ་བོས་ལམ་རིམ་དམར་འཁྲིད་དུ་བཤད་ཟིན་ཏོ། །གཉིས་པ་ལ་གཉིས་ལས། ཞི་ལྷག་ཟུང་འབྲལ་བསྒོམ་པའི་དགོས་པ་དང། བསྒོམ་ཚུལ་དངོས་སོ། །དང་པོ་ནི། རྩེ་གཅིག་བསམ་གཏན་སོགས་བརྒྱད་དེ། དམིགས་པ་ལ་རྩེ་གཅིག་པའི་བསམ་གཏན་ཙམ་ལ་འཁོར་བ་ཡི་རྩ་བ་མ་རིག་པ་རྩད་ནས་གཅོད་པའི་ནུས་པ་མ་མཐོང་ཞིང། དེ་ལྟ་ནའང་དམིགས་པ་ལ་རྩེ་གཅིག་ཏུ་གནས་པའི་ཞི་གནས་ཀྱི་ལམ་དང་བྲལ་བའི་ཐོས་བསམ་གྱི་ཤེས་རབ་ཀྱིས་ཀྱང། བདེན་མེད་སོགས་ཀྱི་དཔྱད་དོན་ལ་ཇི་ཙམ་དཔྱད་ཀྱང་ཉོན་མོངས་རྩད་ནས་མི་ལྡོག་པས་ན། སྔོན་གྱི་དམ་པས་དེ་ཁོ་ན་ཉིད་ཀྱི་དོན་ཚུལ་བཞིན་དུ་གོ་བའི་ཡིན་ལུགས་ཀྱི་དོན་ལ་ཕུ་ཐག་ཆོད་པའི་ཤེས་རབ་དེ། དེ་ཁོ་ན་ཉིད་ལ་བྱིང་རྒོད་ཀྱི་རྣམ་རྟོག་གིས་གཡོ་བ་མེད་པའི་ཞི་གནས་ཀྱི་རྟ་ལ་བསྐྱོན་ནས་ནི། རྟག་ཆད་ཀྱི་མཐའ་དང་བྲལ་བའི་དབུ་མའི་རིགས་པ་གཅིག་དུ་བྲལ་ལ་སོགས་པའི་མཚོན་རྣོན་གྱིས་ནི། རྟག་པ་དང་ཆད་པའི་མཐར་འཛིན་པའི་མཐར་འཛིན་གྱི་དམིགས་པའི་གཏད་སོ་བདེན་གྲུབ་ཏུ་བཟུང་བ་ལྟར་བདེན་འཛིན་གྱི་ཞེན་ཡུལ་ཐམས་ཅད་འཇིག་པར་བྱེད་པའི་རིགས་པ་ཡང་དག་པའི་དོན་ལ་ཚུལ་བཞིན་དུ་དཔྱོད་པའི་ཡངས་པའི་ཤེས་རབ་ཀྱིས་དེ་ཁོ་ན་ཉིད་རྟོགས་པའི་བློ་གྲོས་རྒྱས་པར་མཛད་དོ། །གཉིས་པ་ནི། རྩེ་གཅིག་གོམས་པས་སོགས་དྲུག་སྟེ། རང་གི་དམིགས་པ་གང་རུང་ཅིག་ལ་དམིགས་ནས་མི་དཔྱོད་པར་རྩེ་གཅིག་ཏུ་གནས་པ་གོམས་པས་ཏིང་ངེ་འཛིན་འགྲུབ་པར་འགྱུར་བ་ནི་གཞིར་བཅས་ཤིང་སྨོས་པ་ཅི་འཚལ་ཏེ་བརྗོད་མི་དགོས་ཀྱི། དམིགས་པ་ལ་རྩེ་གཅིག་ཏུ་གནས་བཞིན་པའི་ངང་ནས་དོན་ལ་ཚུལ་བཞིན་དུ་དཔྱོད་པ་ཡི་སོ་སོར་རྟོགས་པ་ཡི་ཤེས་རབ་ཀྱིས་ཀྱང་ཡིན་ལུགས་ཀྱི་དོན་ལ་གཡོ་བ་མེད་པར་ཤིན་ཏུ་བརྟན་པར་གནས་པ་ཡི་ཏིང་འཛིན་སྐྱེད་པར་ནུས་པ་བློས་མཐོང་ནས་ཞི་ལྷག་གཉིས་ལས་ཕྱོགས་རེ་ཙམ་མ་ཡིན་པར་ཟུང་འབྲེལ་སྒྲུབ་པར་བྱེད་པ་ལ་བརྩོན་པ་རྣམས་ཡ་མཚན་ནོ་སྟེ་ངོ་མཚར་ཆེའོ། །རང་གི་དམིགས་པ་ལ་གཏོད་དེ་རྩེ་གཅིག་ཏུ་གནས་པ་གོམས་ཀྱང་ཞི་གནས་འགྲུབ་མོད་ལྷག་མཐོང་འགྲུབ་པ་ལ་ནི། དེ་ཙམ་གྱིས་མཆོག་པར་རྩེ་གཅིག་ཏུ་གནས་བཞིན་པའི་དད་ནས་དཔྱད་སྟོབས་ཀྱིས་ཤིན་སྦྱངས་ཀྱི་བདེ་བ་འདྲེན་ནུས་པ་དང་དགོས་པའི་དོན་ནོ། །གསུམ་པ་ནི། མཉམ་གཞག་ནམ་མཁའ་སོགས་དྲུག་སྟེ། དེ་ཁོ་ན་ཉིད་ཀྱི་དོན་ལ་མཉམ་པར་བཞག་པའི་མཉམ་གཞག་དགག་བྱ་བདེན་གྲུབ་བཀག་པའི་ནམ་མཁའ་ལྟ་བུའི་སྟོང་ཉིད་སྒོམ་པ་དང། དེ་ལས་ཡངས་ནས་རྗེས་ཐོབ་ཏུ་བདེན་པར་སྣང་༄༅། །ཡང་དེར་མ་གྲུབ་པའི་སྒྱུ་མ་ལྟ་བུའི་སྟོང་པ་ཉིད་གཉིས་པོ་བསྒོམས་ནས། སྦྱིན་པ་ལ་སོགས་པ་ཐབས་དང། སྟོང་ཉིད་རྟོགས་པའི་ཤེས་རབ་གཉིས་ཀ་ཟུང་དུ་འབྲེལ་བ་ཡི་སྒོ་ནས་རྒྱལ་སྲས་སྤྱོད་པ་རྒྱ་མཚོའི་ཕ་རོལ་ཏུ་འགྲོ་བ་ནི་རྒྱལ་དང་དེ་སྲས་རྣམས་ཀྱིས་བསྔགས་སོ། །རྗེས་ཐོབ་ཏུ་སྒྱུ་མ་ལྟ་བུ་བསྒོམས་པ་ནི་སྟོང་ཉིད་སྒོམ་པ་དངོས་མིན་ནའང་ཕྱོགས་མཐུན་ཡིན་པས་དེང་བཏགས་པ་ཙམ་མོ། །ཇི་སྐད་བཤད་པ་དེ་ལྟར་རྟོགས་ནས་ཐབས་ཤེས་ཕྱོགས་རེ་བའི་ལམ་གྱིས་ནི་ཚིམ་པ་མེད་པ་སྐལ་བཟང་རྣམས་ཀྱི་ལུགས་སོ། །གཉིས་པ་ཐུན་ཐོར་མ་ཡིན་པ་སྔགས་ཀྱི་ཐེག་པ་ལ་བསླ་ཚུལ་ནི། དེ་ལྟར་རྒྱུ་དང་འབྲས་བུའི་ཐེག་ཆེན་གྱི། །སོགས་དྲུག་སྟེ། སྔར་བཤད་པ་དེ་ལྟར་ལམ་བགྲོད་ཚུལ་དེ་ནི། རྒྱུ་ཕ་རོལ་ཏུ་ཕྱིན་པའི་ཐེག་པ་དང། འབྲས་བུ་སྔགས་ཀྱི་ཐེག་པའི་ཐེག་ཆེན་གྱི་ལམ་མཆོག་གཉིས་གར་དགོས་པའི་ཐུན་མོང་བའི་ལམ་ཡིན་ལ། དེ་ཇི་ལྟ་བ་བཞིན་རྒྱུད་ལ་བསྐྱེད་དེ༑ དེ་ནས་སྣགས་ལ་མཁས་པའི་དེད་དཔོན་གྱི་མགོན་རྡོ་རྗེ་སློབ་དཔོན་ལ་བརྟེན་ནས་རྒྱུད་སྡེའི་རྒྱ་མྶོ་ཆེན་པོར་ཞུགས་ནས་མདོ་སྔགས་གཉིས་ཀྱི་ལམ་གྱི་གནད་ཡོངས་སུ་རྫོགས་པའི་མན་ངག་བསྟེན་པའི་གང་ཟག་དེས། དལ་འབྱོར་གྱི་ལུས་རྟེན་འདི་ཐོབ་པ་དོན་དང་ལྡན་པར་བྱས་པ་ཡིན་ནོ༑ ༑རྗེ་བཙུན་སོགས་སྔར་ལྟར་རོ། །གསུམ་པ་མཇུག་ནི། རང་གི་ཡིད་ལ་སོགས་དྲུག་སྟེ། རང་གི་ཡིད་ལ་གོམས་པར་བྱ་བའི་ཕྱིར་དང། དོན་གཉེར་ཅན་གྱི་སྐལ་བཟང་གཞན་ལའང་ཕན་པར་བྱ་བའི་ཕྱིར་དུ། རྒྱལ་བ་དགྱེས་པའི་ཡོངས་སུ་རྫོགས་པའི་ལམ་མདོར་བསྡུས་ནས། གོ་སླའི་ངག་གིས་བཤད་པའི་དགེ་བ་དེས་ནི། ཕྱོགས་རེ་ཙམ་མ་ཡིན་པར་འགྲོ་བ་ཀུན་ཀྱང་རྣམ་དག་ལམ་བཟང་འདི་དང། འབྲལ་བ་མེད་པར་གྱུར་ཅིག་ཅེས་ནི་སྨོན་ལམ་འདེབས་སོ། །རྗེ་བཙུན་བླ་མས་སོགས་སམ། རྣལ་འབྱོར་ངས་ཀྱང་སྨོན་ལམ་དེ་ལྟར་བཏབ། །ཐར་འདོད་ཁྱེད་ཀྱང་དེ་བཞིན་འདེབས་འཚལ་ལོ། །ཞེས་པ་གང་བྱས་ཀྱང་དོན་ནི་སླའོ། །འདིའི་འཕྲོར་སྟོན་པ་བླ་ན་མེད་པའི་བསྟན་པ་དང། །ཞེས་སོགས་ཚིགས་བཅད་གཅིག་དང། ང་ཡི་བླ་མ་བློ་བཟང་གྲགས་པ་དེ། །ཞེས་སོགས་ཚིགས་བཅད་གཅིག་དང། དང་པོར་རྒྱ་ཆེན་ཐོས་པ་མང་དུ་བཙལ། །ཞེས་སོགས་ཚིགས་བཅད་གཅིག་སྟེ་ཚིགས་བཅད་གསུམ་འབྱུང་བ་ནི། གཞན་གྱིས་འདོན་ཆ་ལ་ཁ་བཀང་བ་སྟེ། འདིའི་མཇུག་དངོས་མིན་པས་མ་བྲིས་སོ། །ཞེས་བྱང་ཆུབ་ལམ་གྱི་རིམ་པའི་ཉམས་ལེན་གྱི་རྣམ་གཞག་མདོར་བསྡུས་ཏེ་བརྗེད་བྱང་དུ་བྱས་པ་འདི་ནི། མང་དུ་ཐོས་པའི་དགེ་སློང་སྤོང་བ་པ་བློ་བཟང་གྲགས་པའི་དཔལ་གྱིས་འབྲོག་རི་བོ་ཆེ་དགེ་ལྡན་རྣམ་པར་རྒྱལ་བའི་གླིང་དུ་སྦྱར་བའོ། །ཞེས་པ་ནི་སླའོ། །འདིར་བྱང་ཆུབ་ལམ་གྱི་རིམ་པའི་གནད་ཚང་བས། འདོན་པའི་ཚེ་དལ་བུས་དོན་ཡིད་ལ་བཤད་པར་བྱས་ན་ལམ་རིམ་གྱི་དོན་བསྡུས་ནས་ཉམས་སུ་བླངས་པར་འགྱུར་ཡང། ཚིག་ཙམ་ཏོན་པས་ནི་མིན་ནོ། །དེས་ན་འདི་ཁ་ཏོན་བྱེད་པས་ནི། དབྱངས་ཀྱི་སྒྲ་སྙན་པ་ཙམ་ལ་ཆགས་པར་མི་བྱའོ། །འདིར་ཉམས་སུ་ལེན་ཚུལ་རྒྱས་པར་ལམ་རིམ་དང་སྦྱར་ནས་འཆད་ནུས་ནའང་མང་དུ་དོགས་ཤིང། དམིགས་སྐོར་ཞིག་ལོགས་སུ་བྲིས་ཡོད་པས་རྒྱས་པར་མ་བཤད་ལ། ཚིག་དོན་རྫོགས་པར་སྔར་གཞན་གྱིས་བཀྲལ་བ་མ་མཐོང་ཡང་རང་བློས་ཅི་ནུས་ཀྱིས་བཀྲལ་བའོ༎ ༎སྨྲས་པ། རྒྱལ་བའི་གསུང་རབ་ཀུན་གྱི་སྙིང་པོའི་དོན། །ཉམས་སུ་ལེན་པའི་དོམ་པར་རབ་བསྒྲིལ་བ། །བྱང་ཆུབ་བགྲོད་པའི་ལམ་གྱི་རིམ་པ་འདི། །ཐུབ་བསྟན་རྒྱལ་མཚན་རྩེ་ན་ནོར་བུ་བཞིན། །དེ་ཕྱིར་ཚུལ་འདིར་ཉམས་ལེན་ཕྱོགས་ཙམ་ཞིག །བགྱིས་པ་ཡིས་ཀྱང་སྐལ་བར་ལྡན་འགྱུར་ན། །ཇི་སྐད་བཤད་བཞིན་རྫོགས་པར་ཉམས་ལེན་དུ། །བསྟར་༄༅། །བ་གང་ཡིན་ངོ་མཚར་ཅིས་མི་ཆེ། །གཞུང་དོན་ཆ་ཤས་ཡོངས་རྫོགས་ཀུན་དའི་གཉེན། །ཐུབ་བསྟན་ལྷ་ལམ་མཛེས་པར་བྱ་ཕྱིར་དང། །འགྲོ་འདིའི་ཁམས་ཀྱི་སྒྲིབ་མུན་བསལ་བའི་ཆེད། །ལྷག་བསམ་དག་པའི་མགྱོགས་འགྲོའི་ཤུགས་ཀྱིས་དྲངས། །འདིར་འབད་འོ་མཚོ་ལས་འཁྲུངས་རྣམ་དཀར་གྱི། །དགེ་ལེགས་བདུད་རྩིས་དོན་གཉེར་ལྡན་པ་ཡི། །འཆི་མེད་ཡིད་ཀྱི་རེ་བ་ཡོངས་རྫོགས་ཏེ། །གཏན་བདེ་སྲོག་གི་ཉེར་འཚོར་འགྱུར་བར་ཤོག །ཅེས་བྱང་ཆུབ་ལམ་གྱི་རིམ་པའི་བསྡུས་དོན་གྱི་ཚིག་འཕྲེལ་སྙིང་པོ་མདོར་བསྡུས་གསལ་བ་ཞེས་བྱ་བ་འདི་ནི། དད་དང་རྣམ་དཔྱོད་ཟུང་དུ་འབྲེལ་བའི་ཁྱའོ་ལུང་དགའ་བཅུ་པ་བློ་བཟང་བསམ་འགྲུབ་ཀྱིས་བསྐུལ་བའི་ངོར། ཤཱཀྱའི་དགེ་སློང་བཤད་སྒྲུབ་མིང་ཅན་གྱིས་འཕྲལ་དུ་མགྱོགས་ལས་སུ་བྲིས་པ་སྟེ། ཕྱིས་བར་དུ་བརྐོ་བའི་ཚེ་ཡི་གེ་བའང་བསྐུལ་བ་པོ་ཉིད་དོ། །འདིས་ཀྱང་རྒྱལ་བའི་བསྟན་པ་རིན་པོ་ཆེ་ཕྱོགས་ཐམས་ཅད་དུ་དར་ཞིང་རྒྱས་ལ་ཡུན་རིང་དུ་གནས་པར་གྱུར་ཅིག ། ༎དགེའོ། །

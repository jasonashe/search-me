རྒྱ་གར་སྐད་དུ། ཨཱརྱ་སརྦ་ཏ་ཐཱ་ག་ཏོཥྞཱི་ཥ་སི་ཏཱ་ཏ་པ་ཏཱི་ནཱ་མ་ཨ་པ་རཱ་ཛི་ཏ་པྲ་བྱཾ་གི་ར་མ་ཧཱ་བི་དྱཱ་ར་ཛྙཱི། བོད་སྐད་དུ། འཕགས་པ་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པའི་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་ཆེན་མོའོ༎ དཀོན་མཆོག་གསུམ་ལ་ཕྱག་འཚལ་ལོ། །སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་ལ་ཕྱག་འཚལ་ལོ། །འདི་སྐད་བདག་གིས་ཐོས་པ་དུས་གཅིག་ན། བཅོམ་ལྡན་འདས་ལྷའི་ཡུལ་སུམ་ཅུ་རྩ་གསུམ་གྱི་གནས་ཆོས་བཟང་ལྷའི་མདུན་ས་ན་དགེ་སློང་གི་དགེ་འདུན་ཆེན་པོ་དང། བྱང་ཆུབ་སེམས་དཔའི་དགེ་འདུན་ཆེན་པོ་དང། ལྷ་རྣམས་ཀྱི་དབང་པོ་བརྒྱ་བྱིན་དང་ཐབས་ཅིག་ཏུ་བཞུགས་ཏེ། དེ་ནས་བཅོམ་ལྡན་འདས་གདན་བཤམས་པ་ལ་བཞུགས་ནས་གཙུག་ཏོར་ལ་རྣམ་པར་གཟིགས་པ་ཞེས་བྱ་བའི་ཏིང་ངེ་འཛིན་ལ་སྙོམས་པར་ཞུགས་སོ། །བཅོམ་ལྡན་འདས་སྙོམས་པར་ཞུགས་མ་ཐག་ཏུ་གཙུག་ཏོར་གྱི་གསེབ་ནས་གསང་སྔགས་ཀྱི་གཞི་འདི་དག་བྱུང་ངོ། །ན་མོ་བྷ་ག་བ་ཏེ་ཨུཥྞཱི་ཥ་ཡ་ཤུདྡྷེ་བ་རི་ཛེ་བི་མ་ལེ་སྭཱཧཱ། བཅོམ་ལྡན་འདས་གཙུག་ཏོར་ཐོགས་པ་མེད་པ་ལ་ཕྱག་འཚལ་ལོ། །སངས་རྒྱས་ལ་ཕྱག་འཚལ་ལོ། །ཆོས་ལ་ཕྱག་འཚལ་ལོ། །དགེ་འདུན་ལ་ཕྱག་འཚལ་ལོ། །ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་བྱེ་བ་ཕྲག་བདུན་པོ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །བྱམས་པ་ལ་སོགས་པ་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཉན་ཐོས་ཀྱི་དགེ་འདུན་དང་བཅས་པ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །འཇིག་རྟེན་ན་དགྲ་བཅོམ་པ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །རྒྱུན་ཏུ་ཞུགས་པ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །ལན་གཅིག་ཕྱིར་འོང་བ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །ཕྱིར་མི་འོང་བ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །འཇིག་རྟེན་ན་ཡང་དག་པར་སོང་བ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །ཡང་དག་པར་ཞུགས་པ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །ལྷའི་དྲང་སྲོང་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །རིག་སྔགས་འཆང་གྲུབ་པའི་དྲང་སྲོང་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །རིག་སྔགས་འཆང་གྲུབ་པའི་དྲང་སྲོང་དམོད་༄༅། །པའི་མཚོན་ཆ་ཅན་དམོད་པས་ཕན་འདོགས་ནུས་པ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། །ཚངས་པ་ལ་ཕྱག་འཚལ་ལོ། །དབང་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་ལེགས་ལྡན་དྲག་པོ་དཀའ་ཐུབ་ཟློག་པའི་བདག་པོ་དང་ལྷན་ཅིག་པ་ལ་ཕྱག་འཚལ་ལོ། །ཆུ་ལྷ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་ལེགས་ལྡན་སྲེད་མེད་ཀྱི་བུ་ཕྱག་རྒྱ་ཆེན་པོ་ལྔས་ཕྱག་བྱས་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་ལེགས་ལྡན་དགའ་བྱེད་དབང་ཕྱུག་ནག་པོ་ཆེན་པོ་གྲོང་ཁྱེར་སུམ་བརྩེགས་འཇིག་པར་བྱེད་པ་ཁ་ཆེའི་དུར་ཁྲོད་ཆེན་པོ་ན་གནས་པར་མོས་པ་མ་མོའི་ཚོགས་ཀྱིས་ཕྱག་བྱས་པ་དང་ལྷན་ཅིག་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་པདམའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་རྡོ་རྗེའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་ནོར་བུའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་རྒྱལ་པོའི་གདུང་ལ་ཕྱག་འཚལ་ལོ༎ བཅོམ་ལྡན་འདས་ལས་ཀྱི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་རིན་པོ་ཆེའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་གཞོན་ནུའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་གླང་པོའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་གཏི་མུག་གི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་ཆགས་པའི་གདུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་དཔའ་བརྟན་པའི་སྡེ་མཚོན་ཆའི་རྒྱལ་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་འོད་དཔག་མེད་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་མི་བསྐྱོད་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་རྡོ་རྗེ་འཛིན་པ་རྒྱ་མཚོ་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་སྨན་གྱི་བླ་བཻ་ཌཱུརྱའི་འོད་ཀྱི་རྒྱལ་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་དོན་ཡོད་གྲུབ་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་སཱ་ལའི་དབང་པོའི་རྒྱལ་པོ་མེ་ཏོག་ཤིན་ཏུ་རྒྱས་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་པདམའི་མཆོག་གི་རྒྱལ་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་རྣམ་པར་གཟིགས་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་༄༅། །པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་གཙུག་ཏོར་ཅན་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་ཐམས་ཅད་སྐྱོབ་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་ལོག་པར་དད་སེལ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་གསེར་ཐུབ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་འོད་སྲུང་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་ཤཱཀྱ་ཐུབ་པ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་རིན་ཆེན་ཟླ་བ་ལ་ཕྱག་འཚལ་ལོ༎ བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་རིན་པོ་ཆེ་ཏོག་གི་རྒྱལ་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་ཀུན་ཏུ་བཟང་པོ་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་རྣམ་པར་སྣང་མཛད་ལ་ཕྱག་འཚལ་ལོ། །བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་དགྲ་བཅོམ་པ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་པདམ་རྒྱས་པ་དང་ཨུཏྤ་ལའི་དྲིའི་ཏོག་གི་རྒྱལ་པོ་ལ་ཕྱག་འཚལ་ལོ། །དེ་རྣམས་ལ་ཕྱག་འཚལ་ནས་བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པའི་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་འདི་རབ་ཏུ་བརྗོད་པར་བྱའོ། །འཐབ་པ་དང། འཐབ་མོ་དང། རྩོད་པ་དང། འགྱེད་པ་ཐམས་ཅད་རབ་ཏུ་འཇོམས་པར་བྱེད་པ། འབྱུང་པོ་དང་གདོན་ཐམས་ཅད་ཟློག་པར་བྱེད་པ། ཕ་རོལ་གྱི་རིག་སྔགས་ཐམས་ཅད་གཅོད་པར་བྱེད་པ། དུས་མ་ཡིན་པར་འཆི་བ་ལས་ཡོངས་སུ་སྐྱོབ་པ། སེམས་ཅན་བཅིངས་པ་ཐམས་ཅད་ཐར་པར་བྱེད་པ། སྡང་བ་དང་རྨི་ལམ་ངན་པ་ཐམས་ཅད་འཇོམས་པར་བྱེད་པ༑ གནོད་སྦྱིན་དང་སྲིན་པོའི་གདོན་ཐམས་ཅད་རྣམ་པར་བརླག་པར་བྱེད་པ། གདོན་སྟོང་ཕྲག་བརྒྱད་ཅུ་རྩ་བཞི་པོ་རྣམས་རྣམ་པར་འཇོམས་པར་བྱེད་པ། རྒྱུ་སྐར་ཉི་ཤུ་རྩ་བརྒྱད་རབ་ཏུ་སྒྲུབ་པར་བྱེད་པ༑ དགྲ་ཐམས་ཅད་ཟློག་པར་བྱེད་པ། གཟའ་ཆེན་པོ་བརྒྱད་རྣམ་པར་བརླག་པར་བྱེད་པ། སྡང་བ་དང་དྲག་པོ་དང་རྨི་ལམ་ངན་པ་རྣམས་རྣམ་པར་འཇོམས་པར་བྱེད་པ། དུག་དང་མཚོན་དང་མེ་དང་ཆུ་ལས་སྐྱོབ་པར་བྱེད་པ། ངན་འགྲོའི་འཇིགས་པ་ཐམས་ཅད་ལས་སྒྲོལ་བ། དུས་མ་ཡིན་པར་འཆི་བ་བརྒྱད་ཀྱི་བར་དུ་ཡོངས་སུ་སྐྱོབ་པར་བྱེད་པ་སྟེ། གཞན་གྱིས་མི་ཐུབ་དྲག་ཤུལ་ཆེ༎ ༄༅། །སྟོབས་ཆེན་གཟི་བརྗིད་ཆེན་མོ་སྟེ། །གཏུམ་པ་ཆེན་མོ་དཀར་མོ་ཆེ། །གསལ་བ་ཆེན་མོ་སྟོབས་ཆེན་མོ། །འབར་བ་ཆེན་མོ་བྲེང་བ་ཆེ། །གོས་དཀར་མོ་ནི་ཆེན་མོ་སྟེ༑ ༑འཕགས་མ་སྒྲོལ་མ་ཁྲོ་གཉེར་ཅན། །རྡོ་རྗེ་ཕྲེང་བ་རྣམ་བསྒྱིངས་མ། །པདམའི་སྣང་བ་རྡོ་རྗེའི་མཚན། །གཞན་གྱིས་མི་ཐུབ་ཕྲེང་བ་ཅན། །རྡོ་རྗེ་མཁར་བསྣམས་འཇོམས་པ་མོ། ། ཞི་བའི་ལྷ་རྣམས་ཀྱིས་མཆོད་མ། །གཟུགས་ཞི་དཀར་མོ་ཆེན་མོ་སྟེ། །འབར་བ་དང་ནི་གོས་དཀར་མོ། །འཕགས་མ་སྒྲོལ་མ་སྟོབས་ཆེན་མོ། །འཆི་མེད་རྡོ་རྗེ་ལུ་གུ་རྒྱུད། །རྡོ་རྗེ་གཞོན་ནུ་རིག་འཛིན་མ། །ལག་ན་རྡོ་རྗེའི་རྡོ་རྗེ་དང། །རིག་སྔགས་གསེར་གྱི་ཕྲེང་བ་ཅན། །ལེ་བརྒན་རྩི་དང་རིན་ཆེན་མ། །རྣམ་པར་སྣང་མཛད་རིགས་ཀྱིའོ། །དེ་བཞིན་གཤེགས་རིགས་གཙུག་ཏོར་མ། །རྣམ་པར་བསྒྱིངས་མའི་ཕྲེང་བ་དང། །རྡོ་རྗེའི་གསེར་འོད་ལྟ་བུའི་སྤྱན། །རྡོ་རྗེ་སྒྲོ་དང་དཀར་མོ་དང། །པདམའི་སྤྱན་དཔལ་སངས་རྒྱས་ཅན། །དེ་བཞིན་རྡོ་རྗེ་ཟླ་བའི་འོད། །དེ་བཞིན་རྡོ་རྗེ་འཛིན་མ་དང། །རྡོ་རྗེ་ཕྲེང་བ་སྒྱུ་འཕྲུལ་ཆེའི། །ལྷ་མོ་དང་ནི་གསེར་འོད་ཅན། །སྤྱན་མཛེས་མ་དང་དཀར་མོ་དང། །པདམ་ཡི་ནི་སྤྱན་ཅན་དང། །རྣམ་དུལ་ཞི་བཞི་སེམས་དང་ནི། །ཟླ་འོད་བདག་གིས་ཡོན་ཏན་ཤེས། །ཕྱག་རྒྱའི་ཚོགས་དེ་དག་དང། མ་མོའི་ཚོགས་དང་བཅས་པ་ཐམས་ཅད་ཀྱིས་བདག་འཁོར་དང་བཅས་པ་དང། སེམས་ཅན་ཐམས་ཅད་ལ་སྲུང་བ་མཛད་དུ་གསོལ། ཨོཾ་རི་ཥི་ག་ཎ་པྲ་ཤཔྟ་སརྦ་ཏ་ཐཱ་ག་ཏོཥྞཱི་ཥ་སི་ཏཱ་ཏ་པ་ཏྲེ་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །ཛམྦྷ་ནི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །སྟམྦྷ་ནི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །མོ་ཧ་ན་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། པ་ར་བིདྱཱ་སཾ་བྷཀྵཱ་ཎ་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །སརྦ་དུཥྚ་སྟམྦྷ་ན་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །སརྦ་པ་ར་བིདྱཱ་ཙྪེ་ད་ན་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །སརྦ་དུཥྚཱ་ནཱཾ་སྟམྦྷ་ན་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །སརྦ་ཡཀྵ་རཱཀྵ་ས་གྲ་ཧཱ་ཎཱཾ་བི་དྷྭཱཾ་ས་ན་ཀ་རཱ་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །ཙ་ཏུ་རཱ་ཤཱི་ཏི་ནཱཾ་གྲ་ཧ་ས་ཧ་སྭཱ་ཎཱཾ་བི་དྷྭ་ན་ས་ན་ཀ་རཱི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །ཨཥྚཱ་བྲིཾ་ཚྷི་ནཱཾ་ནཱ་ཀྵ་ཏྲཱ་ཎཱཾ། ས་དྷ་ན་ཀ་རི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །ཨཥྚ་ཎཱཾ་མ་ཧཱ་གྲ་ཧཱ་ཎཱཾ་བི་དྷྭཾ་ས་ན་ཀ་རི་ཧཱུཾ་དྲཱུཾ་ཧྲཱིཿཥྚོཾ། །རཀྵ་རཀྵ་མ་སརྦ་སཏྭཱཾ་ཤྩ། བཅོམ་ལྡན་འདས་མ་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་རྡོ་རྗེ་གཙུག་ཏོར་ཕྱིར་ཟློག་པ་ཆེན་མོ་ཕྱག་སྟོང་མངའ་བ་ཆེན་མོ༑ དབུ་སྟོང་མངའ་བ་ཆེན་མོ། སྤྱན་བྱེ་བ་བརྒྱ་ཕྲག་སྟོང་མངའ་བ་ཆེན་མོ། མི་ཕྱེད་འབར་བའི་མཚན་རྟགས་ཅན། རྡོ་རྗེ་ལྟོ་བ་ཡངས་པ་ཆེན་པོས་ས་གསུམ་གྱི་དཀྱིལ་འཁོར་ལ་མངའ་མཛད་མ་ལ་ཕྱག་འཚལ་ལོ། །ཨོཾ་སྭསྟི་རྦྷནྟུ་མ་མ་ས་པ་རི་དཱ་ར་སྱ་སརྦ་སཏྭཱ་ཤྩ་སྭཱཧཱ། རྒྱལ་པོའི་འཇིགས་པ་དང། ཆོམ་རྐུན་གྱི་འཇིགས་པ་དང། མེའི་འཇིགས་པ་དང། ཆུའི་འཇིགས་པ་དང། དུག་གི་འཇིགས་པ་དང། མཚོན་གྱི་འཇིགས་པ་དང། དགྲའི་འཇིགས་པ་དང། ཕ་རོལ་འཁྲུལ་འཁོར་གྱི་འཇིགས་པ་དང། མུ་གེའི་འཇིགས་པ་དང། དགྲའི་འཇིགས་༄༅། །པ་དང། ཐོག་གི་འཇིགས་པ་དང། དུས་མ་ཡིན་པར་འཆི་བའི་འཇིགས་པ་དང། ས་གཡོ་བའི་འཇིགས་པ་དང། སྐར་མདའ་ལྟུང་བའི་འཇིགས་པ་དང། རྒྱལ་པོའི་ཆད་པའི་འཇིགས་པ་དང། གཅན་གཟན་གཏུམ་པོའི་འཇིགས་པ་དང། ཀླུའི་འཇིགས་པ་དང། གློག་གི་འཇིགས་པ་དང། བྱེ་མ་ཚ་མོའི་འཇིགས་པ་དང། ནམ་མཁའ་ལྡིང་གི་འཇིགས་པ་དང། ཡམས་ཀྱི་ནད་དང། གནོད་པ་དང། འགོ་བའི་ནད་དང། འཁྲུག་པ་ཐམས་ཅད་ཀྱི་འཇིགས་པ་དང། གནོད་གྱི་འཇིགས་པ་དང། ལྷའི་གདོན་དང། ཀླུའི་གདོན་དང། གནོད་སྦྱིན་གྱི་གདོན་དང། སྲིན་པོའི་གདོན་དང། དྲི་ཟའི་གདོན་དང། ལྷ་མ་ཡིན་གྱི་གདོན་དང། ནམ་མཁའ་ལྡིང་གི་གདོན་དང། རླུང་ལྷའི་གདོན་དང། མིའམ་ཅིའི་གདོན་དང། ལྟོ་འཕྱེ་ཆེན་པོའི་གདོན་དང། མིའི་གདོན་དང། མི་མ་ཡིན་པའི་གདོན་དང། འབྱུང་པོའི་གདོན་དང། ཡི་དྭགས་ཀྱི་གདོན་དང། ཤ་ཟའི་གདོན་དང། གྲུལ་བུམ་གྱི་གདོན་དང། སྲུལ་པོའི་གདོན་དང། ལུས་སྲུལ་པོའི་གདོན་དང། སྐེམ་བྱེད་ཀྱི་གདོན་དང། སྨྱོ་བྱེད་ཀྱི་གདོན་དང། གྲིབ་གནོན་གྱི་གདོན་དང། བརྗེད་བྱེད་ཀྱི་གདོན་དང། གནོན་པོའི་གདོན་དང། མཁའ་འགྲོ་མའི་གདོན་དང། ལུས་ཆགས་མའི་གདོན་དང། ནམ་གྲུའི་གདོན་དང། ཚེར་མ་འདོན་པའི་གདོན་དང། རེ་ལྡེ་འདོན་པའི་གདོན་དང། བྱའི་གདོན་དང། མ་མོ་དགའ་བར་བྱེད་པའི་གདོན་དང། ཀུན་ཏུ་ཕྲེང་བར་བྱེད་པའི་གདོན་དང། འཕྱང་བའི་གདོན་དང། མཁའ་འགྲོ་མའི་གདོན་དང། ལུས་མཁའ་འགྲོ་མའི་གདོན་དང། ཀ་ཊཾ་ཀ་ཊའི་ཚོགས་ཅན་གྱི་གདོན་དང། གདོན་ཐམས་ཅད་དང། མདངས་འཕྲོག་པ་དང། མངལ་ཟ་བ་དང། ཁྲག་འཐུང་བ་དང། ཞག་ཟ་བ་དང། ཤ་ཟ་བ་དང། ཚིལ་ཟ་བ་དང། རྐང་ཟ་བ་དང། བཙས་པ་ཟ་བ་དང། སྲོག་འཕྲོག་པ་དང། གཏོར་མ་ཟ་བ་དང། ཕྲེང་བ་ཟ་བ་དང། དྲི་ཟ་བ་དང། མེ་ཏོག་ཟ་བ་དང། བདུག་པ་ཟ་བ་དང། འབྲས་བུ་ཟ་བ་དང། ལོ་ཏོག་ཟ་བ་དང། བསྲེག་བྱ་ཟ་བ་དང། རྣག་འཐུང་བ་དང། བཤང་བ་ཟ་བ་དང། བཅིན་འཐུང་བ་དང། མཆིལ་མ་ཟ་བ་དང། སྣབས་ཟ་བ་དང། དར་སྣབས་ཟ་བ་དང། སྐྱུག་པ་ཟ་བ་དང། ནོར་བུ་ཟ་བ་དང། མི་གཙང་བ་ཟ་བ་དང། སྟོར་ཁུང་ནས་འབབ་པ་འཐུང་བ་དང། ནོར་འཕྲོག་པ་དང། སེམས་འཕྲོག་པ་རྣམས་དང། དེ་དག་ཐམས་ཅད་དང། གདོན་ཐམས་ཅད་ཀྱིས་བྱས་པའི་རིག་སྔགས་ཐམས་ཅད་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཀུན་ཏུ་རྒྱུས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མཁའ་འགྲོ་དང་མཁའ་འགྲོ་མས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཚངས་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །བརྒྱ་བྱིན་གྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་༄༅། །བུས་གདབ་བོ། །སྲེད་མེད་ཀྱི་བུས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཕྱུགས་བདག་ཆེན་པོ་དང་དྲག་པོས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ནག་པོ་ཆེན་པོས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མ་མོའི་ཚོགས་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མིའི་ཐོད་པ་ཅན་གྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །རི་ཁྲོད་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། ། རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གཡུང་པོས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །སོར་མདའ་གདོང་གིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །རྡོ་རྗེ་གཞོན་ནུ་མས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གཤེན་རྗེའི་གཤེད་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གཤིན་རྗེའི་ཕོ་ཉས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཀླུ་མ་རུངས་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མིའི་ལས་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ལོག་འདྲེན་གྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། ། གཞོན་ནུས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །རྒྱལ་པོ་ཆེན་པོ་བཞིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །སྲིང་མོ་བཞིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མཁའ་ལྡིང་ཡང་དག་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །རྒྱལ་བར་བྱེད་པ་དང། སྦྲང་རྩིར་བྱེད་པ་དང། གྲུབ་པར་བྱེད་པ་དང། དོན་ཀུན་སྒྲུབ་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །བྲིང་གི་རི་ཏི་དང། དགའ་བྱེད་དབང་ཕྱུག་དང། སྨིན་དྲུག་གི་བུ་དང། ཟླ་བ་དང། ཉི་མ་དང། ཚོགས་ཀྱི་བདག་པོ་དང་ལྷན་ཅིག་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །དགེ་སྦྱོང་གཅེར་བུས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །དགྲ་བཅོམ་པས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །སྤྱན་རས་གཟིགས་དབང་ཕྱུག་གིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །འདོད་ཆགས་དང་བྲལ་བས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གསང་བ་པའི་བདག་པོ་ལག་ན་རྡོ་རྗེས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གང་དང་གང་དུ་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །གང་གིས་བྱེད་དུ་བཅུག་པ་དེས་བྱས་པའི་༄༅། །རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །དགེ་སྦྱོང་མགོ་རེག་གིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཕོ་ཉ་དང་ཕོ་ཉ་མོ་དང། བྲན་ཕོ་དང་བྲན་མོས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །དྲང་སྲོང་གི་མཆོག་ཐམས་ཅད་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ལྷའི་ཚོགས་ཐམས་ཅད་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །མི་ཕན་པར་འདོད་པ་ཐམས་ཅད་ཀྱིས་བྱས་པའི་རིག་སྔགས་རལ་གྲིས་གཅད་དོ། །རྡོ་རྗེ་ཕུར་བུས་གདབ་བོ། །ཨོཾ་བཅོམ་ལྡན་འདས་མ་བདག་འཁོར་དང་བཅས་པ་དང། སེམས་ཅན་ཐམས་ཅད་ཀྱི་འཇིགས་པ་ཐམས་ཅད་དང། གནོད་པ་དང། ནད་འགོ་བ་དང། འཁྲུག་པ་ཐམས་ཅད་དང། སྡང་བ་དང་རབ་ཏུ་སྡང་བ་ཐམས་ཅད་དང་ཕྱིར་རྒོལ་བའམ། མི་མཛའ་བའམ། མི་ཕན་པར་འདོད་པ་ཐམས་ཅད་ལས་སྲུང་དུ་གསོལ། སྲུང་དུ་གསོལ། དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན། སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་ཀྱིས་ཕྱག་བྱས་པ། མེ་དང་ཉི་མ་ལྟར་འོད་གསལ་ཞིང་དཀར་བ་རྒྱས་པ་མངའ་བ་ཁྱོད་ལ་ཕྱག་འཚལ་ལོ། །ཨོཾ་ཛྲ་ལ་ཛྭ་ལ། དྷ་ཀ་དྷ་ཀ །ཁཱ་ད་ཁཱད། ད་ར་ད་ར། བི་ད་ར་བི་ད་ར། ཙྪིནྡ་ཙྪིནྡ། བིནྡ་བིནྡ། ཧཱུཾ་ཧཱུཾ་ཕཊ་ཕཊ་སྭཱཧཱ། སརྦ་དུཥྚཱན་ཧཱུཾ་དྲཱུཾ། སརྦ་དུ་རླཾ་གི་ཏེ་བྷྱཿ ཕཊ། སརྦ་དུརླཾ་ཧི་ཏེ་བྷྱཿཕཊ། སརྦ་དུརྩྪ་ཡེ་བྷྱཿཕཊ། སརྦ་དི་ག་བྷྱཿ ཕཊ། སརྦ་དུརྦྷུ་ཀྟེ་བྷྱཿཕཊ། སརྦ་དུརྩྪདྡྷི་ཏེ་བྷྱཿ ཕཊ། སརྦ་ཨ་བ་དྷུ་ཏི་བྷྱཿ ཕཊ། སརྦ་དུ་སྐྲྀ་ཏེ་དྷྱཿ ཕཊ། སརྦ་དུཿབྲེ་ཀྵི་ཏེ་བྷྱཿཕཊ། སརྦ་ཛྭ་རེ་བྷྱཿཕཊ། སརྦཱ་པ་སྨཱ་རེ་བྷྱཿ ཕཊ། སརྦ་སྟྭ་ར་ཀེ་བྷྱཿཕཊ། སརྦ་ཌཱ་ཀི་ནཱི་བྷྱཿཕཊཿསར་རེ་བ་ཏཱི་བྷྱཿཕཊ། སརྦ་ཀ་ཏ་བཱ་སི་ནི་བྷྱཿཕཊ། སརྦ་ཡ་མ་ཡེབྷྱཿཕཊ། སརྦ་ཤ་ཀུ་ནི་བྷྱཿཕཊ། སརྦ་མཱ་ཏྲྀ་ནནྡི་ཀེ་བྷྱཿཕཊ། སརྦ་ག་རེ་བྷྱཿཕཊ། སརྦ་བི་སེ་བྷྱཿཕཊ། སརྦ་ཡོ་གེ་བྷྱཿཕཊ། སརྦ་ཨ་ལམང་ཀེ་བྷྱཿ ཕཊ༑ སརྦ་བྷ་ཡེ་བྷྱཿ ཕཊ། སརྦོ་པ་དྲ་བེ་བྷྱཿཕཊ། སརྦ་ཏཱིརྠི་སརྦོ་པ་སརྒོ་པ་ཡ་སེ་བྷྱཿཕཊ། སརྦ་ཨུ་ཏྲཱ་སེ་བྷྱཿཕཊ། སརྦ་པཱ་ཏ་ཀེ་བྷྱཿཕཊ། སརྦ་བྱཱུ་དྷི་བྷྱཿཕཊ། སརྦ་ཤྲ་མ་ཎེ་བྷྱཿཕཊ། སརྦ་གྲ་ཧེ་བྷྱཿཕཊ། སརྦ་ཏཱིརྠི་ཀེ་བྷྱཿ ཕཊ། སརྦ་པྲ་ཏྱིརྠ་ཀེ་བྷྱཿ ཕཊ། སརྦ་པཱ་ཏ་ཀེ་བྷྱཿཕཊ། སརྦོནྨཱ་དེ་བྷྱཿ ཕཊ། སརྦཙྪ་ཡེ་བྷྱཿཕཊ། སརྦ་བིདྱཱ་དྷ་རེ་བྷྱཿ ཕཊ། ཛ་ཡ་ཀ་ར་མ་དྷུ་ཀ་ར་སིདྡྷི་ཀ་ར་སརྦ་ཨརྠ་སཱ་དྷ་ཀེ་བྷྱཿཕཊ། སརྦ་བིདྱཱ་ཙཱརྱེ་བྷྱཿ ཕཊ། སརྦ་བིདྱཱ་རཱ་ཛེ་བྷྱཿཕཊ། སརྦ་སཱ་དྷ་ཀེ་བྷྱོ་བིདྱ་ཙརྱ་བྷྱཿཕཊ། ཙ་ཏུརྦྷྱོ་བྷ་གི་ནི་བྷྱཿཕཊ། བཛྲ་ཀཽ་མཱ་རཱི་ཡེ་བིདྱཱ་རཱ་ཛྙཱི་ཡེ་བྷྱཿ ཕཊ། སརྦ་བིགྷྣཾ་བི་ནཱ་ཡ་ཀཱ་ནཱཾ་ཕཊ། བ་ར་དཱ་ཡེ་ཕཊ། པ་ར་བི་དྲཱ་བ་ཎ་ཀ་རཱ་ཡ་ཕཊ། སརྦ་ཨ་སུ་རེ་བྷྱཿཕཊ། སརྦ་ག་རུ་ཌེ་བྷྱཿཕཊ། སརྦ་མ་ཧོ་ར་གེ་བྷྱཿཕཊ། སརྦམཱ་ནུ་ཥྱཱ་བྷྱཿ ཕཊ། མཱ་རུ་ཏེ་བྷྱཿཕཊ། ༄༅། །སརྦ་པི་ཤཱ་ཙེ་བྷྱཿ ཕཊ། སརྦ་ཀུམྦྷ་ཎྜེ་བྷྱཿ ཕཊ། བཛྲི་ཤྲཱིངྒ་ལཱ་ཡ་མནྟ་པྲ་ཏྱཾངྒི་རཱ་བྷྱཿཕཊ། སརྦོ་པ་སརྒེ་བྷྱཿཕཊ། སརྦ་རཱཀྵ་ས་བྷྱཿ ཕཊ། མཧཱ་པྲ་ཏྱིངྒ་རེ་བྷྱཿ ཕཊ༑ ཙིནྡ་ཙིནྡ་ཕཊ། བིནྡ་བིནྡ་ཕཊ། ཧཱུཾ་ཧཱུཾ་ཕཊ་སྭཱཧཱ། ཧེ་ཧེ་ཕཊ། ཧོ་ཧོ་ཕཊ། ཨ་མོ་གྷྭ་ཡ་ཕཊ། ཨ་པྲ་ཏི་ཧ་ཏ་ཡ་ཕཊ། བ་ར་པྲ་དཱ་ཡ་ཕཊ། ཨ་སུ་ར་བི་དྲཱ་བ་ཎ་ཀ་རཱ་ཡ་ཕཊ། སརྦ་དེ་བེ་བྷྱཿ ཕཊ། སརྦ་ནཱ་གེ་བྷྱཿ ཕཊ། སརྦ་ཡཀྵེ་བྷྱཿ ཕཊ། སརྦ་རཱཀྵ་སེ་བྷྱཿ ཕཊ། སརྦ་གནྡྷ་རྦེ་བྷྱཿ ཕཊ། སརྦ་ཀི་ནཱ་རེ་བྷྱཿཕཊ། སརྦ་པྲེ་ཏེ་བྷྱཿཕཊ། སརྦ་བྷཱུ་ཏེ་བྷྱཿ ཕཊ། སརྦ་ཀུམྦྷཎྜེ་བྷྱཿ ཕཊ། སརྦ་པཱུ་ཏ་ནེ་བྷྱཿ ཕཊ། སརྦ་ཀ་ཊ་པཱུ་ཏ་ནེ་བྷྱཿ ཕཊ། སརྦ་སྐནྡེ་བྷྱཿ ཕཊ། སརྦོནྨ་དེ་བྷྱཿ ཕཊ། བཛྲ་སཱཾ་ཀ་ལཱ་ཡ་མ་ཧཱ་པྲ་ཏྱཾ་གི་ར་རཱ་ཛཱ་ཡ་ཕཊ། ཀཱ་ལཱ་ཡ་ཕཊ། མ་ཧཱ་ཀཱ་ལཱ་ཡ་ཕཊ༑ མཱ་ཏྲི་ག་ཎཱ་ཡ་ཕཊ། བིཥྞུ་བཱི་ཡེ་ཕཊ། མ་ཧཱ་མཱ་ཏྲི་ག་ཎ་མ་སྐྲྀ་ཏི་ཡེ་ཕཊ། མ་ཧེ་ཤྭ་རཱ་ཡེ་ཕཊ། བྲཧྨ་ཎཱི་ཡེ་ཕཊ། ཨགནཱི་ཡེ་ཕཊ། མ་ཧཱ་ཀཱ་ལཱི་ཡེ་ཕཊ། ཀཱ་ལ་དཎྜི་ཡེ་ཕཊ། ཨིནྡྲི་ཡེ་ཕཊ། རཽ་དྲཱི་ཡེ་ཕཊ། ཙ་མུཎྜཱི་ཡེ་ཕཊ། བཱ་རཱ་ཧཱི་ཡེ་ཕཊ། མ་ཧཱ་བཱ་རཱ་ཧཱི་ཡེ་ཕཊ། རཱ་ཏྲི་ཡེ་ཕཊ། ཀཱ་ལ་རཱ་ཏྲི་ཡེ་ཕཊ། ཡ་མ་དཎྜཱི་ཡེ་ཕཊ། ཀ་པཱ་ལཱི་ཡེ་ཕཊ། མ་ཧཱ་ཀ་པཱ་ལཱི་ཡེ་ཕཊ། ཀཽ་མཱ་རི་ཡེ་ཕཊ། ཡ་མཱི་ཡེ་ཕཊ། བཱ་ཡུ་བེ་ཕཊ། ཀུ་བེ་རཱ་ཡེ་ཕཊ། ནཻ་རི་ཏཱི་ཡེ་ཕཊ། བ་རུ་ཎཱི་ཡེ་ཕཊ། མཱ་རུ་ཏཱི་ཡེ་ཕཊ། མ་ཧཱ་མཱ་རུ་ཏཱི་ཡེ་ཕཊ། སོ་མཱི་ཡེ་ཕཊ། ཨི་ཤཱ་ནཱི་ཡེ་ཕཊ། པོཀྐ་སཱི་ཡེ་ཕཊ༑ ཨརྠ་རྦ་ཎཱི་ཡེ་ཕཊ། ས་པ་རཱི་ཡེ་ཕཊ། ཀྲྀཥྞ་ས་པ་རཱི་ཡེ་ཕཊ། ཡ་མ་དཱུ་ཏི་ཡེ་ཕཊ། ནི་ས་ཏི་བཱ་ཙ་རེ་བྷྱཿ ཕཊ། ཏྲི་སན་ཡཱ་བ་རེ་བྷྱཿཕཊ། དྷ་ར་ཎཱི་ཡི་ཕཊ། ཨ་ཏི་མུཀྟི་ཀསྨཱི་ར་མཧཱ་ཤྨ་ཤཱ་ནཱི་བ་སཱི་ནཱི་ཡེ་ཕཊ། ཨི་ཏི་བྷྱཿ། སརྦ་བྷ་ཡེ་བྷྱཿ། སརྦ་དོ་ཥེ་བྷྱཿ ཕཊ། ཨོཾ་ཧཱུཾ་ཥྚོཾ་བྷནྡྷ་བནྡྷ་སརྦ་དུཥྚཱ་ན་རཀྵ་རཀྵ་མཱཾ་སརྦ་ས་ཏྭཱཾ་ཤྩ་སྭཱཧཱ། གང་སུ་དག་བདག་འཁོར་དང་བཅས་པ་དང། སེམས་ཅན་ཐམས་ཅད་ལ་སྡང་བ་སྡང་བའི་སེམས་དང་ལྡན་པ་རྣམས་དང། དྲག་ཤུལ་དྲག་ཤུལ་གྱི་སེམས་དང་ལྡན་པ་རྣམས་དང། སྡིག་པ་སྡིག་པའི་སེམས་དང་ལྡན་པ་རྣམས་དང། འཁྲུག་པ་འཁྲུག་པའི་སེམས་དང་ལྡན་པ་རྣམས་དང། མི་མཛའ་བ་མི་མཛའ་བའི་སེམས་དང་ལྡན་པ་དེ་རྣམས་ལས་བདག་འཁོར་དང་བཅས་པ་དང། སེམས་ཅན་ཐམས་ཅད་ལ་སྲུང་བ་མཛད་དུ་གསོལ། ལོ་བརྒྱ་འཚོ་བར་གྱུར་ཅིག །སྟོན་བརྒྱ་མཐོང་བར་ཤོག་ཅིག །གང་སུ་དག་གནོད་སྦྱིན་གྱི་གདོན་རྣམས་དང། མདངས་འཕྲོག་པ་རྣམས་དང། མངལ་ཟ་བ་རྣམས་དང། ཁྲག་འཐུང་བ་རྣམས་དང། ཞག་ཟ་བ་རྣམས་དང། ཤ་ཟ་བ་རྣམས་དང། ཚིལ་ཟ་བ་རྣམས་དང། རྐང་ཟ་བ་རྣམས་དང། བཙས་པ་ཟ་བ་རྣམས་དང། སྲོག་འཕྲོག་པ་རྣམས་དང། གཏོར་མ་ཟ་བ་རྣམས་དང། ཕྲེང་བ་ཟ་བ་རྣམས་དང། དྲི་ཟ་བ་རྣམས་དང། མེ་ཏོག་ཟ་བ་རྣམས་དང། བདུག་པ་ཟ་བ་རྣམས་དང། འབྲས་བུ་ཟ་བ་རྣམས་དང། ལོ་ཏོག་ཟ་བ་རྣམས་དང། བསྲེག་བྱ་ཟ་བ་རྣམས་དང། ནོར་འཕྲོག་པ་༄༅། །རྣམས་དང། སེམས་འཕྲོག་པ་རྣམས་དང། རྣག་ཟ་བ་རྣམས་དང། བཤང་བ་ཟ་བ་རྣམས་དང། གཅིན་འཐུང་བ་རྣམས་དང། མཆིལ་མ་འཐུང་བ་རྣམས་དང། རྣབས་ཟ་བ་རྣམས་དང། ངར་སྣབས་ཟ་བ་རྣམས་དང། སྐྱུག་པ་ཟ་བ་རྣམས་དང། ནོར་བུ་ཟ་བ་རྣམས་དང། མི་གཙང་བ་ཟ་བ་རྣམས་དང། སྟོར་ཁུང་ནས་འབབ་པ་འཐུང་བ་རྣམས་དང། སྡིག་པའི་སེམས་དང་ལྡན་པ་རྣམས་དང། སྡང་བའི་སེམས་དང་ལྡན་པ་རྣམས་དང། དྲག་ཤུལ་གྱི་སེམས་དང་ལྡན་པ་རྣམས་དང། ལྷའི་གདོན་རྣམས་དང། ཀླུའི་གདོན་རྣམས་དང། གནོད་སྦྱིན་གྱི་གདོན་རྣམས་དང། སྲིན་པོའི་གདོན་རྣམས་དང། དྲི་ཟའི་གདོན་རྣམས་དང། ལྷ་མ་ཡིན་གྱི་གདོན་རྣམས་དང། ནམ་མཁའི་ལྡིང་གི་གདོན་རྣམས་དང། མིའམ་ཅིའི་གདོན་རྣམས་དང། ལྟོ་འཕྱེ་ཆེན་པོའི་གདོན་རྣམས་དང། མིའི་གདོན་རྣམས་དང། མི་མ་ཡིན་པའི་གདོན་རྣམས་དང། རླུང་ལྷའི་གདོན་རྣམས་དང། ཡི་དྭགས་ཀྱི་གདོན་རྣམས་དང། ཤ་ཟའི་གདོན་རྣམས་དང། འབྱུང་པོའི་གདོན་རྣམས་དང། གྲུལ་བུམ་གྱི་གདོན་རྣམས་དང། སྲུལ་པོའི་གདོན་རྣམས་དང། ལུས་སྲུལ་པོའི་གདོན་རྣམས་དང། སྐེམ་བྱེད་ཀྱི་གདོན་རྣམས་དང། སྨྱོ་བྱེད་ཀྱི་གདོན་རྣམས་དང། གྲིབ་གནོན་གྱི་གདོན་རྣམས་དང། བརྗེད་བྱེད་ཀྱི་གདོན་རྣམས་དང། གནོན་པོའི་གདོན་རྣམས་དང། མཁའ་འགྲོ་མའི་གདོན་རྣམས་དང། ནམ་གྲུའི་གདོན་རྣམས་དང། ཀུན་ཏུ་འབྲེང་བའི་གདོན་རྣམས་དང། ཚེར་མ་གདོན་པའི་གདོན་རྣམས་དང། བྱའི་གདོན་རྣམས་དང། མ་མོ་དགའ་བར་བྱེད་པའི་གདོན་རྣམས་དང། བུམ་པ་ལྟ་བུའི་གདོན་རྣམས་དང། འཕྱང་བའི་གདོན་རྣམས་དང། ལུས་མཁའ་འགྲོ་མའི་གདོན་རྣམས་དང། ཀ་ཊཾ་ཀ་ཊའི་ཚོགས་ཅན་གྱི་གདོན་རྣམས་དང། གདོན་ཐམས་ཅད་བསལ་དུ་གསོལ། རིམས་ཉིན་གཅིག་པ་རྣམས་དང། ཉིན་གཉིས་པ་རྣམས་དང། ཉིན་གསུམ་པ་རྣམས་དང། ཉིན་བཞི་པ་རྣམས་དང། ཉིན་བདུན་པ་རྣམས་དང། ཟླ་བ་ཕྱེད་པ་རྣམས་དང། ཟླ་བ་གཅིག་པ་རྣམས་དང། ཉིན་མོ་པ་རྣམས་དང། ཉི་མ་ཕྱེད་པ་རྣམས་དང། ཡུད་ཙམ་པ་རྣམས་དང། རྟག་པའི་རིམས་རྣམས་དང། རིམས་ལྡང་དུབ་པ་རྣམས་དང། འབྱུང་པོའི་གདོན་རྣམས་དང། ཡི་དྭགས་ཀྱི་རིམས་རྣམས་དང། ཤ་ཟའི་རིམས་རྣམས་དང། མིའི་རིམས་རྣམས་དང། མི་མ་ཡིན་པའི་རིམས་རྣམས་དང། རྒྱུན་མི་འཆད་པའི་རིམས་རྣམས་དང། རླུང་ལས་གྱུར་པ་རྣམས་དང། མཁྲིས་པ་ལས་གྱུར་པ་རྣམས་དང། བད་ཀན་ལས་གྱུར་པ་རྣམས་དང། འདུས་པ་ལས་གྱུར་པ་རྣམས་དང། རིམས་ཐམས་ཅད་དང། ཀླད་པ་ན་བ་བདག་དང། སེམས་ཅན་ཐམས་ཅད་ལ་བསལ་དུ་གསོལ། བདག་དང་སེམས་ཅན་ཐམས་ཅད་ཀྱི་གཞོགས་ཕྱེད་ན་བ་དང། ཡི་ག་འཆུས་པ་དང། མིག་ནད་དང། སྣ་ནད་དང། ཁ་ནད་དང། མགུལ་ནད་དང། སྙིང་ནད་དང། ལྐོག་མའི་ལྷ་གོར་༄༅། །ན་བ་དང། རྣ་བ་ན་བ་དང། སོ་ན་བ་དང། བྲང་ན་བ་དང། སྙིང་ན་བ་དང། གནད་ན་བ་དང། རྩིབས་ལོགས་ན་བ་དང། རྒྱབ་ན་བ་དང། ལྟོ་ན་བ་དང། རྐེད་པ་ན་བ་དང། མདོམས་ན་བ་དང། གཞང་ན་བ་དང། སྐྱེ་གནས་ན་བ་དང། རབ་ཏུ་སྐྱེ་གནས་ན་བ་དང། བརླ་ན་བ་དང། བྱིན་པ་ན་བ་དང། ལག་པ་ན་བ་དང། རྐང་པ་ན་བ་དང། ཡན་ལག་ན་བ་དང། ཉིང་ལག་ན་བ་བསལ་དུ་གསོལ། བདག་དང་སེམས་ཅན་ཐམས་ཅད་ལ་འབྱུང་པོ་དང། ཡི་དྭགས་དང། རོ་ལངས་དང། མཁའ་འགྲོ་མ་དང། རིམས་དང། ཤུ་བ་དང། མིག་སེར་དང། གཡའ་བ་དང། འབྲུམ་བུ་ཕྲ་མོ་དང། མཛེ་དང། ཕོལ་མིག་དང། བས་བླྔགས་དང། མཆིན་ནད་དང། མཚན་པར་རྡོལ་བ་དང། གཡན་པ་དང། གཞང་འབྲུམ་དང། མེ་དབལ་དང། ལྷོག་པ་དང། སྐེམ་པ་དང། དབུགས་མི་བདེ་བ་དང། སྐྲག་པ་དང། ལུས་པ་དང། བརྒྱལ་པ་དང། དབྱིག་དུག་དང། དུག་སྦྱར་མ་དང། དུག་དང། བྱད་དང། མེ་དང། དམུ་ཆུ་དང། མི་ངས་དང། འཐབ་མོ་དང། འཁོན་དང། གནོད་པ་དང། དུས་མ་ཡིན་པར་འཆི་བ་དང། སྦྲང་བུ་ཏྲེ་མ་བུ་ག་དང། ཏྲཻ་ལ་ཏ་ཀ་དང། སྡིག་པ་དང། སྦྲུལ་དང། སྲེ་མོ་དང། སེངྒེ་དང། སྟག་དང། དོམ་དང་དྲེད་དང། གནག་དང། ཆུ་སྲིན་དང། སྤྱང་ཀི་དང། ཆོམ་པོ་དང། དེ་དག་ལ་སོགས་པ་སྲོག་འཕྲོག་པར་བྱེད་པ་གཞན་ཐམས་ཅད་ཀྱང་བསལ་དུ་གསོལ། གདུགས་དཀར་པོ་ཅན་རྡོ་རྗེ་གཙུག་ཏོར་ཕྱིར་ཟློག་པ་ཆེན་མོའི་རིག་སྔགས་ཀྱི་མཐུས་དཔག་ཚད་བཅུ་གཉིས་ཀྱི་ནང་ཚུན་ཆད་དམ། དཔག་ཚད་ལྔ་བརྒྱའི་ནང་ཚུན་ཆད་དུ་རིག་སྔགས་བཅིང་བར་བགྱིའོ། །གཟི་བརྗིད་བཅིང་བར་བགྱིའོ། ། རིག་སྔགས་ཐམས་ཅད་བཅིང་བར་བགྱིའོ། །གཞན་གྱི་རིག་སྔགས་ཐམས་ཅད་བཅིང་བར་བགྱིའོ། །མཚམས་གཅད་པར་བགྱིའོ། །ས་བཅིང་བར་བགྱིའོ། །ཕྱོགས་བཅུ་བཅིང་བར་བགྱིའོ། ། ནམ་མཁའ་བཅིང་བར་བགྱིའོ། །གཞན་གྱི་སྡེ་རབ་ཏུ་རེངས་པར་བགྱིའོ། །ཏདྱ་ཐཱ། ཨོཾ་ཨ་ན་ལེ་ཨ་ན་ལེ། ཨ་ཙ་ལེ་ཨ་ཙ་ལེ། ཁ་ཁ་མེ་ཁ་ཁ་མེ། བི་ཥ་དེ་བི་ཥ་དེ། བཱི་རེ་བཱི་རེ། བཻ་ར་བཻ་ར། སཽ་མེ་སཽ་མེ༑ ཤཱནྟེ་ཤཱནྟེ། དཱནྟེ་དཱནྟེ། བཛྲ་དྷ་ར་བནྡྷ་བནྡྷ་ནི་བཛྲ་པཱ་ཎི་ཕཊ། ཨོཾ་ཧཱུཾ་ཥྚོཾ་ཕཊ་སྭཱཧཱ། ཨོཾ་བཛྲ་པཱ་ཎི་བནྡྷ་བནྡྷ་བཛྲ་བཱ་ཤེ་ན་སརྦ་དུཥྚ་བིགྷྣཱ་བི་ནཱ་ཡ་ཀཱ་ན་ཧཱུཾ་ཧཱུཾ་ཕཊ་ཕཊ། ཧཱུཾ་དྲཱུཾ་བནྡྷ་ཕཊ། རཀྵ་རཀྵ་མཱཾ་སརྦ་སཏྭཱཾ་ཤྩ་སྭཱཧཱ། གང་གིས་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པའི་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་ཆེན་མོ་འདི་གྲོ་གའམ། རས་སམ། ཤིང་ཤུན་ལ་བྲིས་ཏེ་ལུས་སམ། མགུལ་དུ་ཐོགས་པ་བྱས་ནས་བཅང་བར་བྱའོ། །བཀླག་པར་བྱ་སྟེ། དེ་དག་ལ་ཇི་སྲིད་འཚོའི་བར་དུ་དུག་གིས་མི་ཚུགས་མཚོན་གྱིས་མི་ཚུགས། རིམས་ཀྱིས་མི་ཚུགས། ནད་ཀྱིས་མི་ཚུགས། ཆད་པས་མི་ཚུགས། མེ་དང་ཆུས་མི་ཚུགས། བསྒྱུར་བའི་ལས་ཐམས་ཅད་ཀྱིས་མི་ཚུགས། དབྱིག་དུག་གིས་མི་༄༅། །ཚུགས༑ དུག་སྦྱར་མས་མི་ཚུགས། བྱད་ཀྱིས་མི་ཚུགས། དུས་མ་ཡིན་པར་འཆི་བའི་དུས་བྱེད་པར་མི་འགྱུར་རོ། །གདོན་ཐམས་ཅད་དང། བགེགས་དང། ལོག་འདྲེན་ཐམས་ཅད་ཀྱི་ཡིད་དུ་འོང་ཞིང་དགའ་བར་འགྱུར་རོ། །བསྐལ་པ་བྱེ་བ་སྟོང་ཕྲག་བརྒྱད་ཅུ་རྩ་བཞིར་སྐྱེ་བ་དང་སྐྱེ་བར་ཚེ་རབས་དྲན་པར་འགྱུར་རོ། །རྡོ་རྗེའི་རིགས་བྱེ་བ་ཁྲག་ཁྲིག་འབུམ་ཕྲག་བརྒྱད་ཅུ་རྩ་བཞིའི་རིག་སྔགས་ཀྱི་ལྷས་ཀྱང་རྟག་ཏུ་རྒྱུན་མི་འཆད་པར་མཉམ་པར་བཞག་ནས་དེ་ལ་རྟག་པར་སྲུང་བ་དང། སྐྱོབ་པ་དང། སྦེད་པར་བྱེད་དོ། །རྡོ་རྗེ་ཕོ་ཉ་མོ་མངག་གཞུག་བརྒྱད་ཅུ་རྩ་བཞིས་ཀྱང་རྟག་པར་ཡོངས་སུ་སྐྱོང་བར་བྱེད་དེ། དེ་རྣམས་ཀྱི་ཡིད་དུ་འོང་བ་དང། དགའ་བར་འགྱུར་རོ། །ནམ་དུ་ཡང་གནོད་སྦྱིན་ཉིད་དང། སྲིན་པོ་ཉིད་དང༑ འབྱུང་པོ་ཉིད་དང། ཡི་དྭགས་ཉིད་དང། ཤ་ཟ་ཉིད་དང། སྲུལ་པོ་ཉིད་དང། ལུས་སྲུལ་པོ་ཉིད་དང། མི་དབུལ་པོར་མི་འགྱུར་རོ། །སངས་རྒྱས་བཅོམ་ལྡན་འདས་གྲངས་མེད་དཔག་ཏུ་མེད་པ་གངྒཱའི་ཀླུང་གི་བྱེ་མ་སྙེད་ཀྱི་བསོད་ནམས་ཀྱི་ཕུང་པོ་དང་ལྡན་པར་འགྱུར་རོ། །དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པའི་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་ཆེན་མོ་འདི་འཆང་ན། ཚངས་པར་མི་སྤྱོད་པ་ཡང་ཚངས་པར་སྤྱོད་པར་འགྱུར་རོ། །ཐུབ་པར་མ་བྱས་པ་ཡང་ཐུབ་པར་བྱས་པར་འགྱུར་རོ། །མི་གཙང་བ་ཡང་གཙང་བར་བྱས་པར་འགྱུར་རོ། །བསྙེན་གནས་བྱས་པ་ཅན་མ་ཡིན་པ་ཡང་བསྙེན་གནས་བྱས་པ་ཅན་དུ་འགྱུར་རོ། །གང་དག་མཚམས་མེད་པ་ལྔ་བྱས་པ་དེ་ཡང་སྡིག་པ་མ་ལུས་པ་འབྱང་བར་འགྱུར་རོ། །སྔོན་གྱི་ལས་ཀྱི་སྒྲིབ་པ་ཡང་མ་ལུས་པར་ཟད་པར་འགྱུར་རོ། །བུད་མེད་གང་ལ་ལས་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བའི་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པ་ཆེན་མོའི་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་ཆེན་མོ་འདི་འཆང་ན། བུ་འདོད་པ་དེས་ཀྱང་བུ་རྙེད་དེ། ཆེ་དང། བསོད་ནམས་དང། སྟོབས་ཀྱང་རབ་ཏུ་ལྡན་ནོ། །འདི་ནས་ཤི་འཕོས་ནས་ཀྱང་འཇིག་རྟེན་གྱི་ཁམས་བདེ་བ་ཅན་དུ་སྐྱེ་སྟེ། འདོད་ཆགས་དང། ཞེ་སྡང་དང། གཏི་མུག་དང། ང་རྒྱལ་དང། རྒྱགས་པ་ཐམས་ཅད་དང་བྲལ་བར་འགྱུར་རོ། །གང་ལ་ལ་ཞིག་མི་ངས་དང། བ་ལང་གི་ནད་དང། ཕྱུགས་ནད་དང། ཡམས་ཀྱི་ནད་དང། གནོད་པ་དང། འགོ་བའི་ནད་དང། འཁྲུག་པ་ཐམས་ཅད་དང། ཕ་རོལ་གྱི་དམག་ཚོགས་འོང་བ་ལ་བཅོམ་ལྡན་འདས་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་མི་ཕམ་པ། དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་པ་ཆེན་མོ་འདི་རྒྱལ་མཚན་གྱི་རྩེ་མོ་ལ་བཏགས་ཏེ་རིམ་གྲོ་ཆེར་མཆོད་པ་ཆེན་པོ་བྱས་ནས་གྲོང་ཁྱེར་ཐམས་ཅད་ཀྱི་སྒོའམ། གཙུག་ལག་ཁང་ངམ༑ ༄༅། །གྲོང་ངམ། གྲོང་ཁྱེར་རམ། ལྡོངས་སམ། གྲོང་རྡལ་ལམ། དུར་ཁྲོད་དམ། རིའམ། དགོན་པའི་གནས་སུ་གཟུགས་སོ། །གཞན་གྱིས་མི་ཐུབ་མ་ཕྱིར་ཟློག་མ་ཆེན་མོ་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་འདི་རིམ་གྲོ་ཆེར་བྱས་ཏེ་བཙུགས་པ་ཙམ་གྱིས་གཡུལ་ཞི་བར་བྱས་པར་འགྱུར་རོ། །ཡམས་ཀྱི་ནད་དང། གནོད་པ་དང། འགོ་བའི་ནད་དང། འཁྲུག་པ་ཐམས་ཅད་དང། ཕ་རོལ་གྱི་དམག་ཚོགས་རྣམས་རབ་ཏུ་ཞི་བར་འགྱུར་རོ། །ཀླུའི་རྒྱལ་པོ་མཐའ་ཡས་དང། ཀླུའི་རྒྱལ་པོ་དུང་སྐྱོང་དང། ཀླུའི་རྒྱལ་པོ་ནག་པོ་ཆེན་པོ་དང། ཀླུའི་རྒྱལ་པོ་དགའ་བོ་དང། ཉེ་དགའ་བོ་དང། དེ་དག་དང། གཞན་ཡང་ཀླུའི་རྒྱལ་པོ་ཐམས་ཅད་ཀྱིས་ཀྱང་དུས་དུས་སུ་ཆར་འབེབས་སོ། །དུས་དུས་སུ་གྲོག་འབྱིན་ཏོ། །དུས་དུས་སུ་འབྲུག་སྒྲོག་གོ །ནད་དང་གནོད་པ་ཐམས་ཅད་ཉེ་བར་ཞི་བར་འགྱུར་རོ། །ཨོཾ་ཧཱུཾ་ཥྚོཾ་བནྡྷ་བནྡྷ་སརྦ་དུཥྚན་རཀྵ་མ་མ་ས་པ་རི་བཱ་ར་སྱ་སརྦ་ས་ཏྭཱ་ཤྩ་སྭཱཧཱ། ཨོཾ་ཧཱུཾ་ཥྚོ་བནྡྷ་བནྡྷ་སརྦ་དུཥྚ་ན་རཀྵ་མ་མ་ས་པ་རི་བཱ་ར་སྱ་ས་ཏྭཾ་ཤྩ་བཛྲ་པཱ་ཎི་ཧཱུཾ་ཕཊ་སྭཱཧཱ། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏོཥྞཱི་ཥ་ཨ་བ་ལོ་ཀི་ཏེ་མཱུརྟ་ནི་ཏེ་ཛོ་རཱ་ཤི། ཨོཾ་ཛྭ་ལ་ཛྭ་ལ། ཁཱ་ད་ཁཱ་ད། དྷ་ཀ་དྷ་ཀ། ད་ར་ད་ར། བི་ད་ར་བི་ད་ར། ཙྪིནྡ་ཙྪིནྡ། བྷིནྡ་བྷིནྡ་ཧཱུཾ་ཧཱུཾ་ཕཊ་ཕཊ་རཀྵ་རཀྵ་མ་མ་ས་པ་རི་ཝ་ར་སྱ་སརྦ་སཏྭཱ་ཤྩ་སྭཱཧཱ། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏོཥྞཱི་ཥ་སི་ཧ་ཏ་བ་ཏྲཻ་ཧཱུཾ་ཕཊ་སྭཱཧཱ། ཧཱུཾ་ཧཱུཾ་རཀྵ་རཀྵ་མ་མ་ས་པ་རི་ཝཱ་ར་སྱ་སརྦ་ས་ཏྭཱཾ་ཤྩ་ཧཱུཾ་ཕཊ་སྭཱཧཱ། ཏདྱ་ཐཱ། ཨོཾ་ཨ་ན་ལེ་ཨ་ན་ལེ། ཨ་ཙ་ལེ་ཨ་ཙ་ལེ། ཁ་ས་མེ་ཁ་ས་མེ། བཱི་རེ་བཱི་རེ། བཻ་ར་བཻ་ར། སཽ་མེ་སཽ་མེ། སརྦ་བུདྡྷ་ཨདྷིཥཊཿཨཱ་ན་ཨ་དྷིཥཊཿཨཱི་ཏེ་སརྦ་ཏ་ཐཱ་ག་ཏོཥྞཱི་ཥ་སི་ཏཱ་ཏ་པ་ཏྲེ་སརྦ་དུཥྚ་ཙི་ཏ་ཏཱ་ན་ཧཱུཾ་ཕཊ་སྭཱཧཱ། སངས་རྒྱས་ཀྱ་རྣལ་འབྱོར་གྱིས་ཤིག་པ་ཐམས་ཅད་ལ་ལན་གསུམ་བཟླས་པར་བྱའོ། །བཅོམ་ལྡན་འདས་ཀྱིས་དེ་སྐད་ཅེས་བཀའ་སྩལ་ནས་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་དང། ལྷ་དང། མི་དང། ལྷ་མ་ཡིན་དང། ནམ་མཁའ་ལྡིང་དང། མིའམ་ཅི་དང། དྲི་ཟར་བཅས་པའི་འཇིག་རྟེན་ཡིད་རངས་ཏེ་བཅོམ་ལྡན་འདས་ཀྱིས་གསུངས་པ་ལ་མངོན་པར་བསྟོད་དོ། །འཕགས་པ་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་གཙུག་ཏོར་ནས་བྱུང་བ་གདུགས་དཀར་པོ་ཅན་ཞེས་བྱ་བ་གཞན་གྱིས་མི་ཐུབ་པ་ཕྱིར་ཟློག་པ་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་ཆེན་མོ་རྫོགས་སོ༎ ༎

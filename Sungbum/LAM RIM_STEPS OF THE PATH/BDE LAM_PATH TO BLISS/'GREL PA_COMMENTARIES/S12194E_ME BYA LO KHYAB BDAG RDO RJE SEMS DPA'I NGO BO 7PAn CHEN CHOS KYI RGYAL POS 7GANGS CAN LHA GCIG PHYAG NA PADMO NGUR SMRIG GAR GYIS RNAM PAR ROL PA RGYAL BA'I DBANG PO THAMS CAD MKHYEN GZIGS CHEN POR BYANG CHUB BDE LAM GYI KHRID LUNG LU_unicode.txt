༄༅། །མེ་བྱ་ལོ་ཁྱབ་བདག་རྡོ་རྗེ་སེམས་དཔའི་ངོ་བོ ༧པཎ་ཆེན་ཆོས་ཀྱི་རྒྱལ་པོས ༧གངས་ཅན་ལྷ་གཅིག་ཕྱག་ན་པདམོ་ངུར་སྨྲིག་གར་གྱིས་རྣམ་པར་རོལ་པ་རྒྱལ་བའི་དབང་པོ་ཐམས་ཅད་མཁྱེན་གཟིགས་ཆེན་པོར་བྱང་ཆུབ་བདེ་ལམ་གྱི་ཁྲིད་ལུང་ལུང་རིགས་ཀྱི་མཐའ་དཔྱོད་དང་བཅས་ཏེ་རྒྱས་པར་སྩལ་སྐབས། གཞུང་གི་ཚིག་ཟིན་ལ་མི་གསལ་བའི་གསུང་བཤད་ཕལ་མོ་ཆེའི་ཟིན་བྲིས་བདེ་ལམ་ཡང་གསལ་སྒྲོན་མེ་ཞེས་བྱ་བ་བཞུགས་སོ༎ ༄༅༅། །རྗེ་བཙུན་བླ་མ་དམ་པ་དམིགས་པ་མེད་པའི་ཐུགས་རྗེ་ཆེན་པོ་དང་ལྡན་པ་དེ་ཉིད་ཀྱི་ཞབས་ལ་སྒོ་གསུམ་གུས་པ་ཆེན་པོས་ཕྱག་འཚལ་ལོ། །ཚེ་རབས་ཐམས་ཅད་དུ་རྗེས་སུ་བཟུང་དུ་གསོལ། འདིར་རྒྱལ་བའི་གདུང་འཚོབ་ཀུན་གཟིགས་པཎ་ཆེན་ཆོས་ཀྱི་རྒྱལ་པོས། བྱང་ཆུབ་བདེ་ལམ་གྱི་ཁྲིད་ལུང་སྩལ་སྐབས་གཞུང་གི་རྩ་ཚིག་ལ་མི་གསལ་བའི་གསུང་བཤད་རགས་བ་རྣམས་ཟིན་བྲིས་སུ་བཀོད་པ་ལ། ༈ ཉིན་དང་པོར། དཔལ་སྒྱུ་འཕྲུལ་དྲ་བ་ལས། ཆོས་ཀྱི་རྒྱལ་མཚན་ལེགས་པར་འཛུགས། ཞེས་དྲངས་ནས་སྤྱི་བཤད་མཛད་པའི་ཚེ། གངྒཱའི་ཀླུང་གི་བྱེ་མ་མང་བའི་དོན་ནི། འཛམ་གླིང་འདིར་མ་ཟད་གླིང་བཞི་བྱེ་བ་ཕྲག་བརྒྱ་པོ་རེ་རེར་ཡང། གངྒཱའི་ཀླུང་རེ་ཡོད་ཅིང། དེ་རྣམས་ཀྱི་བྱེ་མ་ནི་བྱེ་བ་གཞན་༄༅། །ལས་ཀྱང་ཞིབ་པའི་རྒྱུ་མཚན་གྱིས་ཡིན་གསུངས། རྒྱལ་མཚན་ཞེས་པ་ནི་དཔེར་ན་ལྷ་དང་ལྷ་མིན་འཐབ་པའི་ཚེ། ལྷ་མིན་ཕམ་ཞིང་ལྷ་རྣམས་རྒྱལ་ན། རྒྱལ་བའི་བརྡ་རུ་མཚོན་འཕྱར་བ་དེ་ལ་ཟེར། ཆོས་ཀྱི་རྒྱལ་མཚན་ལེགས་པར་འཛུགས་ཞེས་པའི་སྐབས་ཀྱི་རྒྱལ་མཚན་ནི། བྱང་ཆུབ་ཀྱི་ཤིང་དྲུང་དུ་བདུད་ལས་རྒྱལ་ཏེ་ཆོས་ཀྱི་འཁོར་ལོ་བསྐོར་བ་ལ་ཟེར། འདིར་གང་ལས་རྒྱལ་ན་བདུད་བཞི་ལས་རྒྱལ། གང་རྒྱལ་ན། བདག་ཅག་གི་སྟོན་པ་ལྟ་བུ་རྒྱལ། ཞེས་གསུངས། དེ་མཛད་པ་པོ་ནི་འདིའི་ཡང་མཛད་པ་པོ་ཡིན་གསུངས་པའི་སྐབས་ལས་འཕྲོས་ནས། མངོན་པར་རྟོགས་ཞེས་པའི་དོན་ལ་རང་རྒྱུད་ལ་མངོན་པར་རྟོགས་ཚུལ་དང། གཞན་རྒྱུད་ལ་མངོན་པར་རྟོགས་ཚུལ་གཉིས་ཡོད། དང་པོ་ནི་རང་གིས་ཐོས་པའི་ལུང་དོན་ལ་བསམས་ནས། ཐོས་བསམ་སྒོམ་གསུམ་གྱི་ཤེས་རབ་སྐྱེས་པ་ལ་བྱེད། གཉིས་པ་ནི་བཅོམ་ལྡན་འདས་ཐུབ་པའི་དབང་པོ་སྐུ་མདོག་གསེར་བཅོམ་ལྟ་བུ་ཞེས་སོགས་ཐུབ་པའི་སྐུའི་རྣམ་འགྱུར་སོགས་གཞན་གྱིས་གོ་བའི་ཐབས་ལ་བྱེད། སྦྱོར་བའི་ཆོས་དྲུག་གི་སྐབས་སྔོན་འགྲོ་གནས་ཁང་གི་ཕྱོགས་བཞིར་མཐོ་བཞི་བརྩིགས་ནས་རྒྱལ་པོ་ཆེན་པོ་བཞིར་བསྐྱེད། ཡེ་ཤེས་པ་འགུག་པ་སོགས་བྱེད་རྒྱུ་བྱུང་ན་རབ་མ་བྱུང་ཡང་རྒྱལ་པོ་ཆེན་པོ་བཞིར་མོས་ནས། འཕྲིན་ལས་བཅོལ་ན། བགེགས་མི་འབྱུང་བའི་དགོས་པ་ཆེ། རྒྱུ་མཚན་སངས་རྒྱས་ཀྱི་བསྟན་པ་མིའི་ནང་ནས་འོད་སྲུང་ཆེན་པོ་དང། ལྷའི་ནང་ནས་རྒྱལ་ཆེན་བཞིར་གཏད་པའི་རྒྱུ་མཚན་གྱིས་ཡིན་གསུངས། དེའི་ཤེས་བྱེད་དུ་མདོ་དྲན་པ་ཉེར་གཞག་གི་ཁུངས་གསུངས། རྗེ་རིན་པོ་ཆེ་ནི་ཉན་ཐོས་འོད་སྲུང་ཆེན་པོའི་སྐུའི་སྐྱེ་བ་ཡིན། དེའི་རྒྱུ་མཚན་རྨི་ལམ་དུ་ཡང་རྗེ་བཙུན་ཙོང་ཁ་པས་ཞེས་སོགས་ཀྱིས་ཀྱང་འགྲུབ། ཙོང་ཁ་བརྒྱད་ཅུར་སྔོན་ཚེ་བྲམ་ཟེའི་ཁྱེའུར་འཁྲུངས་པའི་ཚེ། ཞེས་པ་ནི་འོད་སྲུང་ཆེན་པོ་བྲམ་ཟེའི་རིགས་ཡིན་པ་ལ་དགོངས་པ་འདྲ། བྲམ་ཟེའི་ཁྱེའུར་གྱུར་པའི་སྐབས་སེམས་བསྐྱེད་པའི་མཚམས་སྦྱོར་བྱེད་མཁན་ནི། རྒྱལ་སྲས་མོས་པའི་བློ་གྲོས་ཡིན། ཡང་འོད་སྲུང་ཆེན་པོའི་སློབ་མ་དགེ་སློང་པདམའི་ངད་ལྡན་དུ་ཡང་གྱུར། ས་དང་པོ་ལ་གནས་པའི་བྱང་སེམས་རྣམས་ཀྱི་རྣམ་སྨིན་གྱི་སྐྱེ་བ་ནི་བརྒྱ་དུས་གཅིག་ཏུ་ལེན། སྨོན་ལམ་གྱི་སྐྱེ་བ་ནི་གྲངས་མེད་ལེན་གསུངས། དེའི་རྒྱུ་མཚན་ཞིབ་པ་མི་གསུངས། དཀོན་མཆོག་གསུམ་སྤྱན་འདྲེན་པའི་ཚེ་གནས་ཁང་བྱི་དོར་བྱས་པས་དཀོན་མཆོག་གསུམ་ལ་མི་བརྙས་པའི་རྒྱུ་དང། མ་འོངས་༄༅། །པར་སངས་རྒྱས་ཀྱི་ཞིང་ཁྱད་པར་ཅན་འགྲུབ་པའི་རྟེན་འབྲེལ་འགྲིག །མཆོད་པའི་སྐབས་འབེན་གུང་རྒྱལ་གྱི་རྣམ་ཐར་གསུངས། སངས་རྒྱས་ཀྱི་འོད་ཟེར་ལ་སྐུ་གསུང་ཐུགས་གསུམ་གྱི་འོད་ཟེར་གསུམ་ཡོད། བདག་ལྟ་བུ༑ བླ་མའི་སྐུར་ལམ་གྱིས་སྟོན་པ་ལྟ་བུ་ནི་སྐུའི་འོད་ཟེར་ཡིན། སངས་རྒྱས་ཀྱི་གསུང་ཐམས་ཅད་ནི་གསུང་གི་འོད་ཟེར་ཡིན། ཐུགས་ཀྱི་བྱིན་རླབས་ཀྱིས་གདུལ་བྱའི་མི་ཤེས་པའི་མུན་པ་སེལ་བ་ནི་ཐུགས་ཀྱི་འོད་ཟེར་ཡིན། མདོར་ན་འོད་ཟེར་ཞེས་པ་ནི་སྣང་བར་བྱེད་པའི་དོན་ཡིན་ཞིང། དེས་སེམས་ཅན་གྱི་མ་རིག་པའི་མུན་པ་སེལ་བར་བྱེད་པ་དེ་ཀ་རང་ལ་བྱ། འདིའི་སྐབས་སུ་ཨོ་རྒྱན་པ་རིན་ཆེན་དཔལ་གྱིས་བྱ་ཕོ་ལ་སྒོམ་བཏབ་པའི་ལོ་རྒྱུས་གསུངས། བླ་ན་མེད་པའི་བྱང་ཆུབ་ཏུ་སེམས་བསྐྱེད་ཅེས་པའི་སྐབས་བྱད་པར་སེམས་བསྐྱེད་པ་དང་ཆུབ་པར་སེམས་བསྐྱེད་ལུགས་གཉིས་གསུངས། སངས་རྒྱས་བྱང་སེམས་ཀྱི་མཐུ་བསམ་གྱིས་མི་ཁྱབ་པའི་ཤེས་བྱེད་ལ། མགོན་པོ་ཁྱོད་ཀྱི་ཞལ་གྱི་དབུགས་ཙམ་ཡང། ཞེས་སོགས་འདྲེན་པ་གནང་ནས། དེའི་དཔེར་བརྗོད་ལ་པཎ་ཆེན་རིན་པོ་ཆེ་བློ་བཟང་ཆོས་རྒྱན་འབྲུག་གི་ཡུལ་དུ་ཕེབས་བའི་ལམ་དུ་སྦལ་པ་ཆེན་པོ་གཅིག་ལ་སྦལ་པ་ཆུང་ངུ་མང་པོས་བསྐོར་ནས་ཟ་བར་གཟིགས། འདི་ག་དུས་ནས་བྱུང་དྲིས་པས། སྔ་མོ་ནས་ཡོད་པར་འདུག་ཟེར། དེར་ལྗགས་ཆབ་གཅིག་བཏབ་ནས་ཕེབས། སླར་ཚུར་ལམ་གཟིགས་དུས་དེ་ཐམས་ཅད་ཤེ་བར་གྱུར་བའི་རྣམ་པར་གསུངས། ཕྱི་རོལ་པ་ལ་ཐར་ལམ་མེད་པའི་ཤེས་བྱེད་ལ་དཔང་ནས་སྐྱེས་དང། ཉེ་རྒྱལ་གཉིས་ཀྱིས་མུ་སྟེགས་ཀྱི་སྟོན་པ་དྲུག་ལ་བདུད་རྩི་བཙལ་བས་མ་བྱུང་བ་གསུངས། ༈ ཉིན་གཉིས་པར་ཡང་གནས་ཁང་བྱི་དོར་བྱེད་པ་ལས་འཕྲོས་ནས། འཕགས་པ་ལམ་ཕྲན་གྱིས་རྡུལ་སྤངས་སོ་དྲི་མ་སྤངས་སོ་ཞེས་འདོན་བཞིན་དགེ་འདུན་གྱི་མཆིལ་ལྷམ་ཕྱིས་པའི་མཐུས་དྲི་མ་སྲབ་པོར་སོང་ནས་མྱུར་དུ་དགྲ་བཅོམ་ཐོབ་པའི་རྟོགས་བརྗོད་དང། མཛེས་དགའ་དགེ་འདུན་གྱི་གཙུག་ལག་ཁང་གི་ཕྱག་བདར་བྱེད་བཅུག་པས་འདོད་ཆགས་སྲབ་པའི་ལོ་རྒྱུས་གསུངས། མཆིལ་ལྷམ་ཞེས་པ་ནི་འཕགས་ཡུལ་ཚད་པ་ཆེ་བས་རྐང་པའི་མཐིལ་ཙམ་མ་གཏོགས་གཞན་ལ་གཡོགས་མ་མེད་པའི་ལྷམ་ཞིག་ཡིན་གསུངས། སྤྱོད་ལམ་གང་བདེར་གནས། ཞེས་པ་ནི། ཉལ་བ་དང། འཁྲས་པ་སོགས་ཀྱིས་གནས་ན་ཏིང་ངེ་འཛིན་སྐྱེ་སླ་ཞིང། ལུས་ཀྱི་ཀུན་སྤྱོད་ཆོས་བརྒྱད་ལྡན་གྱིས་གནས་ན་ཏིང་ངེ་འཛིན་སྐྱེ་དཀའ་བའི་གང་ཟག་གི་རིགས་འགའ་ཞིག་ལ་དགོངས་ནས་འདིར་འཕགས་པ་བ་ལང་བདག་གི་ལོ་རྒྱུས་གསུངས། དེ་འདྲའི་རིགས་ཅན་ལས་གཞན་གྱིས་ཏིང་ངེ་འཛིན་སྒྲུབ་པ་སོགས་ཀྱི་ཚེ་ལུས་ཆོས་བརྒྱད་ལྡན་དུ་བྱས་ནས་དགོས་པ་ཆེན་པོ་ཡོད་པས་ངེས་པར་བྱ། འདིའི་ཤེས་བྱེད་ལ་སློབ་དཔོན་འཕགས་པ་ལྷ་ལ་རྟ་དབྱངས་ཀྱིས་ཆོས་ཐོག་མར་ཉན་པའི་སྐབས། ལག་པ་གཉིས་ཀྱིས་རྐང་པ་ཡ་གཅིག་གི་པུས་མོ་བསྡམས་ནས་ཆོས་ཉན་པ་ན། འཕགས་པ་ལྷས་ཁྱོད་ནད་པ་ཡིན་ནམ་ཞེས་བཀའ་བཀྱོན་པའི་ལོ་རྒྱུས་གསུངས་ཤིང། དེ་ལ་བརྟེན་ནས་རྟ་དབྱངས་ཀྱིས་བླ་མ་༄༅། །ལྔ་བཅུ་པ་བརྩམས་ཚུལ་ཟུར་ཅམ་གསུངས། མདོར་ན་ཆོས་ཉན་པའི་ཚེ་སྤྱོད་ལམ་མཐའ་གཉིས་སུ་མ་ལྷུང་བ་ཞིག་དགོས་ཤིང། མཐའ་གཉིས་ནི་བླ་མ་དང་ཆོས་ལ་བརྙས་ཐབས་སུ་འགྱུར་བ་དང། བདག་ཉིད་ངལ་ཞིང་དུབ་པའི་མཐའ་གཉིས་ཡིན། དེ་ཡང་བྱེ་བྲག་སྨྲ་བའི་ནད་ཚན་ས་སྒྲོག་རི་པ་སོགས་ཀྱི་ལུགས་ལ་ལག་པ་ཐལ་མོ་སྦྱར་ཞིང་རྐང་མཐིལ་དང་ཚང་ར་ཙམ་མ་གཏོགས་ས་ལ་མ་ཐུག་བར་ཆོས་ཉན་དགོས་པས་དེ་འདྲ་བ་ངལ་ཞིང་དུབ་པའི་མཐར་འགྱུར་གསུངས། ཚོགས་ཞིང་གི་གཙོ་བོ་གདན་སེངྒེ་བརྒྱད་ཀྱིས་བཏེག་པ་དང། བཞིས་བཏེག་པའི་ལུགས་གཉིས་ཡོད་ཅིང་སྤྱིར་སེང་གེས་བཏེག་པ་ནི་སེངྒེ་དེས་རི་དྭགས་ཀྱི་ཚོགས་གཞན་ཐམས་ཅད་ཟིལ་གྱིས་གནོན་པ་ལྟར་བླ་མ་ཐུབ་པའི་དབང་པོས་ཀྱང་བདུད་དང། མུ་སྟེགས་སོགས་ཟིལ་གྱིས་གནོན་པའི་མཚོན་བྱེད་ཡིན། སེངྒེ་བཞི་བྱེད་པ་ནི་མི་འཇིགས་པ་བཞི་མཚོན་བྱེད་ཡིན་གསུངས། བརྒྱད་ལ་བཤད་པ་མ་མཛད། དེའི་སྟེང་གི་ཁྲི་རིན་པོ་ཆེས་བྱེད་པ་ནི་གདུལ་བྱའི་རེ་བ་སྐོང་བའི་བརྡ་ཡིན། པདམ་ནི་ཉེས་པས་མ་གོས་པ་དང་ཉི་ཟླ་ནི་ཐབས་ཤེས་མཚོན་པའི་བརྡ་ཡིན་གསུངས། ཚོགས་ཞིང་རྣམས་ཀྱི་གནས་ལྡར་ཡི་གེ་ལྔ་འགོད་པ་ནི་སྐུ་གསུང་ཐུགས་ཡོན་ཏན་འཕྲིན་ལས་ལྔའམ། ཆོས་དབྱིངས་ཡེ་ཤེས་སོགས་ཡེ་ཤེས་ལྔ་མཚོན་བྱེད་ཡིན། རང་གི་མཐའ་སྐོར་དུ་སེམས་ཅན་ཐམས་ཅད་མིའི་རྣམ་པར་སྒོམ་ཀྱང་ཆོག་ལ། ཡང་ན་སོ་སོའི་རྣམ་པ་མ་བསྒྱུར་བར་གྲོག་སྦུར་སོགས་ཀྱི་རྣམ་པ་དེ་ཀར་གནས་བཞིན་པའི་དང་ལ། ཚོགས་ཞིང་རྣམས་ཀྱི་བྱིན་རླབས་བསམ་གྱིས་མི་ཁྱབ་པའི་སྟོབས་ཀྱིས་སྡིག་སྒྲིབ་དག་པའི་མོས་པ་བྱེད་གསུངས། མྱུར་བ་མྱུར་བར་ཞེས་པའི་མྱུར་བ་དང་པོ་ནི་བསྐལ་ཆེན་གྲངས་མེད་གསུམ་ལ་བལྟོས་མི་དགོས་པའི་དོན་དང། མྱུར་བ་གཉིས་པ་ནི་སྙིགས་དུས་ཀྱི་ཚེ་ཐུང་གཅིག་ལ་སངས་རྒྱས་བའི་དོན་ཡིན། ཡང་དབེན་ས་བཀའ་བརྒྱུད་རང་གི་མཐུན་མོང་མིན་པའི་བཞེད་སྲོལ་ལ་མྱུར་བ་དང་པོ་སྙིགས་དུས་ཀྱི་ཚེ་ཐུང་གཅིག་དང། མྱུར་བ་གཉིས་པ་ནི་བླ་མའི་རྣལ་འབྱོར་ལམ་གྱི་སྲོག་ཤིང་དུ་བྱས་ནས་སྒོམ་པའི་ཐབས་མཁས་ཀྱི་མཐུས་མི་ལོ་བཅུ་གཉིས་ཚུན་ལ་རྡོ་རྗེ་འཆང་གི་གོ་འཕང་ཐོབ་ནུས་པའི་དོན་ཡིན་གསུངས། བླ་མའི་རྣལ་འབྱོར་ལམ་གྱི་སྲོག་ཏུ་བྱེད་ལུགས་ནི༑ ཉམས་ལེན་གང་བྱེད་ཀྱི་གནས་སྐབས་ཀུན་ཏུ་བླ་མ་བསྒོམ་བཞིན་པའི་ངང་ནས་བྱེད་པའི་དོན་ཡིན། གནད་དེ་སྟོན་ཚུལ་ནི་ཚོགས་ཞིང་གསལ་འདེབས་པའི་ཚེ་ཡང་མདོ་སྔགས་གཉིས་ཀ་ནས་འབྱུང་བའི་ཚོགས་ཞིང་གསལ་འདེབས་པ་དང་སྡུད་པའི་ཚེ་རྒྱ་ཆེན་སྤྱོད་བརྒྱུད་རྣམས་བྱམས་པ། ཟབ་མོ་ལྟ་བརྒྱུད་རྣམས་འཇམ་དབྱངས་ཆོས་འབྲེལ་ཡོད་པའི་བླ་མ་རྣམས་རྩ་བའི་བླ་མ་བྱིན་རླབས་ཉེ་བརྒྱུད་རྣམས་རྡོ་རྗེ་འཆང་ལ་བསྡུ། མཐར་གཞན་རྣམས་གཙོ་བོ་ལ་ཐིམ་ཞིང༑ རྡོ་རྗེ་འཆང་གཙོ་བོའི་སྤྱི་གཙུག་ཏུ་བྱོན་ནས་གཙོ་བོ་ལ་ཐིམ་པས་ཐུབ་དབང་དང་རྡོ་རྗེ་འཆང་དབྱེར་མེད་དུ་གྱུར། སླར་ཐུབ་པའི་དབང་པོ་ཡང་རང་ལ་ཐིམ་པས་དབྱེར་མེད་དུ་བྱེད་ཅིང། དེ་འདྲའི་ཐུབ་དབང་དེ་རྣམ་པ་ཐུབ་པའི་༄༅། །དབང་པོའི་རྣ་མ་པ་ཅན་ཡིན་ཀྱང། དོན་ལ་རང་གི་དྲིན་ཅན་རྩ་བའི་བླ་མ་དེ་རང་ཡིན་པར་སྒོམ་དགོས་པའི་གནད་ཀྱིས་ཡིན་གསུང་བ་འདྲ་ཞིག་གོ་བྱུང། འདི་སྐོར་ལས་འཕྲོས་ནས། ཡེ་ཤེས་ཇི་སྙེད་ཁྱབ་གྱུར་པ། ། སྐུ་ཡིས་ཀྱང་ནི་ད་སྙེད་ཁྱབ། ཅེས་གསུངས་བས་ལོངས་སྐུའི་མཁྱེན་པས་གང་དུ་ཁྱབ་པར་སྐུས་ཀྱང་ཁྱབ་སྐུ་དང་མཁྱེན་པ་གཉིས་ནི་ངོ་བོ་གཅིག་ཡིན། དེས་ན་སྤྲུལ་སྐུ་གང་དུ་བཞུགས་པ་དེ་ཀ་རང་ལ་ལོངས་སྐུ་ཡང་བཞུགས་ཟེར་དགོས་པ་འདྲ་མོ་ཞིག་ཀྱང་འདུག་ཅེས་པའི་དོགས་སློང་ཞིག་ཀྱང་མཛད་བྱུང། གོང་གི་སེང་ཁྲི་སོགས་ཀྱི་བརྡ་མཚོན་ལས་འཕྲོས་ནས་བརྡའི་སྒོ་ནས་ཆོས་སྟོན་པའི་དཔེ་ལ་ཨོ་རྒྱན་གྱི་ཡུལ་གྱི་དཔལ་ཕག་ཆང་བའི་རྣམ་ཐར་ཟུར་ཙམ་ཞིག་གསུངས། གཞན་ཡང་རྒྱལ་པོ་མ་སྐྱེས་དགྲའི་འགྱོད་པ་སེལ་བའི་མདོ་ལས་འབྱུང་བ། རྒྱལ་པོས་རྗེ་བཙུན་འཇམ་དབྱངས་འཁོར་དང་བཅས་པ་གདུགས་ཚོད་ལ་སྤྱན་དྲངས་ཤིང་གདུགས་ཚོད་གསོལ་བའི་མཐར་རྗེ་བཙུན་འཇམ་དབྱངས་ལ་རིན་ཐང་ཚད་མེད་པའི་གོས་རིན་པོ་ཆེ་ཞིག་འབུལ་བར་བརྩམས་པ་ན། འཇམ་དབྱངས་ཡལ། དེ་ནས་འོད་སྲུང་ཆེན་པོ་ལ་འབུལ་བར་བརྩམས་པ་ན་དེ་ཡང་ཡལ། མཐར་གཞན་སུ་ཡང་མི་དམིགས་པ་ན་རྒྱལ་པོ་རང་ཉིད་ཀྱིས་གྱོན་པས་རང་གི་ལུས་ཀྱང་ཡལ་བ་ན། སྟོང་པ་ཉིད་ཀྱི་དོན་རྟོགས་པའི་སྔོན་བྱུང་ཡང་གསུངས། ཕྱག་ལ་དབྱེ་ན་ལུས་ངག་ཡིད་གསུམ་གྱི་ཕྱག་གསུམ་ཡོད་ཅིང་ཐལ་མོ་སྤྱི་བོ་མགྲིན་པ་སྙིང་ཁ་གསུམ་དུ་སྦྱར་བས་ལུས་ངག་ཡིད་གསུམ་གྱི་ཕྱག་མཚོན། ཡང་སྤྱི་བོར་སྦྱར་བས་སྤྱི་གཙུག་ལྟར་མི་མངོན་པའི་གཙུག་ཏོར་འགྲུབ་པའི་ནུས་བ་འཇོག །མཎྜལ་གྱི་རྒྱ་རབ་གསེར་དངུལ་སོགས་རིན་པོ་ཆེ་ལས་གྲུབ་པའི་མཎྜལ་འབུལ་བར་གསུངས་པ་ནི། གསེར་དངུལ་སོགས་རིན་པོ་ཆེ་ཡིན་པའི་རྒྱུ་མཚན་ཙམ་གྱིས་མིན། ལོངས་སྤྱོད་རབ་ཡོད་པ་ཞིག་གིས་གསེར་དངུལ་སོགས་ཀྱི་མཎྜལ་འབུལ་དགོས་ཞེས་པའི་དོན་ཡིན། མཎྜལ་གྱི་ཚོམ་བུ་འགོད་པའི་ཚེ། ཤར་གྱི་ཚོམ་བུ་བླ་མ་སོགས་འབུལ་ཡུལ་གྱི་ཕྱོགས་སུ་འགོད་པ་ནི། དེ་རྣམས་ཀྱི་ཡོན་ཏན་དང་བྱིན་རླབས་རང་ལ་འཇུག་པའི་ཆེད་ཡིན། ཤར་གྱི་ཚོམ་བུ་རང་ཉིད་ཀྱི་ཕྱོགས་སུ་འགོད་པ་ནི། རང་གི་ལུས་ལོངས་སྤྱོད་སོགས་བླ་མ་དང་ཚོགས་ཞིང་རྣམས་ལ་འབུལ་བའི་ཆེད་ཡིན་གསུངས། བཤགས་པ་བྱེད་པའི་ཚེ་ཕྱིན་ཆད་ལ་སྡོམ་སེམས་མེད་ན་གླང་པོ་ཆེའི་ཁྲུས་དང་འདྲ་ཚུལ་ནི་གླང་པོ་ཆེ་ཚད་པས་གདུངས་པའི་ཚེ་ཆུ་ནང་དུ་ཞུགས་ནས་ཁྲུས་བྱེད། ཁྲུས་ཟིན་ནས་ལུས་ཀྱི་གཤེར་དེ་བསྐམ་པའི་ཕྱིར་བྱེ་མ་ལ་འདྲེ་ལྡོག་བྱེད། ཡང་བྱེ་མ་འགོས་པ་དེ་ཁྲུ་བའི་ཕྱིར་ཆུ་ནང་དུ་ཞུགས་ནས་ཁྲུས་བྱེད། ཡང་ཁྲུས་བྱས་པའི་གཤེར་དེ་བསྐམ་པའི་ཕྱིར་བྱེ་མ་ལ་འདྲེ་ལྡོག་བྱེད་པ་དང་འདྲ་གསུངས། ཡང་རྒྱ་ཆེན་སྤྱོད་པའི་ལམ་དང་དེའི་སྙིང་པོ་དང་ཡང་སྙིང་གསུམ་ཡོད། ཕྱི་མ་ནི་སློབ་མི་སློབ་ཀྱི་ཟུང་འཇུག་ལྟ་བུ་ལ་བྱེད། རྡུལ་གཅིག་སྟེང་ན་རྡུལ་སྙེད་སངས་རྒྱས་རྣམས་ཞེས་པའི་སྐབས་སུ་སངས་རྒྱས་དང་ས་མཐོན་པོ་ལ་གནས་པའི་༄༅། །བྱང་སེམས་ཀྱིས་རྡུལ་ཕྲ་རབ་གཅིག་གི་ནང་དུ་ཕྱོགས་བཅུའི་རྡུལ་སྙེད་ཀྱི་སངས་རྒྱས་ཀྱི་ཞིང་ཁམས་ཤོང་བར་མཛད་ནུས་པ་ནི། སངས་རྒྱས་བྱང་སེམས་དེ་རྣམས་ཀྱིས་ཆོས་རྣམས་ཀྱི་གནས་ལུགས་སྟོང་པ་ཉིད་མངོན་སུམ་དུ་རྟོགས་ཤིང། རྡུལ་གཅིག་གི་ཆོས་ཉིད་དང་འཇིག་རྟེན་གྱི་ཁམས་ཀྱི་ཆོས་ཉིད་གཉིས་ལ་རྒྱ་ཆེ་ཆུང་མེད་པའི་སྟབས་ཀྱིས་ཡིན་གསུངས་ནས། ཤེས་བྱེད་ལ། གང་ལ་སྟོང་པ་ཉིད་རུང་བ། །དེ་ལ་ཐམས་ཅད་རུང་བར་འགྱུར། །གང་ལ་སྟོང་ཉིད་མི་རུང་བ། །དེ་ལ་ཐམས་ཅད་རུང་མི་འགྱུར། ཞེས་འདྲེན་པར་མཛད། ༈ ཉིན་གསུམ་པར་ཐོག་མར། ཚུལ་གནས་ཐོས་དང་བསམ་གཏན་པས་ཞེས་སོགས་དྲངས་ནས། གསུང་རབ་ཀྱི་ཚིག་ཐོས་ནས་འདིའི་དོན་ནི་འདིའོ་སྙམ་ནས་ཡིད་དཔྱོད་ཀྱིས་གོ་བ་ནི་ཐོས་བྱུང། རྒྱུ་མཚན་ཡང་དག་ལ་བརྟེན་ནས་དོན་དེ་ལ་ངེས་ཤེས་འདྲེན་ནུས་པ་ནི་བསམ་བྱུང་ཡིན་གསུངས། ཡང་སངས་རྒྱས་ཞལ་བཞུགས་པའི་དུས་ཉན་ཐོས་འགའ་ཞིག་གིས༑ བདག་གིས་སངས་རྒྱས་ཀྱི་ཆོས་ཀྱི་སྐུ་མཐོང་ཡང་གཟུགས་ཀྱི་སྐུ་མ་མཐོང་བས་བལྟར་འགྲོ། ཞེས་པའི་སྐབས་ཀྱི་ཆོས་སྐུ་ནི་བདག་མེད་སྟོང་པ་ཉིད་ཀྱི་དོན་བསྒོམ་པའི་སྟོབས་ཀྱིས་ཉོན་མོངས་ཟད་པའི་ཆ་ལ་བྱེད་པ་ཡོད་དེ། དེ་ཙམ་མིན། རྒྱུ་མཚན་གང་གི་ཕྱིར་ན། ཉན་ཐོས་སློབ་པ་འགའ་ཞིག་གིས་ཀྱང་བདག་གིས་སངས་རྒྱས་ཀྱི་ཆོས་ཀྱི་སྐུ་མཐོང་ཞེས་སོགས་ཟེར་བས་སོ། །འོ་ན་དེ་འདྲའི་ཆོས་སྐུ་དེ་གང་ལ་བྱེད་ཅེ་ན། སོ་སོར་ཐར་པ་དང། སོ་སོར་ཐར་པའི་མདོ་གཉིས་ལ་བྱེད་དགོས་གསུངས། དེའི་ཤེས་བྱེད་ལ་སོ་སོར་ཐར་པའི་མདོ་ལས། ང་ནི་མྱ་ངན་འདས་གྱུར་ན། །འདི་ནི་ཁྱོད་ཀྱི་སྟོན་པ་ཞེས། ཞེས་པ་འདི་འདྲེན་པར་མཛད། ཡང་སངས་རྒྱས་ཀྱིས་ཞིང་འདིར་གདུལ་བྱ་རྣམས་ལ་གཙོ་བོར་སྒྲའི་སྒོ་ནས་ཆོས་སྟོན་པ་བཞིན་དུ། སངས་རྒྱས་ཀྱི་ཞིང་འགའ་ཞིག་ཏུ་གཙོ་བོར་དྲི་དང་རོ་དང་རེག་བྱའི་སྒོ་ནས་ཆོས་སྟོན་པ་ཡོད་གསུངས། དྲིའི་སྒོ་ནས་ཆོས་སྟོན་པ་ཡོད་པའི་ཤེས་བྱེད་སྐབས་ལི་ཙ་བྱི་དྲི་མེད་གྲགས་པའི་རྟོགས་བརྗོད་གསུངས། བཤེས་གཉེན་བསྟེན་ཚུལ་སྐབས་འཕགས་པ་ཐོགས་མེད་ཀྱིས་རྗེ་བཙུན་བྱམས་པ་ཁྱི་མོར་མཐོང་བའི་སྐབས་ལས་འཕྲོས་ཏེ། འཇམ་དཔལ་རྩ་རྒྱུད་ལས། ས་བླའི་ཕོ་ཉ་མོ་ཞེས་བརྗོད་ཅེས་པའི་སྐབས་ཀྱི་ས་བླའི་པོ་ཉ་ནི་འཕགས་པ་ཐོགས་མེད་ལ་དང་པོར་ཞལ་གཟིགས་པའི་རྗེ་བཙུན་དེ་ཡིན་གསུངས་པ་འདྲ་ཞིག་བྱུང་ཡང་སྐབས་འདིར་གྲོང་ཁྱེར་སྒྲ་སྒྲོགས་ཀྱི་གཙུག་ལག་ཁང་གི་ཕྱི་རོལ་དུ། ལྷའི་དབང་པོ་བརྒྱ་བྱིན་གྱིས་ཞབས་འབྲིང་པ་བྱས་ནས། སངས་རྒྱས་འཆག་ཅིང་བཞུགས་པའི་སྐབས་དགེ་སློང་ལེགས་སྐར་ཉལ་བར་རིངས་ནས་སྒྲ་སྒྲོགས་ཀྱི་ཡུལ་གྱི་གནོད་སྦྱིན་བོང་བུ་ཞེས་པར་བརྫུས་ནས་སངས་རྒྱས་ལ་སྡིག་རི་བསྐུལ་བའི་ལོ་རྒྱུས་གསུངས། ཡང་སློབ་དཔོན་སངས་རྒྱས་ཡེ་ཤེས་ཀྱིས་སློབ་དཔོན་འཇམ་དཔལ་བཤེས་གཉེན་བསྟེན་ཚུལ་དང་མི་ལས་མར་པ་བསྟེན་ཚུལ་རྣམས་ཆེས་ངོ་མཚར་མོད་ཀྱང། ནཱ་རོས་ཏཻ་ལོ་པ་བསྟེན་ཚུལ་དང་གུ་རུ་ཞི་བ་སྦས་པས་བསྟེན་ཚུལ་རྣམས་སྔ་མ་དེ་རྣམས་ལས་༄༅། །ཆེས་ལྷག་ཅིང་བྱུང་ན་དེ་འདྲ་ཞིག་དགོས་ཞེས་རྗེ་བློ་བཟང་བཟོད་པའི་ཞལ་སྔ་ནས་གསུང་གསུངས། ༈ ཉིན་བཞི་པ་ལ་ཐོག་མར། རྣམ་པ་ཐམས་ཅད་མཁྱེན་ཉིད་ལམ། ཞེས་པ་ནས། དགོས་པ་ཡིན་ཞེས་པའི་བར་དྲངས་ནས། སངས་རྒྱས་རྣམས་ཀྱིས་ཐབས་རྣམ་མཁྱེན་ལ་བརྟེན་ནས། གདུལ་བྱ་རྣམས་ཀྱི་ཁམས་དང་བསམ་པ་དང་བག་ལ་ཉལ་ཇི་ལྟ་བ་བཞིན་གཟིགས་ནས། དེ་དང་འཚམས་པའི་ཆོས་སྟོན་ཞེས་པའི་སྐབས་ཀྱི་ཁམས་ལ་ཡང་ཁམས་དྲུག་ཡོད། དེ་ཡང་སེམས་ཅན་འགའ་ཞིག་གི་ལུས་སའི་ཁམས་ཤས་ཆེ་བ་དང། འགའ་ཞིག་ཆུའི་ཁམས་ཤས་ཆེ་བ་སོགས་ཡིན། གཟུགས་མེད་ཀྱི་དགྲ་བཅོམ་འགའ་ཞིག་ཀྱང་སངས་རྒྱས་ཀྱི་མདུན་དུ་ཆོས་ཉན་པ་ཡོད་དེ། དེའི་ཚུལ་ཡང་དགྲ་བཅོམ་དེ་རྣམས་ཀྱི་སེམས་སྟེང་ན་སྔོན་ཐོག་མ་མེད་པ་ནས་གཟུགས་ཅན་དུ་ཡང་ཡང་སྐྱེས་དུས་ཀྱི་དབང་པོ་ལྔའི་བག་ཆགས་ཡོད་པ་དེ། སངས་རྒྱས་ཀྱི་འོད་ཟེར་གྱིས་རེག་པའི་མཐུས་སད་ནས་ཆོས་ཉན་པ་ཡིན། དེ་ལ་གཟུགས་མེད་ཀྱང་སངས་རྒྱས་ཀྱི་འོད་ཟེར་གྱིས་རེག་ནུས་པ་ནི་བསམ་གྱིས་མི་ཁྱབ་པའི་མཐུའི་ཁྱད་པར་ཡིན་གསུངས། ཡང་སངས་རྒྱས་ཀྱི་མདུན་དུ་གཟུགས་ཁམས་ཀྱི་ལྷ་ཞིག་ཆོས་ཉན་དུ་འོངས་པའི་ཚེ། བྱེ་ཚན་གྱི་སྟེང་དུ་མར་གསར་བཞག་པ་བཞིན་ལུས་ས་ལ་བྱིང་བ་ན། དེ་ལན་མ་མཁའི་འདུ་ཤེས་སྒོམ་དུ་བཅུ་ག་པས་ལམ་བྱིང་བ་ཡང་ཁམས་ཀྱི་ཁྱད་པར་གྱིས་ཡིན་གསུང་བ་འདྲ་ཞིག་ཀྱང་གོ་བྱུང། གཟུགས་མེད་ཀྱི་གང་ཟག་ཚེ་མ་འཕོས་བཞིན་དུ་ཆོས་ཉན་པ་ཡོད་པ་དཀོན་མཆོག་བརྩེགས་པ་སོགས་ལས་ཀྱང་གསུངས། སྐབས་འདིའི་བསམ་པ་ནི་ང་རྒྱལ་དང། ཕྲག་དོག་སོགས་ཀྱི་བསམ་པ་གང་ཤས་ཆེ་བདེ་ལ་བྱེད། བསམ་པ་དེ་འདྲ་བའི་གང་ཟག་གི་བསམ་པ་དང་འཚམས་པར་ཆོས་བསྟན་ནས། འདུལ་ཚུལ་གྱི་དཔེར་བརྗོད་ལ། ལྟེང་རྒྱས་འོད་སྲུངས་ཀྱིས་རང་ཉིད་དགྲ་བཅོམ་པ་ཡིན་སྙམ་པའི་མངོན་པའི་ང་རྒྱལ་གཞོམ་ཕྱིར་དུ་སྒྱུ་འཕྲུལ་རིམ་པར་བསྟན་ནས་སླར་དགྲ་བཅོམ་ལ་བཀོད་པའི་བར་གྱི་གཏམ་རྒྱུད་རྒྱས་པར་གསུངས། གཞན་ཡང་བྲམ་ཟེ་པདམའི་སྙིང་པོ་དང། གཙུག་ཕུད་ཅན་གཉིས་ཀྱི་གཏམ་རྒྱུད་ཀྱང་བྱུང། དགེ་བའི་བཤེས་གཉེན་བསམ་པས་བསྟེན་ཚུལ་གྱི་སྐབས་སུ་བླ་མའི་ངོས་ནས་ཡོན་ཏན་ཆེ་ཆུང་སོགས་ཇི་ལྟར་ཡིན་ཀྱང། སློབ་མའི་ངོས་ནས་ཡིད་ཆེས་ཡོན་ན་རྟོགས་པ་ཁྱད་པར་ཅན་སྐྱེ་བའི་དཔེར། སྔོན་མ་བསད་པའི་མཚམས་མེད་བྱས། ཕྱིས་བསྙེན་པར་རྫོགས་ནས་དགེ་སློང་གི་སྡོམ་པ་མ་སྐྱེས་ཀྱང། མིང་དགེ་སློང་བདེ་བྱེད་ཟེར་བས་མཐའ་འཁོབ་ཏུ་དགེ་འདུན་གྱི་སྡེ་སྤེལ་བའི་ཕྱིར་རང་ཉིད་ཀྱིས་མཁན་སློབ་བྱས་ནས་རབ་བྱུང་བསྙེན་རྫོགས་བྱས་པ་རྣམས་ཡིད་ཆེས་ཀྱི་དབང་གིས་སྡོམ་པ་སྐྱེས་པར་མ་ཟད་དགྲ་བཅོམ་ཡང་ཐོབ་པ་བྱུང་བ་དང། བདེ་བྱེད་རང་ཉིད་བསྲོ་ཁང་ལ་གནས་བཞིན་པའི་ངང་ནས་ཚེ་འཕོས་ནས་དམྱལ་པར་སྐྱེས་ཀྱང། སྔར་དགེ་འདུན་གྱི་སྡེ་སྤེལ་བའི་མཐུས་དམྱལ་བ་ལས་མྱུར་དུ་ཐར་ཏེ་རྒྱལ་ཆེན་རིགས་བཞི་པའི་ལྟར་སྐྱེས་ནས། སངས་རྒྱས་ཀྱིས་ཆོས་བསྟན་པས་བདེན་པ་མཐོང་༄༅། །པའི་གཏམ་རྒྱུད་རྒྱས་པར་གསུངས། སྦྱོར་བས་བསྟེན་ཚུལ་གྱི་སྐབས། འབྲོམ་རིན་པོ་ཆེས་ཇོ་བོ་སེ་བཙུན་གྱི་ཕྱག་ཕྱི་མཛད་པའི་རྟོགས་བརྗོད་གསུངས། གཞན་ཡང་རང་གི་ལུས་ངག་ཡིད་གསུམ་བླ་མ་ལ་འབུལ་ཞིང། ཕུལ་ནས་ལུས་ཀྱིས་བླ་མའི་གཡོག་བྱེད། ངག་གིས་བླ་མའི་ཡོན་ཏན་སོགས་བརྗོད། ཡིད་ཀྱིས་བླ་མས་གང་གསུངས་ཉམས་སུ་ལེན་པར་བྱེད་གསུངས། དབང་པོའི་སྒོ་སྡོམ་པའི་སྐབས་ཉོན་མོངས་གང་ཤས་ཆེ་ཞིང་འབྱུང་ཉེ་བ་དེའི་ཡུལ་དང་མ་ཐུག་པ་བྱེད་དགོས་ཤིང་ཡང་ན་ནི་རལ་སྐོར་མཁན་གྱི་དཔེ་བཞིན་ཉོན་མོངས་གང་སྐྱེས་བྱུང་བ་དེ་ཐད་ཀ་ཐད་ཀར་འབུར་འཇོམས་གཏོང་བ་ཡིན། ཕྱི་མ་འདི་གང་ཟག་དབང་སྣོན་གྱི་ལུགས་ཡིན། རལ་གྲིའི་འཁོར་ལོ་ཞེས་པ་འདི་འཁོར་ལོ་གཅིག་ལ་རལ་གྲིའི་རྩིབས་མང་པོ་ཡོད་པ་ཞིག་མིན་ཀྱང། སྐོར་བ་པོའི་མྱུར་ཁྱད་ཀྱི་དབང་གིས་རྩིབས་ལྔ་པ་ནས་བརྒྱ་རྩ་བརྒྱད་པའི་བར་གྱི་ཐ་སྙད་བྱེད་པ་ཡོད་གསུངས། ཡང་བླ་མ་བསྟེན་ཚུལ་སྐོར་ལ། ནཱ་རོའི་རྣམ་ཐར་ཟུར་ཙམ་ཞིག་གསུངས་པའི་སྐབས། ནཱ་རོ་པ་སྲིན་བུ་པད་པས་གང་བའི་ཆུ་ནང་དུ་ལྷུང་ཞེས་པའི་སྲིན་བུ་པད་པ་ལ་ཐང་ལ་གནས་པ་དང་ཆུ་ལ་གནས་པ་གཉིས་ཡོད་ཅིང་ཆུ་ལ་གནས་པ་དེས་མི་སོགས་ཀྱི་ལུས་ཀྱི་ནང་དུ་ཞུགས་ནས་ཁྲག་འཐུང་བར་བྱེད་ཤེས་རྒྱུན་པར་ལུས་ཁབ་ཙམ་ལས་ཀྱང་མེད་ཀྱང། མིའི་ལུས་ལ་ཞུགས་ནས་ཁྲག་གཞིབ་སྟེ་ཕྱིར་བྱུང་བ་ན་མཛུབ་མོ་ཙམ་དུ་འགྲོ་ལ་དེ་བོད་ན་ནི་མེད་གསུངས། ཡང་དྲུག་སྡེ་སངས་རྒྱས་ཀྱི་སྤྲུལ་བ་ཡིན་ཚུལ་ཡང་ཡང་གསུངས། ༈ ཉིན་ལྔ་པར། གཞན་འབྱོར་ལྔའི་ནང་གི་སངས་རྒྱས་བྱུང་བ་དང་ཞེས་པ་ནི་རང་ཉིད་སྐྱེས་པའི་བསྐལ་པ་དེར་སངས་རྒྱས་བྱུང་བ་ཙམ་མིན་གྱིས། རང་མིར་སྐྱེ་བ་བླང་བའི་དུས་དེར་སངས་རྒྱས་ཞལ་བཞུགས་པ་ཞིག་ལ་བྱེད། བསྟན་པ་གནས་ཞེས་པ་ཡང་ཉན་པའི་དབང་དུ་བྱས་ནས། དོན་དམ་པའི་བསྟན་པ་གནས་པ་ལ་བྱེད་ཅིང། དེ་ཡང་རྒྱུན་ཞུགས་ཐོབ་པ་སོགས་བདེན་པ་གསར་དུ་མཐོང་བ་ཡོད་པ་ལ་བྱེད། དེས་ན་དལ་འབྱོར་ཚང་ན་དལ་འབྱོར་མཚན་ཉིད་པ་ཚང་བས་མ་ཁྱབ་གསུངས། འཇིག་ཚོགས་ལ་བལྟ་བའི་རིའི་རྩེ་མོ་མཐོ་བ་ཉི་ཤུ་བཅོམ་ཞེས་མཐོ་བ་སྨོས་པའི་རྒྱུ་མཚན་ནི། དཔེར་ན་རི་ཤིན་ཏུ་མཐོ་བ་ཞིག་ཡོད་པ་དེའི་རྩེ་ནས་མར་ཕྱེད་ཙམ་དུ་བཅོམ་ན་རིའི་རྩེ་མོ་མཐོ་བ་བཅོམ་ཞེས་པ་ལྟར། འཇིག་ལྟ་དེ་ལ་ཕྲ་རགས་གཉིས་ཡོད་པའི་རགས་པ་རྟག་ཅིག་རང་དབང་ཅན་གྱིས་བདག་ཏུ་འཛིན་པ་དེ་བཅོམ་པ་ན། འཇིག་ལྟ་དེའི་ཕྱོགས་གཅིག་བཅོམ་པས་མཐོ་བ་བཅོམ་ཞེས་གསུངས་པ་ཡིན་གསུངས། སྔོན་ནས་མི་རྟག་པ་གོམས་མྱོང་པའི་བག་ཆགས་ཁྱད་པར་ཅན་ཡོད་པ་ཞིག་ཡིན་ན། མི་རྟག་པའི་ཆོས་ཅུང་ཟད་ཅིག་བསྟན་པ་ན་དེ་མ་ཐག་མི་རྟག་པ་ལ་གོ་བ་ཁྱད་པར་ཅན་སྐྱེ་བའི་དཔེར། རྒྱལ་པོ་མྱ་ངན་མེད་ཀྱིས་དགེ་འདུན་མང་པོ་གདུགས་ཚོད་ལ་སྤྱན་དྲངས་པ་ན། བཤུལ་དུ་དགེ་སློང་གློ་བུར་བ་དྲུག་ཅུ་ཡོངས་ཤིང། དེ་རྣམས་ལ་ཡང་བཟའ་བ་དང་བཅའ་བ་མང་པོ་བསྟབ་བན་གྲལ་མགོར་འདུག་པའི་དགེ་སློང་ལ། དེའི་གྲལ་གཞུག་ན་འདུག་པ་དེས་༄༅། །ཁྱོད་ཀྱིས་མ་བརྟགས་པར་གྲལ་མགེར་བསྡད་པ་མ་ལེགས། རྒྱུ་མཚན་གང་གི་ཕྱིར་ན་ད་ལྟ་རྒྱལ་པོ་ཆོས་ཉན་པའི་ཕྱིར་འོང་བས་དེ་ལ་ཁྱེད་ཀྱི་ཇི་ལྟར་བྱེད་ཅེས་སྨྲས་པས། གྲལ་མགོའི་དགེ་སློང་དེ་ཇི་ལྟར་བྱེད་སྙམ་དུ་སེམས་ཁོང་དུ་ཆུད་ཅིང་འདུག་པ་ན། ཀུན་དགའ་ར་བ་དེར་བསྟན་པ་ལ་དགའ་བའི་ཤིང་གི་ལྷ་ཞིག་ཡོད་པ་དེས། དགེ་སློང་དེ་ལ་ཁྱེད་འཇིགས་མི་དགོས་ད་གཟོད་རྒྱལ་པོ་ཆོས་ཉན་དུ་བྱུང་དུས། རྒྱལ་པོ་ཆེན་པོ་ས་གཞི་རི་དང་བཅས་པ་འདི་ཡང་འཇིག་པར་འགྱུར་ན་ཁྱོད་ཀྱི་རྒྱལ་སྲིད་ལྟ་ཅི་སྨོས། དེ་བས་ན་དོན་འདི་བསམ་པར་རིགས་སོ་སྨྲོས་ཤིག་ཅེས་སྨྲས་སོ། །དེ་ནས་རྒྱལ་པོ་དེས་དགེ་སློང་དེ་ལ་རས་ཡུག་ཆེན་རིན་ཐང་ཆེ་བ་ཞིག་ཕུལ་ནས་ཆོས་ཉན་པར་བརྩམས་པ་ན། དགེ་སློང་དེས་ཀྱང་སྔར་ཤིང་གི་ལྷས་བསླབས་པ་ལྟར་སྨྲས་པ་ན་རྒྱལ་པོ་དེ་མ་ཐག་ཕོ་བྲང་དུ་ལོག་ནས་དེ་ཉིད་སེམས་ཤིང་ངེས་པ་རྙེད་པའི་ལོ་རྒྱུས་གསུངས། ཡང་བྱང་ཆུབ་སེམས་དཔའ་ཞིག་གིས་དཔེར་ན་བྱེ་བྲག་སྨྲ་བའི་གཞུང། དོན་དམ་དོན་བྱེད་ནུས་པ་གང་ཞེས་སོགས་ཀྱི་གཞུང་འདོན་པ་དང་དེའི་དོན་ཆེད་དུ་བྱ་བ་སེམས་ཅན་གཞན་གྱི་དོན་དུ་སེམས་པར་བྱེད་པ་ངེས་པར་ཡོད་ཅིང། དེ་ལྟ་ན་དེ་འདྲའི་གཞུང་དང་དེའི་དོན་གྱི་བདེན་གཉིས་ཀྱི་འཇོག་ཚུལ་སེམས་པའི་བློ་སོགས་ཐེག་ཆེན་གྱི་ཆོས་ཡིན་ནམ་མིན་ཞེས་སོགས་ཀྱི་དཔྱད་པ་ཞིག་ཀྱང་གསུངས། ཡང་མཛོད་བསླབ་བཏུས་དང་ལམ་རིམ་ཆེན་མོ་རྣམས་སུ་དམྱལ་བ་ཡང་གསོས་ཀྱི་ས་གཞི་ལྕགས་ཀྱི་ས་གཞི་ཡིན་པར་མ་གསུངས་ཀྱང། དྲན་པ་ཉེར་གཞག་ཏུ་ལྕགས་ཀྱི་ས་གཞིར་གསུངས། ཡང་དམྱལ་སྲུངས་རྣམས་ཀྱང་སེམས་ཅན་དེ་སྔོན་གཏི་མུག་ཤས་ཆེ་བའི་སྟོབས་ཀྱིས་མི་དགེ་བའི་ལས་བསགས་པ་ཞིག་ཡིན་ན་ཕག་གི་གདོང་པ་ཅན་དང། ཞེ་སྡང་གིས་ཡིན་ན་སྦྲུལ་དང་སེང་གེ་དང་སྟག་སོགས་ཀྱི་མགོ་བརྙན་ཅན་འབྱུང་བར་གསུངས། ཡང་ཤར་ཆེན་རིན་ཆེན་རྒྱལ་མཚན་པས། རྒྱལ་ཚབ་རིན་པོ་ཆེར་སྦྱིན་བདག་གི་ངོ་འཛིན་གནང་དགོས་ཚུལ་ཞུས་པའི་ཚེ་དབུ་ཞྭ་ཕུད་ནས་དབུ་དཀར་པོ་ཡིན་པ་དེ་བསྟན་ནས་ངའི་མགོ་འདི་ལ་ལྟོས་དང་དེ་འདྲའི་ལོང་འདུག་གམ་ཞེས་གསུངས་པའི་ལོ་རྒྱུས་དང། ཡང་འབྲོམ་རིན་པོ་ཆེས་རྭ་སྒྲེང་གི་ཤུག་གསེབ་ཏུ་ཤངས་སྣབས་ཀྱང་ཕྱིས་ལོང་མེད་པར། འཇིག་རྟེན་མཁྱེན་པ་རྙེད་དང་མ་རྙེད་དང། ཞེས་སོགས་ཀྱི་ཚིགས་བཅད་འདོན་གྱིན་མི་རྟག་པ་སོགས་བསྒོམས་པའི་ལོ་རྒྱུས་ཀྱང་གསུངས། ༈ ཉིན་དྲུག་པར། ཐོག་མར། བདག་ནི་སངས་རྒྱས་ཕྱོགས་མི་འཛིན། ཞེས་སོགས་དྲངས་ནས་སྤྱི་བཤད་མཛད་རྗེས། དམྱལ་བའི་སྡུག་བསྔལ་གྱི་སྐབས་དཔེར་ན་ཐིག་ནག་ལྟ་བུར་ལུས་གན་རྐྱལ་ལམ་ཁ་སྤུབ་ཏུ་བསྙལ་ནས་ལྕགས་ཀྱི་སྐུད་པ་མེ་འབར་བ་ཐིག་བརྒྱད་སོགས་བཏབ་ནས་སོག་ལེ་མེ་འབར་བས་གཤག་ཅིང་ལུས་དུམ་བུར་བྱས་པ་ན་དུམ་བུ་སོ་སོས་ཀྱང་སྡུག་བསྔལ་མྱོང་ཞིང། དེར་མ་ཟད་ཁྲག་དང་ཆུ་སེར་ས་ལ་ཐིག་པ་དེ་དག་ལ་བརྟེན་ནས་ཀྱང་སྡུག་བསྔལ་མྱོང་ཞེས་གསུངས། དུད་འགྲོའི་སྡུག་བསྔལ་གྱི་སྐབས་ཕྱིའི་རྒྱ་མཚོ་ཆེན་པོར་ནང་ནས་ཕྱིར་རིམ་པ་བཞིན་ལོངས་སྤྱོད་ཀྱི་མཚོ། ལོངས་སྤྱོད་༄༅། །ལྡན་གྱི་མཚོ། ལོངས་སྤྱོད་ཆེན་པོའི་མཚོ་ཞེས་པ་རིམ་པ་གསུམ་ཡོད། མཚོ་ནང་མ་ལ་སྟེང་ནས་འོག་ཏུ་དཔག་ཚད་ཉི་ཁྲི། བར་བར་བཞི་ཁྲི་ཐ་མར་བརྒྱད་ཁྲི་ཡོད། དེ་རྣམས་ལ་དུད་འགྲོའི་རིགས་མཐའ་ཡས་པ་གནས་པའི་གཟུགས་ཆེ་ཤོས་ཆུ་སྲིན་ཡིན། དེ་ཡང་མཚོ་ནང་མ་ལ་ཆུ་སྲིན་ཉའི་རིགས་ཅན་ཞེས་པ་དང་བར་མ་ལ་ཉ་མིད་ཅེས་པ་དང། ཕྱི་མར་ཉ་མིད་མིད་པ་ཞེས་པ་གསུམ་གནས། དེ་གསུམ་གྱི་ཕྱི་མ་ཕྱི་མས་ནང་མ་ནང་མ་རིམ་བཞིན་ཁྱུར་མིད་གཏོང་ནུས་པར་མདོ་ལས་གསུང་གསུངས། ཡང་སངས་རྒྱས་ཀྱི་སྐུའི་ཡོན་ཏན་བཤད་སྐབས། ཁྱོད་སྐུ་མཚན་གྱིས་སྤྲས་གྱུར་པ། ཞེས་སོགས་དེང་སངས་གཟུགས་ཅན་གྱི་རྒྱན་དང་དཔེ་རྒྱན་དུ་འཆད་ཀྱང་བསྡུ་བའི་རྒྱན་གཙོ་ཆེ་བ་འདྲ་གསུངས། དཀོན་མཆོག་གསུམ་ལ་སྐྱབས་སུ་འགྲོ་བའི་ཚེ། སངས་རྒྱས་ལ་ཐོབ་བྱ་དང། ཆོས་ལ་ཉམས་ལེན་དང། དགེ་འདུན་ལ་དེའི་གྲོགས་སུ་དམ་བཅའ། ཡང་སངས་རྒྱས་གཅིག་པུའང་དེ་གསུམ་དུ་དམ་བཅས་ནས་དཀོན་མཆོག་གསུམ་པོ་རེ་རེར་ཡང་གསུམ་ཀ་ཚང་བའི་ཚུལ་བསམ་གསུངས། ངན་སོང་གི་སྡུག་བསྔལ་བསམ་ན་འཁོར་བ་སྤྲས་གྱུར་པ། ཞེས་སོགས་དེང་སངས་གཟུགས་ཅན་གྱི་རྒྱན་དང་དགའ་བོའི་ཚ་བོ་གཉིས་ཀྱི་ལོ་རྒྱུས་གསུངས། ཡང་ལས་འབྲས་ཟབ་ཅིང་ཆུད་མི་ཟ་བའི་དཔེར་ཁྱིམ་བདག་དཔལ་སྐྱེས་ཀྱི་རྒྱུད་ཀྱི་ཐར་པ་ཆ་མཐུན་གྱི་དགེ་རྩ་དེ་སྔོན་སངས་རྒྱས་འོང་སྲུང་གི་མཆོད་རྟེན་ཞིག་གི་མཐའ་སྐོར་དུ་རྟའི་སྤངས་ཡོད་པ་ལ་ཁྱིམ་བདག་དེ་སྦྲང་བུར་སྐྱེས་དུས་སྤངས་ཀྱི་དྲི་ལ་རིམ་བཞིན་འཁོར་བས་མཆོད་རྟེན་ལ་ཆེད་དུ་མ་བསམས་ཀྱང་སྐོར་བ་གཅིག་འཁོར་བའི་མཐུ་ལས་བྱུང་བའི་སྔོན་བྱུང་དང། ཀུན་ཏུ་རྒྱུ་དག་གིས་མཽ་འགལ་གྱི་བུ་ཆེན་པོ་བཅོམ་པ་དང། འཕགས་པ་ནག་པོ་ཆེན་པོ་དང་ལེགས་འོངས་དང་འཕགས་པ་སྒུར་ཆུང་སོགས་ཀྱི་སྔོན་བྱུང་གསུངས། ཡང་འདོད་པའི་ཉེས་དམིགས་གསུང་སྐབས་སྤུ་གྲིའི་སོར་ཆགས་པའི་སྦྲང་རྩི་བ་ལྡག་པ་དང་འདྲ་ཞེས་པའི་ཞར་ལ། སྤུ་གྲི་ནི་རྒྱ་མཚོའི་ནང་ནས་འབྱུང་བའི་རྡོ་ཞིག་ལ་འཁོར་ལོ་བྱས་ནས། དེ་འོག་ནས་བསྐོར་བའི་སྟེང་དུ་ལྕགས་ཤིན་ཏུ་ངར་བ་བཟུང་ནས་བརྡར་ན་ཤིན་ཏུ་རྣོ་བར་གྱུར་པ་ཞིག་ཡིན། དེ་འདྲའི་གྲི་དེས་གང་གཅོད་པར་བྱ་བའི་སྤུ་ལ་མ་ཐུག་ཙམ་ནས་སྤུ་ཆད་འགྲོ་བ་ཞིག་ཡིན་གསུངས། ཡང་སྐབས་གཅིག་ལྷའི་སྦྱན་གྱིས་སྲིད་རྩེ་ནས་མནར་མེད་བར་གྱི་གཟུགས་ལིང་པོ་ད་མཐོང་བར་མ་ཟད་དེའི་ཕྱི་ནང་ཐམས་ཅད་མ་འདྲེས་པར་མཐོང། ལྷའི་མིག་གི་མངོན་ཤེས་ཀྱིས་གཟུགས་དེ་རྣམས་དང་དེའི་སྟེང་གི་མྱོང་བའི་འགྱུར་བ་རྣམས་ཀྱང་མཐོད་ཞེས་པའི་ན་ཡམ་གཅིག་ཀྱང་གསུངས། ༈ ཉིན་བདུན་པར། དཔལ་ལྡན་ཟླ་བ་གྲགས་པའི། གང་ཚེ་རང་དབང་འཇུག་ཅིང་མཐུན་གནས་པ། ཞེས་སོགས་དྲངས་ནས་སྤྱི་བཤད་དང་ཀུན་སློང་བཟོ་སྐབས། མདོ་ལས་ལེགས་པར་རབ་ཏུ་ཉོན་ལ་ཡིད་ལ་ཟུང་ཞིག། ཅེས་པའི་ལེགས་པར་ཞེས་པས་ཀུན་སློང་དང་དེར་མ་ཟད་ལུས་ངག་ཡིད་གསུམ་ཀའི་ཀུན་སྤྱོད་བསྟན། རབ་ཏུ་ཞེས་པས་ཉམས་སུ་ལེན་པའི་ཆེད་དུ་ཉན་དགོས་པ་བསྟན་ཡིད་ལ་བཟུང་ཞེས་པས་བསྒོམ་པས་༄༅། །ཉམས་སུ་མྱོང་བར་བྱེད་དགོས་ཚུལ་བསྟན་ཞེས་པའི་བཤད་པའི་རྣམ་གྲངས་ཤིག་ཀྱང་གསུངས། སངས་རྒྱས་ཀྱི་སྐུའི་ཡོན་ཏན་བསམ་སྐབས་བྱང་སེམས་ཤུགས་འཆད་གིས་གཙུག་ཏོར་གྱེན་དུ་འཕགས་པའི་མཐའ་རྟོགས་པའི་ཕྱིར་སྟེང་ཕྱོགས་སུ་འཇིག་རྟེན་གྱི་ཁམས་གྲངས་ལས་འདས་པར་ཕྱིན་ཀྱང་མཐའ་མི་རྙེད་པའི་ལོ་རྒྱུས་གསུངས། གསུང་གི་ཡོན་ཏན་བསམ་སྐབས་ཐོག་མར་དགེ་བ། བར་དུ་དགེ་བ་མཐའ་མར་དགེ་བ་ཞེས་པའི་ཐོག་མར་དགེ་བ་ཞེས་པའི་དོན། ནང་ཉམས་ལེན་ལ་སྦྱར་ན་ཉོན་མོངས་གནོན་པའི་དོན་ཡིན། དེ་བཞིན་དུ་བར་དུ་དགེ་བའི་དོན་ལམ་རྒྱུད་ལ་སྐྱེས་པའི་དོན་ཡིན་མཐའ་མར་དགེ་བའི་དོན་ལམ་མཐར་ཕྱིན་པའི་དོན་ཡིན་གསུངས། ཐུགས་ཀྱི་ཡོན་ཏན་བསམ་སྐབས་ལས་འཕྲོས་ནས། རྒྱལ་བའི་ཐུགས་མཉམ་པར་བཞག་བཞིན་དུ་རྗེས་ཐོབ་ཀྱི་མཛད་པ་མཛད་ཅེས་པའི་ཐ་སྙད་ཐོབ་པ་བཞིན་དུ་སྐུ་དང་གསུང་ལ་ཡང་མཉམ་པར་བཞག་པའི་ཐ་སྙད་ཀྱང་གསུངས། ཡང་སངས་རྒྱས་ཀྱི་ཐུགས་གཟུགས་སྐུའི་རྣམ་པར་ཤར་བ་ཡོད་པ་ལྟར་གསུང་ཡང་གཟུགས་སྐུའི་རྣམ་པར་ཤར་བ་ཡོད་པ་སོགས་གསུངས། ཡང་གསུང་གི་ཡོན་ཏན་གྱི་སྐབས་འཕགས་པ་མཽ་འགལ་གྱི་བུས་གསུང་གི་མཐའ་རྟོགས་པའི་ཕྱིར་ཤར་ཕྱོགས་ཀྱི་འཇིག་རྟེན་ཚད་མེད་པ་འདས་པའི་ཕ་རོལ་ཏུ་ཕེབས་པའི་ལོ་རྒྱུས་གསུངས། ཐུགས་ཀྱི་ཡོན་ཏན་གྱི་སྐབས་སྲིད་མོ་ཐེ་ཚོམ་སེལ་བ་ཞེས་པ་ཏིང་ངེ་འཛིན་གཅིག་ལ་མཉམ་པར་བཞག་པ་ན་དེའི་ཏིང་ངེ་འཛིན་སློང་བའི་ཕྱིར། མཽ་འགལ་གྱི་བུ་དང་འཇམ་དཔལ་སོགས་ཀྱིས་སྒྱུ་འཕྲུལ་བསྟན་ཀྱང་སློང་མ་ནུས་ཤིང། བྱང་སེམས་སྒྲིབ་པ་རྣམ་སེལ་གྱིས་སེ་གོལ་གཏོགས་པས་བསླང་བའི་ལོ་རྒྱུས་གསུངས། དེར་མ་ཟད་བྲམ་ཟེ་ཤིན་ཏུ་བརྐམ་ཆགས་ཅན་ཞིག་ལ་འཕགས་པ་དྲུག་སྡེས་ཐབས་སྣ་ཚོགས་ཀྱིས་གཙུག་ལག་ཁང་རྩིག་པ་སྣ་བརྒྱ་ཡོད་པ་ཞིག་བརྩིགས་པའི་སྦྱིན་བདག་བསྐུལ་ནས་བརྩིག་ཏུ་བཅུག་པ་ན། སངས་རྒྱས་ཀྱིས་ཁང་པ་དེ་མྱུར་དུ་འཇིག་པ་དང། ཞིག་པའི་ཚེ་སྦྱིན་བདག་བྲམ་ཟེ་ཁ་ནས་ཁྲག་ཚ་བ་སྐྱུག་ཅིང་འཆི་བའི་གནས་ཡོད་པ་གཟིགས་ནས་ཐབས་མཁས་ཀྱིས་གཙུག་ལག་ཁང་དེར་ཕེབས་ནས་རྒྱལ་ཆེན་བཞི་དང། བརྒྱ་བྱིན་དང་ཚངས་པ་རྣམས་ལ་མཚན་མོ་རིམ་པར་ཆོས་སྟོན་པར་མཛད་པའི་མཐར་བྲམ་ཟེ་དེ་ཡང་གཏོང་ཕོད་ལ་བཀོད་པའི་ལོ་རྒྱུས་གསུངས། ལས་འབྲས་ཤིན་ཏུ་ཟབ་པའི་དཔེར་འཕགས་སྐྱིས་པོ་དང། མ་ལ་གནོད་གཉིས་ཀྱིས་ཤཱཀྱ་རྣམས་ཁ་བརྟགས་པའི་ཚེ་སངས་རྒྱས་ཀྱང་དབུ་སྙུན་ཚུལ་བསྟན་པའི་ལོ་རྒྱུས་ཀྱང་གསུངས། སྙན་པ་བཟང་ལྡན་གྱི་སྔོན་གྱི་བྱུང་བ་དང་སྤྲེའུ་སྦྲང་རྩི་འབུལ་གྱི་སྔོན་བྱུང་དང། འཕགས་པ་ནག་པོ་ཆེན་པོར་བྱ་རོག་ཞར་མ་ལྟར་བལྟའོ་ཞེས་སྨྲ་མཁན་གྱི་བུ་མོ་སོགས་ཀྱི་སྔོན་བྱུང་གསུངས། ལས་དུ་མས་ལུས་གཅིག་འཕེན་པའི་དཔེར་ཤེད་བུ་སེར་སྐྱའི་སྐྱེ་བ་འཕེན་པའི་ལས་ལྟ་བུ་གསུངས། ཡང་མཁས་པའི་སྡིག་པ་ལྕི་ཡང་ཡང་བའི་དཔེར་རྒྱལ་པོ་གཏུམ་པོ་རབ་སྣང་གིས་བྲམ་ཟེ་བརྒྱད་ཁྲི་དང་ཕྲ་མེན་མ་ལྔ་བརྒྱ་དུས་གཅིག་ལ་བསད་ཀྱང། འཕགས་པ་ཀ་ཏྱཱ་ཡན་ལ་དད་ཅིང་འགྱོད་པའི་༄༅། །སྟོབས་ཀྱིས་མྱུར་དུ་བདེན་པ་མཐོང་བའི་གཏམ་རྒྱུད་གསུངས། ༈ ཉིན་བརྒྱད་པར། སྡུག་བདེན་གྱི་ཉེས་དམིགས་སྐབས། མངལ་གྱི་གནས་སྐབས་ལྔའི་དང་པོ་མེར་མེར་པོའི་སྐབས་སུ་ཡང་སེམས་ཀྱི་སྣང་བ་ལ་དྲི་ང་ཞིང་མི་བདེ་བའི་ཉམས་འཆར་རྒྱུ་ཞིག་ཡོད། དེ་བཞིན་དུ་ཕྱི་མ་ཕྱི་མ་རྣམས་ཀྱི་སྐབས་སུ་ཡང་རིམ་གྱིས་སྡུག་བསྔལ་ཇེ་ཆེར་འགྲོ་གསུངས། ཡང་ལྷ་མ་ཡིན་རྣམས་རང་གི་གནས་ནས་འཕགས་ཏེ། སུམ་ཅུ་རྩ་གསུམ་གྱི་གནས་ལ་སོང་ནས་ལྷ་རྣམས་ཀྱི་དཔལ་དང་ལོངས་སྤྱོད་ལ་རྟོག་ཅིང། དེ་ནས་ཕྲག་དོག་སྐྱེས་ནས་སླར་ལྷ་རྣམས་དང་འཐབ་པའི་ཚེ་སྡུག་བསྔལ་མྱོང་ལུགས་རྒྱ་ཆེར་གསུངས། དེ་ཡང་ལྷ་མིན་གྱི་དབང་པོ་སྒྲ་གཅན་སོགས་ཀྱིས་གླིང་གསུམ་སོགས་མི་ཡུལ་དུ་ཆོས་སྤྱོད་པ་མང་བའི་ཚེ། ལྷ་རྣམས་རྒྱལ་བར་ཤེས་ནས་ཆོས་བྱེད་པ་ལ་བར་གཅོད་རྩོམ་པའི་ཕྱིར་ཤར་གྱི་ལུས་འཕགས་ཀྱི་གླིང་དང་ནུབ་ཀྱི་བ་ལང་སྤྱོད་ཀྱི་གླིང་གི་ནམ་མཁར་ཤིན་ཏུ་འཇིགས་པའི་གཟུགས་བསྟན་ནས་སེར་བ་དང། དབྱུག་པ་ལྟ་བུ་ཞེས་པའི་ཆར་མ་རུང་བ་ཕབ་སྟེ་ལོ་ཐོག་བཅོམ་ནས་འཚེ་བར་བྱེད། བྱང་གི་སྒྲ་མི་སྙན་དུ་ནི་རླུང་མ་རུང་བ་བཏང་ནས་མ་སྨོས་པའི་ལོ་ཐོག་དང། དཔག་བསམ་གྱི་ཤིང་རྣམས་ཀྱི་འབྲས་བུམ་རུང་བར་བྱས་ནས་འཚེ་བར་བྱེད་གསུངས། ལྷ་རྣམས་གནས་གོང་མ་གོང་མ་ལོངས་སྤྱོད་དང་དཔལ་ཕུན་སུམ་ཚོགས་ཀྱང་ལྷ་མིན་ནི་དེ་ལས་ལྡོག་པར་གསུངས། ཉོན་མོངས་ཐམས་ཅད་ཀྱི་སྟེང་ན་མ་རིག་པ་རེ་ངེས་པར་ཡོད་གསུང་བ་ཞིག་ཀྱང་བྱུང། ངན་སོང་དང་མཐོ་རིས་ཀྱི་ཉེས་དམིགས་རྣམས་དྲན་པ་ཉེར་གཞག་གི་མདོ་ལ་བལྟས་ནས་སྒོམ་དགོས། དེ་ཡང་དྲན་པ་ཉེར་གཞག་གི་མདོ་སྔར་བོད་དུ་འགྱུར་བ་མཇུག་མ་ཚང་ཞིང། ཆག་ལོ་ཙཱ་བ་ཆོས་རྗེ་དཔལ་གྱིས་སླར་པཎྜི་ཏ་རིན་ཆེན་འབྱུང་གནས་ཞི་པ་དང་བཅས་འགྱུར་ཆད་བསབ་ནས་མཇུག་རྫོགས་པར་གྲགས་ཀྱང། ངན་སོང་གསུམ་ནས་འཐབ་བྲལ་གྱི་བར་གྱི་བཤད་པ་མ་གཏོགས། དགའ་ལྡན་ཡན་ཆད་ཀྱི་བཤད་པ་མ་བྱུང་བས་ད་དུང་མཇུག་མ་རྫོགས་པ་འདྲ་གསུངས། ཡང་ཚུལ་ཁྲིམས་ཀྱི་སྒྲ་བཤད་མཛད་པའི་སྐབས་ཤིག་ཚུལ་བཞིན་ཡིད་ལ་བྱེད་པའི་ཁྲིམས་ཡིན་པས་ན་ཞེས་པ་དང། ཚུལ་དང་མཐུན་པའི་ཁྲིམས་ཡིན་པས་ན་ཚུལ་ཁྲིམས་ཞེས་སྒྲ་བཤད་ཚུལ་གཉིས་ཀྱི་རྣམ་གཞག་ཞིག་ཀྱང་གསུངས། ༈ ཉིན་དགུ་པར། རྣམ་འགྲེལ་ལས། སྲིད་པའི་སྲེད་ལས་རྣམས་བརྒལ་བས་ཞེས་སོགས་དྲངས་ནས། སངས་རྒྱས་ཤཱཀྱ་ཐུབ་པ་སངས་རྒྱ་ཁའི་དུས་སྔར་བསྐལ་པ་མང་པོར་བསགས་པའི་འཁོར་བས་བསྡུས་པའི་ལས་ཀྱི་ཆ་མང་དུ་ཡོད་པ་རྣམས་སངས་རྒྱས་པ་ན་རྣམ་སྨིན་འབྱིན་པའི་ནུས་པ་ཕལ་ཆེར་བཅོམ་ཞིང། འགའ་ཞིག་མ་འོངས་པའི་དགེ་སློང་རྣམས་ཀྱི་ལོངས་སྤྱོད་ཀྱི་རྒྱུར་བསྔོ་བར་མཛད་པ་ཡིན། དེའི་རྒྱུ་མཚན་ཡང་སངས་མ་རྒྱས་ན་སླར་འཁོར་ལོས་སྒྱུར་བའི་རྒྱལ་པོ་དྲུག་ཁྲི་ལྔ་སྟོང་བདུན་བརྒྱར་སྐྱེ་བ་བཞེས་རྒྱུ་ཡིན་པ་ལ་སངས་རྒྱས་པའི་ཚེ་འཁོར་སྒྱུར་དེ་སྙེད་དུ་སྐྱེས་དུས་ཀྱི་ལོངས་སྤྱོད་རྣམས་མ་འོངས་པའི་དགེ་འདུན་རྣམས་ཀྱི་ཆེད་དུ་༄༅། །བསྔོས་བས། མི་རྣམས་སེན་མོའི་སྟེང་དུ་ཞིང་རྨོ་ཡང་ངའི་ཉན་ཐོས་རྣམས་བསོད་སྙོམས་ཀྱིས་འཕོངས་པར་མི་འགྱུར་རོ་ཞེས་འདུལ་བ་ལུང་ལས་བཤད་གསུངས། ཡང་མ་འོངས་པའི་ཉན་ཐོས་རྣམས་བསོད་སྙོམས་ཀྱི་མི་ཕོངས་པའི་རྒྱུ་གཅིག་ནི་སྔོན་སློབ་ལམ་གྱི་ཚེ་ཞིང་སྦྱངས་པའི་མཐུ་ཡིན་བྱས་ཀྱང་ཆོག་གསུངས། སྔ་མ་ལྟར་ན་གཞན་གྱིས་བསགས་པའི་ལས་ཀྱི་རྣམ་སྨིན་གཞན་གྱིས་མྱོང་བར་ཐལ་བའི་སྐྱོན་འོང་དམ་སྙམ་ན་སྐྱོན་མེད་དེ། དེ་ནི་བསམ་གྱིས་མི་ཁྱབ་པའི་གནས་ཡིན་པའི་ཕྱིར་གསུངས། ཡང་མེ་ལྔ་བསྟེན་བ་ནི་ཕྱོགས་བཞིར་མེ་ཆེན་པོ་རེ་བཏང་ནས་དབུས་སུ་རང་བསྡད་མེས་རང་གི་ལུས་ལ་མ་རེག་ཙམ་བྱས་སྟེང་ནས་ཉི་མས་གདུང་ཕྱོགས་བཞི་ནས་མེས་གདུང་བའི་དཀའ་ཐུབ་བསྟེན་ནས་རྡེའུ་ལྟ་བུའི་དམིགས་རྟེན་གཅིག་ལ་རྩེ་གཅིག་ཏུ་བསྒྲིམ་སྟེ་བསྒོམ་པ་ནི་འདོད་ཆགས་སྤོང་བའི་ཐབས་སུ་བསམ་པ་ཡིན་ཀྱང། དེས་འདོད་པའི་འདོད་ཆགས་རེ་ཞིག་མགོ་གནོན་ཙམ་ལས་མི་ནུས་གསུངས། ཡང་འདོད་པའི་ལས་གཡོ་བའི་དཔེར་ཕྱི་མ་ལྷར་སྐྱེ་འགྱུར་གྱི་ལས་བསགས་པ་ཞིག་ཡི་དྭགས་ཀྱི་དཔོན་མོར་སྐྱེས་པའི་གཏམ་རྒྱུད་ཀྱང་གསུངས། ཁམས་གོང་གི་ལས་མི་གཡོ་བའི་ལས་ཡིན་པའི་རྒྱུ་མཚན། དེ་རྣམས་ཏིང་ངེ་འཛིན་གྱི་སྟོབས་ཀྱིས་བསགས་པའི་ལས་ཡིན་པས། དེའི་རྣམ་སྨིན་གཞན་དུ་བསྒྱུར་དཀའ་བའི་གནད་ཀྱིས་ཡིན་གསུངས། ཡང་སྟོང་ཉིད་སྒྲ་སྤྱི་ཙམ་གྱི་སྒོ་ནས་རྟོགས་ཚུལ་དང་དོན་སྤྱིའི་སྒོ་ནས་རྟོགས་ཚུལ་དང་མངོན་སུམ་དུ་རྟོགས་ཚུལ་གསུམ་ཡོད་པའི་དཔེ་ནི། རྒྱ་གར་རྡོ་རྗེ་གདན་གྱི་གཙུག་ལག་ཁང་དབུས་སམ་དེ་ལ་བོད་ཀྱི་ཐོག་སོ་དགུ་རེ་ལ་ཐོག་སོ་རེར་བྱས་པའི་ཐོག་སོ་དགུ་བརྩེགས་ཡོད་པས་དེ་ལྟར་ཡོད་དོ་ཞེས་པའི་སྒྲའི་རྗེས་སུ་འབྲངས་ནས་བསམ་པ་ན་གོ་ཚུལ་ཞིག་ཡོང་བ་དེ་ནི་སྒྲ་སྤྱི་ཙམ་གྱི་འཆར་ཚུལ་ཡིན། དེ་རྗེས་རྡོ་རྗེ་གདན་གྱི་བཀོད་པ་རྒྱུས་ཡོད་པ་ཞིག་གིས་དེའི་བཀོད་པ་ཇི་ལྟ་བ་བཞིན་བཟོས་ནས། འདི་འདྲའོ་ཞེས་བསྟན་པ་ན་གོ་ཚུལ་ཞིག་ཡོང་བ་ནི་དོན་སྤྱིའི་འཆར་ཚུལ་ཡིན་དཔེ་དེ་དང་འདྲ་གསུངས། སྐྱེས་བུ་ཆེན་པོའི་ཆོས་སྐོར་སྐབས། གསེར་འགྱུར་རྩི་ཡི་རྣམ་པ་མཆོག་ལྟ་བུ་ཞེས་པའི་གསེར་འགྱུར་གྱི་རྩི་ནི་དངུལ་ཆུ་ལས་བཟོས་པ་ཞིག་ཡིན་ཅིང་མིང་གཞན་ཚོན་ཆེན་པོ་ཡང་ཟེར༑ དེ་ཉིད་རྡོའམ་ལྕགས་སོགས་སྲང་དྲུག་ཡོད་པ་ཞིག་ལ་བསྐུས་ནས་བཞག་ན་ཟླ་བ་དྲུག་ནས་གསེར་དུ་འགྱུར་ནུས་པ་ནས་ཞག་གཅིག་ལ་འགྱུར་ནུས་པའི་བར་ཡོད་གསུང་བ་འདྲ་ཞིག་གོ་བྱུང། ༈ ཉིན་བཅུ་པར། འཇིག་རྟེན་དོན་སྒྲུབ་ཚུལ་ལ་མཐུ་མེད་པའི། །ཐེག་པ་དམན་པ་ནམ་ཡང་འདོད་མི་བྱ། ཞེས་དྲངས་ནས། ཉན་རང་གིས་ཀྱང་སེམས་ཅན་གྱི་དོན་བྱེད་པའི་ཤེས་བྱེད་ལ་མཆོག་ཟུང་སོགས་ཀྱིས་གཞན་དོན་དཔག་མེད་མཛད་པའི་རྟོགས་བརྗོད་གསུངས། འོན་ཀྱང་སངས་རྒྱས་ཀྱིས་གཞན་དོན་བྱེད་པ་དང་མི་འདྲ་ཚུལ་མཐའ་ཡས་པ་ཡོད་ཀྱང། གཙོ་བོ་འཁོར་འདུ་ཐེབས་གཅིག་ལྟ་བུས་ཀྱང་ཁམས་དང་བསམ་པ་དང་བག་ལ་ཉལ་ཇི་ལྟ་བ་བཞིན་མི་ཤེས་པའི་རྒྱུ་མཚན་གྱིས་ཡིན་༄༅། །གསུངས། གཞན་ཡང་ཤཱ་རིའི་བུས་རྫུ་འཕྲུལ་གྱིས་དམྱལ་བའི་གནས་ཆུ་བསིལ་བས་གང་བར་མཛད་ནས་དེ་རྣམས་ཀྱི་སྡུག་བསྔལ་ཞི་བར་མཛད་མོད་ཀྱང། བྱང་ཆུབ་སེམས་དཔའ་སྤྱན་རས་གཟིགས་ཀྱིས་ཆུ་ཐིག་པ་གཅིག་གིས་ཀྱང་དམྱལ་བའི་སྡུག་བསྔལ་ཞི་བར་བྱེད་ནུས་གསུངས། ཡང་སངས་རྒྱས་ཀྱི་བསྟན་པ་ཉམས་པའི་ཚེ་རང་རྒྱལ་དང། དྲང་སྲོང་མངོན་ཤེས་ཅན་རྣམས་ཀྱིས་རྩིས་དང་སྨན་དཔྱད་སོགས་དང་ཞི་གནས་དང་ལྷག་མཐོང་སོགས་ཀྱི་ཆོས་སྟོན་པ་ཡོད། དེ་ཡང་སངས་རྒྱས་ཞལ་བཞུགས་པ་དང་མྱ་ངན་ལས་འདས་པ་དང་བསྟན་པ་ཉམས་པའི་ཚེ་ཆོས་སྟོན་མཁན་རྣམས་ནི་དཔེར་ན། ཉིན་མོའི་ཚེ་ཉི་མའི་འོད་ཀྱིས་སྣང་བར་བྱེད་དེ་ནུབ་ནས་ཟླ་བ་དང་དེ་ནུབ་ནས་རྒྱུ་སྐར་གྱི་འོད་ཀྱིས་སྣང་བ་དང་འདྲ་གསུངས། ཡང་མཁའ་ལྡིང་ལ་རྒྱ་མཚོ་ཆེན་པོ་ནས་ཀླུའི་རྒྱན་ཆ་ཟབ་དང། ཀླུ་དང་རྒྱན་སྦྲགས་ནས་ཟ་བའི་རིགས་གཉིས་ཡོད། ཕྱི་མ་དེ་ལ་འཆི་ལྟས་འཕོག་པ་ན་མིག་གིས་ཇི་ཙམ་བལྟས་ཀྱང་ཀླུ་མི་མཐོང། དེའི་ཚེ་ཁོང་ཁྲོ་སྐྱེས་ནས་རི་རབ་ཀྱི་རྩེ་ནས་རྒྱ་མཚོར་ལན་བདུན་གྱི་བར་དུ་ཞུགས་ཀྱང་ཟས་མ་རྙེད་པ་ན། ཁོངས་པའི་ནང་ནས་ལྟོགས་པའི་མེ་འབར་ནས་ལུས་སྲེག་པར་བྱེད་ཅིང། མཐར་སྙིང་འབའ་ཞིག་ལུས་པ་དེ་རྒྱ་མཚོར་ལྷུངས་ནས་རིན་པོ་ཆེ་ཐམས་ཅད་ཡོངས་སུ་བསྡུས་པ་ཞེས་བྱ་བའི་རིན་པོ་ཆེར་འགྱུར་གསུངས། ཡང་བྱང་སེམས་དུད་འགྲོར་སྐྱེས་ཀྱང་ཐབས་མཁས་ཀྱིས་གཞན་དོན་མི་འདོར་བའི་དཔེར་རི་བོང་གི་སྐྱེས་རབས་ཟུར་ཙམ་གསུངས། ཡང་སྐབས་ཤིག་ཐུན་མོང་དང་ཐུན་མོང་མིན་པའི་སྐབས་མ་ཕྱེ་བར་སྤྱིར་རྒྱལ་པོ་སྤྱི་བོ་སྐྱེས་ལ་སངས་རྒྱས་ཟིན་དང། བྱང་སེམས་ཀྱི་ཆ་གཉིས་ཡོད་གསུང་བ་འདྲ་ཞིག་ཀྱང་བྱུང། ཡང་རྒྱལ་པོ་དེའི་རྟོགས་བརྗོད་བོད་དུ་འགྱུར་བའི་མདོ་སྡེ་དག་གི་ནང་དུ་དྲུག་བདུན་ཙམ་བཞུགས་ཀྱང་དེ་དག་གི་ནང་དུ་བློན་པོ་ཆེན་པོ་བདེན་པའི་བློ་གྲོས་ཞེས་པ་ཞིག་ཡོད་པར་མ་གསུངས་ཤིང། དཔག་བསམ་འཁྲི་ཤིང་དུ་དེ་འདྲ་ཡོད་པར་གསུངས་པ་ནི། ཆོས་རྒྱལ་དགེ་དབང་གིས་རྒྱ་གར་དུ་བཞུགས་པའི་མདོ་ཞིག་ལ་ཡོད་པར་གཟིགས་ནས་གསུངས་པ་འདྲ་གསུངས། ༈ ཉིན་བཅུ་གཅིག་པར། ལོང་བས་ཕྱག་དར་ཕུང་པོ་ལས་ཞེས་དྲངས་ནས་སྤྱི་བཤད་མཛད། སངས་རྒྱས་ཀྱི་སྒྲ་བཤད་མཛད་པའི་སྐབས་སངས་ཞེས་པ་ལས་འཕྲོས་ནས་རྩེ་མོའི་སྦྱོར་བ་ཐོབ་པའི་བྱང་སེམས་རྣམས་ཀྱི་རྨི་ལམ་དུ་ཆོས་ཐམས་ཅད་རྨི་ལམ་ལྟ་བུར་བལྟ་བ་ཡོད་པར་མ་ཟད། དེ་ལྟར་བལྟ་བའི་བློ་དེ་ལ་མཉམ་གཞག་དང་རྗེས་ཐོབ་གཉིས་ཀྱང་ཡོད་གསུངས། ཡང་ས་དང་པོར་སྟོང་ཉིད་མངོན་སུམ་དུ་རྟོགས་ཏེ་དེ་ལ་ཆུ་ལ་ཆུ་བཞག་པ་ལྟར་སོང་སྟེ་གཉིས་སྣང་ནུབ་པ་ཡིན་ཀྱང། དེ་དང་ས་གཉིས་པ་ནས་བཅུ་བའི་བར་གྱི་སྟོང་ཉིད་མངོན་སུམ་དུ་རྟོགས་པའི་ཡེ་ཤེས་རྣམས་ལ་གཉིས་སྣང་ནུབ་ཚུལ་མི་འདྲ་བ་རེ་ཡོད་གསུངས། ༈ ཡང་སྤངས་པ་ཁྱད་པར་གསུམ་ལྡན་གྱི་སླར་མི་ལྡོག་པར་སྤངས་པའི་དོན་སྤངས་བ་དེ་ལས་མི་ཉམས་པའི་དོན་ཡིན། ཕྱི་རོལ་པ་འགའ་ཞིག་གིས་བསམ་གཏན་གྱི་དངོས་༄༅། །གཞིའི་ཏིང་ངེ་འཛིན་གྱི་མཐུས་འདོད་ཉོན་རེ་ཞིག་མགོ་གནོན་ཀྱང་སླར་མངོན་གྱུར་དུ་སྐྱེ་བ་ཡོད་པ་དེ་ལས་ཁྱད་པར་དུ་གྱུར་པའི་དོན་ཡིན་གསུངས། དད་པ་དང་མོས་པ་དང་རྗེས་སུ་ཡི་རང་སོགས་བྱས་ནས་བསོད་ནམས་ཚད་མེད་པ་ཐོབ་པའི་དཔེར་རྒྱལ་པོ་གསལ་རྒྱལ་གྱིས་སངས་རྒྱས་འཁོར་བཅས་ཟླ་བ་དྲུག་ཏུ་བསྙེན་བཀུར་བའི་ཚེ། སློང་མོ་བ་དེས་བ་ཞེས་པས་དད་པ་དང་རྗེས་སུ་ཡི་རང་བྱས་བས། སངས་རྒྱས་ཀྱིས་ཉིན་རེ་བཞིན་ཡོན་བསྔོ་བར་མཛད་པའི་ཚེ་རྒྱལ་པོ་དང། སློང་མོ་བདེའི་མ་གཉིས་ཀའི་མིང་ནས་སྨོས་ཏེ་བསྔོ་བ་མཛད་པའི་གཏམ་རྒྱུད་གསུངས། ཡང་བདག་གཞན་མཉམ་བརྗེའི་སྐབས་ལས་འཕྲོས་པའི་སྡུག་བསྔལ་དང་ལེན་གྱི་སྐབས། བྱང་སེམས་རྣམས་ཀྱི་མོས་པའི་མཐུས་གཞན་གྱི་སྡུག་བསྔལ་དངོས་སུ་ལེན་ནུས་པའི་དཔེར་བླ་མ་བྱམས་པའི་རྣལ་འབྱོར་པའི་རྣམ་ཐར་དང། རྗེ་སྨོན་ལམ་དཔལ་བ་གཙང་དུ་བཞུགས་པའི་སྐབས་ཤིག་ཡུལ་དེར་འབྲུམ་ནག་གི་ནད་ཆེན་པོ་ཞིག་བྱུང་བ་ན་རྗེ་རང་ཉིད་སྐུ་འབྲུམ་ཐར་མ་ཡིན་ཀྱང་སྡུག་བསྔལ་དང་ལེན་མཛད་པས་སྐུ་ལུས་ཐམས་ཅད་ཤུ་བ་ལྟ་བུས་ཁྱབ་པར་གྱུར་པ་ན། ཡུལ་གྱི་འབྲུམ་ནད་ཤིན་ཏུ་ཡང་བར་གྱུར་ནས་མྱུར་དུ་ཆད་པའི་ལོ་རྒྱུས་གསུངས། གཞན་ཡང་བྱང་སེམས་རྣམས་ཀྱིས་གཞན་གྱི་སྡུག་བསྔལ་དང་ལེན་བྱེད་པའི་ཚེ་གཞན་གྱི་སྡུག་བསྔལ་རང་གིས་བླངས་ཀྱང་རང་ལ་སྡུག་བསྔལ་མི་ཡོང་བ་དང་ཡོང་བ་གཉིས་ཡོད་གསུངས། བདག་གཞན་བརྗེ་བའི་ཚེ་བདག་གཅེས་འཛིན་གྱི་བློ་དེ་རང་གི་སྙིང་ཁར་འོད་ལྟ་བུའི་རྣམ་པར་བསྒོམ་སྟེ། གཞན་གྱི་སྡུག་བསྔལ་ཚུར་སྣ་བུག་གཡོན་ནས་ཞུགས་ཏེ་འོད་དེ་ཉིད་ནག་ཐུམ་གྱིས་བསད་པའི་མོས་པ་བྱེད་པ་ཡང་ཡོད། ཡང་བདག་གཅེས་འཛིན་གྱི་བློ་ལ་དགྲར་བལྟས་ནས་ཆགས་ཆེ་ཉམ་ང་ཐམས་ཅད་དེའི་ཐོག་ཏུ་བསྒྱུར་བའི་མོས་པ་བྱེད་པ་ཡོད། འདི་བླ་མ་དྷརྨཱ་རཀྵི་ཏ་དང་བོད་ཀྱི་བླ་མ་སྔ་མ་རྣམས་ཀྱི་ཕྱག་ལེན་དུ་བྱུང་ཡང་མདོ་རྒྱུད་གཞན་ན་མི་གསལ་གསུངས། ཡང་བདག་གཅེས་འཛིན་གྱི་བློན་མ་སྐྱེས་བྱུང་བའི་ཚེ། བློ་དེ་ཉིད་རང་བཞིན་མེད་པར་གཏན་ལ་ཕབ་སྟེ་ཞི་བར་བྱེད་པ་ནི་དབང་རྣོན་གྱི་ལུགས་ཡིན་གསུངས། ཚུལ་ཁྲིམས་ཉེས་སྤྱོད་དྲི་མ་འཁྲུད་པའི་ཆུ་ཞེས་སོགས་རྐང་པ་གསུམ་གྱིས་རིམ་པ་བཞིན། སྡོམ་པའི་ཚུལ་ཁྲིམས་དང་སེམས་ཅན་དོན་བྱེད་དང་དགེ་བ་ཆོས་སྡུད་ཀྱི་ཚུལ་ཁྲིམས་གསུམ་བསྟན་ཅེས་གསུངས། སེམས་ཅན་དོན་བྱེད་ཀྱི་ཤེས་རབ་ནི་སེམས་ཅན་རྣམས་ཀྱི་ཁམས་དང་བསམ་པ་དང་བག་ལ་ཉལ་རྣམས་ཤེས་པའི་ཤེས་རབ་ཅིག་ལ་བྱེད་གསུངས། ༈ ཉིན་བཅུ་གཉིས་པར། རྒྱ་ཆེ་རོལ་པར། བསོད་ནམས་རྣམ་སྨིན་བདེ་སྟེར་སྡུག་བསྔལ་ཐམས་ཅད་སེལ་྾྾྾། ཞེས་སོགས་དྲངས་ནས་སྤྱི་བཤད་སྐབས། སངས་རྒྱས་ཀྱིས་བྱང་ཆུབ་ཤིང་དྲུང་དུ་རྩྭའི་གདན་འདིང་བའི་དགོས་པ་ནི། དྲེགས་པ་དང་རྒྱ་གས་པ་དང་ང་རྒྱལ་བཅོམ་པའི་སྒོ་ནས་སངས་རྒྱ་དགོས་ཞེས་མཚོན་པའི་བརྔ་ཡིན། དེའི་ཚེ་སྨིན་མཚམས་ཀྱི་མཛོད་སྤུ་ལས་འོད་ཟེར་སྤྲོས་ཏེ་བདུད་ཀྱི་སྡེ་གནོན་པར་མཛད་པ་ནི། བདུད་ཀྱི་གནས་ན་ཡོད་པའི་གང་ཟག་༄༅། །མང་པོ་བྱང་ཆུབ་གསུམ་ཅི་རིགས་ལ་འགོད་བཞེད་པ་དང། ལྷའི་བུའི་བདུད་བཅོམ་པའི་དགོས་པ་གཉིས་ཡོད་གསུངས། སྨིན་མཚམས་ནས་འོད་ཟེར་སྤྲོས་པའི་དུས་བདུད་ཀྱི་གནས་ན་ཡོད་པའི་དོགས་པའི་རྒྱལ་མཚན་གཡོ་བར་མཛད་པ་ཡིན་ལ། དོགས་པའི་རྒྱལ་མཚན་ནི་ཆུ་སྲིན་གྱི་རྒྱལ་མཚན་ཡིན། དགའ་བའི་རྒྱལ་མཚན་ནི་དར་གྱི་ལྷབ་ལྷུབ་མང་པོ་ཡོད་པ་ཞིག་ཡིན། དེ་ཡང་དོགས་པའི་རྒྱལ་མཚན་གཡོས་ན་བཟང་བ་ཡིན་གསུངས། ཡང་ཤིང་དྲུང་དེར་བཞུགས་སྐབས་ཚངས་པ་དང་བརྒྱ་བྱིན་སོགས་ལྷ་ཀླུ་དང་བྱང་སེམས་གཞན་སངས་རྒྱས་དང་ཉེ་བར་མི་བཞུགས་པ་ནི། གྲོགས་གཞན་སུ་ཡང་མེད་པར་སངས་རྒྱས་གཅིག་པུས་བདུད་ཀྱི་སྡེམ་ལུས་པ་འཕམ་པར་བྱེད་ནུས་ཞེས་གོ་བའི་ཆེད་ཡིན། ཡང་ཉན་རང་དགྲ་བཅོམ་རྣམས་ཀྱིས་སེར་སྣ་མ་ལུས་པར་སྤངས་ཀྱང་སྦྱིན་པའི་ཕར་ཕྱིན་གྱི་མིང་མི་ཐོབ་པ་ནི་རང་གི་ལུས་ལོངས་སྤྱོད་དགེ་རྩ་དང་བཅས་པ་སེམས་ཅན་ཐམས་ཅད་ལ་སྙིང་ཐག་པ་ནས་མ་བསྔོས་པའི་སྟོབས་ཀྱིས་ཡིན་གསུངས། སྦྱིན་པའི་འཁོར་གསུམ་ཡོངས་སུ་དག་ལུགས་ལ་སྦྱིན་བྱའི་རྫས་སོགས་ཁ་ན་མ་ཐོ་བ་མེད་པ་ལ་བཤད་པ་དང། འཁོར་གསུམ་རང་བཞིན་མེད་པར་རྟོགས་པ་གཉིས་ཡོད་པའི་ཕྱི་མ་གཙོ་ཆེ་གསུངས། ཡང་བྱང་སེམས་རྣམས་ཀྱིས་རང་གི་ལུས་ལོངས་སྤྱོད་དགེ་རྩ་ཐམས་ཅད་བླ་མ་དང་དཀོན་མཆོག་གསུམ་གྱི་ཆེད་དུ་བསྔོ་མོད་ཀྱང། སླར་དེ་ལ་ལོངས་སྤྱོད་པའི་ཚེ་དཀོན་མཆོག་ལ་སེར་སྣ་མེད་པ་དང། རང་ཉིད་ཀྱིས་སྤྱོད་དགོས་པའི་དུས་དྲན་པར་བྱས་ནས་ལོངས་སྤྱོད་གསུངས། ཟང་ཟིང་གི་སྦྱིན་པ་ལ་ནང་དང་ཕྱིའི་སྦྱིན་པ་གཉིས་ཡོད་པའི་དང་པོའི་དཔེར་བརྗོད་ལ་རྒྱལ་པོ་བྱམས་པའི་སྟོབས་ཀྱི་རྟོགས་བརྗོད་གསུངས། ཡང་ལུས་སོགས་དངོས་སུ་སྟེར་ནུས་པའི་དུས་ལ་སླེབ་ཀྱང། སྨྱོ་བ་དང་བདུད་ཀྱི་རིགས་ཀྱི་ལྷ་ངག་ལ་མི་སྦྱིན་པར་གསུངས་པའི་དགོངས་པ་ཅུང་བརྟག་དཀའ་མོད་ཀྱང་རྣམ་པ་གཅིག་ཏུ་ན་དེ་ལ་ཕན་ཆེར་མི་ཐོགས་པ་ལ་དགོངས་ཟེར་དགོས་སམ་སྙམ། འོན་ཀྱང་བདག་ཅག་གི་སྟོན་པ་རྒྱལ་པོ་ཟླ་འོད་དུ་གྱུར་སྐབས་ལྷས་བྱིན་ངང་རྒྱུད་གཅིག་པའི་བྲམ་ཟེ་ཚུལ་ངན་པ་ཞིག་ལ་དབུ་སྦྱིན་པར་མཛད་པས། དེ་ལ་ཕན་མ་ཐོགས་ཀྱང་རྒྱལ་པོ་ཉིད་ཀྱི་སྦྱིན་པའི་ཐུགས་དགོངས་རྫོགས་གསུངས། མི་འཇིགས་པའི་སྦྱིན་པའི་ནང་ཚན་ཆུའི་འབྱུང་བའི་འཇིགས་པ་ལས་སྐྱོབ་པའི་དཔེར། རྗེ་སྨོན་ལམ་དཔལ་བ་དབུས་སུ་བཞུགས་སྐབས་ཤིག་སྐྱེ་ཆུ་ཤིན་ཏུ་ཆེ་བ་ཞིག་བྱུང་དུས་སྨོན་ལམ་དཔལ་འཕགས་པ་ཡིན་ན་ཆུ་ལྡོག་ཅིག་ཅེས་རྡོ་ལ་བྲིས་ནས་རགས་ཀྱི་སྟེང་དུ་བཞག་པ་དེ་མ་ཐག་ཆུའི་གནོད་པ་ལས་གྲོལ་བའི་རྣམ་ཐར་གསུངས། ༈ ཉིན་བཅུ་གསུམ་པར། རྒྱ་ཆེ་རོལ་པར། སྦྱིན་དང་ཚུལ་ཁྲིམས་བཟོད་དང་བརྩོན་འགྲུས་དང་ཞེས་སོགས་དྲངས་ནས་སྤྱི་བཤད་མཛད། སྡུག་བསྔལ་དང་ལེན་གྱི་བཟོད་པའི་སྐབས་འཕགས་པ་རྣམས་ལ་ལུས་ཀྱི་སྡུག་བསྔལ་འབྱུང་ཡང་སེམས་མི་བདེ་བ་མེད་པའི་དཔེར། འཕགས་པ་ཉེ་སྡེ་དང་འཁོན་ཡོད་པའི་མི་ཞིག་སྤྲུལ་གདུག་མྱུར་དུ་དུག་གོ་༄༅། །ཁྱབ་པར་བྱེད་བ་ཞིག་ཏུ་སྐྱེས་ནས་འཕགས་པ་ལ་གཙེས་བ་ན་འཕགས་པའི་ལུས་བྱེ་མའི་ཆང་བུ་ལྟར་གྱུར་ཏེ་ལུས་ལ་སྡུག་བསྔལ་བྱུང་ཡང་སྔོན་གྱི་ལས་བདག་གིར་མཛད་པས་སེམས་ལ་སྡུག་བསྔལ་མ་བྱུང་བའི་སྔོན་བྱུང་གསུངས། བཟོད་པ་དང་བྱམས་པ་སྒོམ་ན་ཕྱི་ནང་གི་གནོད་པས་མི་བརྫི་བར་མ་ཟད། བདེན་བའི་བྱིན་རླབས་འགྲུབ་པའི་དཔེར་དྲང་སྲོང་བཟོད་པ་སྨྲ་བ་དང། རྒྱལ་པོ་བྱམས་པའི་སྟོབས་གཉིས་ཀྱི་རྟོགས་བརྗོད་ཟུར་ཙམ་རེ་གསུངས། འདི་སྐབས་དྲང་སྲོང་བཟོད་པ་སྨྲ་བ་དིང་སང་གི་སྐྱེས་རབས་ཀྱི་བྲིས་ཐང་སོགས་རི་མོ་རྣམས་སུ་རབ་བྱུང་གི་ཆ་ལུགས་ཅན་བྲིས་པ་ཤ་སྟག་འདུག་ཀྱང། སངས་རྒྱས་ཀྱི་བསྟན་པ་མཐུན་སྣང་དུ་འཇིག་རྟེན་ན་མེད་པའི་སྐབས་གཅིག་ཏུ་ཕེབས་པར་འདུག་བས་ཀུན་ཏུ་རྒྱུའི་ཆ་བྱང་ཅན་ནམ་དྲང་སྲོང་གི་ཆ་བྱང་ཅན་ཡིན་པ་འདྲ་གསུངས། དེར་མ་ཟད་སྔོན་འབྲི་ཁུངས་པས་ས་སྐྱར་དམག་དྲངས་ནས་ས་པཎ་གྱི་ཆོས་གྲྭར་སོང་བ་ན་རྗེ་ས་པཎ་གྱིས་འཁོར་རྣམས་ལ་བཟོད་པ་སྒོམ་དུ་བཅུག་པ་དང། རྒྱལ་སྲས་ཐོགས་མེད་ཀྱིས་ཆོས་གྲྭར་ཡང་དམག་བྱུང་ཚེ་བཟོད་པ་སྒོམ་དུ་བཅུག་པའི་མཐུས་བཞེད་དོན་འགྲུབ་པའི་ལོ་རྒྱུས་གསུངས། ཡང་རྒྱལ་པོ་སངས་རྒྱས་ཕྱོགས་དང་ཟབ་མོའི་ཕྱོགས་ཅེས་བྱ་བ་གཉིས་བྱུང་བའི་ཕྱི་མས་འཕགས་པ་ཐོགས་མེད་ཀྱི་སྦྱིན་བདག་མཛད་ནས་ཡུལ་ལྔ་ལེན་དུ་གཙུག་ལག་ཁང་བརྩིགས་ནས་འཕགས་པ་ཆོས་སྟོན་པའི་ཚེ། དམག་བྱུང་དུས་ཀྱང་བཟོད་པ་བསྒོམ་པས་དམག་ལྡོག་པའི་ལོ་རྒྱུས་དང། དེ་ལས་འཕྲོས་ནས་ཏཱ་ར་ནཱ་ཐའི་རྒྱ་གར་ཆོས་འབྱུང་དུ་རྒྱལ་པོ་དེ་གཉིས་ལས་སངས་རྒྱས་ཕྱོགས་ཕྱི་མ་ཡིན་པར་བཤད་ཀྱང། ཁོ་བོ་རང་གི་བསམ་བ་ལ་འཇམ་དཔལ་རྩ་རྒྱུད་ཀྱི་ལུང་བསྟན་དང་བསྒྲིགས་ན། སངས་རྒྱས་ཕྱོགས་ཁུ་བོ་ལྟ་བུ་ཞིག་ཡིན་པས་སྔ་ཞིང། ཟབ་མོའི་ཕྱོགས་ཚ་བོ་ལྟ་བུ་ཡིན་བས་ཕྱི་བར་བྱེད་དགོས་སམ་སྙམ་གསུངས། ཡང་ཞི་གནས་འགྲུབ་ཁའི་ཚེ་རང་གི་མདུན་ན་བཞུགས་པའི་སངས་རྒྱས་ཀྱི་སྐུ་གཟུགས་ལྟ་བུ་ཞིག་དམིགས་རྟེན་དུ་བྱས་ཡོད་ན། ཕྱི་རོལ་གྱི་གཟུགས་དང་རང་གི་ལུས་འོད་ལྟ་བུའི་རྣམ་པར་ཞུ་ནས། མདུན་གྱི་དམིགས་རྟེན་དེ་ལ་ཐིམ་པའི་ཉམས་ཤར་ནས་མདུན་གྱི་དམིགས་རྟེན་མ་གཏོགས་གཞན་གང་གི་ཡང་སྣང་བ་མི་འཆར། དེ་བཞིན་དུ་རང་ཉིད་ལྟར་བསྒོམ་ན་ཡང་དེ་དང་འདྲ༑ ཡང་རང་གི་སེམས་ལ་དམིགས་ནས་ཞི་གནས་སྒོམ་ཀྱང་གཟུགས་སྣང་ཐམས་ཅད་རང་གི་སྙིང་ཁར་ཐིམ་བའི་ཉམས་འཆར། དེའི་གནད་ཀྱིས་ཞི་གནས་གསར་དུ་འགྲུབ་བའི་རྗེས་ཐོབ་ཀྱི་ཚེ། རང་གི་ལུས་གསར་དུ་གྲུབ་པ་ལྟ་བུའི་ཉམས་འཆར་བ་ཡིན་གསུངས། ཡང་ཞི་གནས་འགྲུབ་ཁའི་ཚེ་ཀླད་པ་ལྕི་བེམ་པ་སྙམས་བྱེད་པ་འོང་བ་ཡང་ཤ་ཝགས་མཚམས་སུ་ལུས་ཀྱི་གནས་ངན་ལེན་བསྐྱེད་པའི་རླུང་ཡོད་པ་རྣམས་རིམ་གྱིས་འདུས་ནས་སྤྱི་བོར་ཡལ་འགྲོ་བའི་རྟགས་ཡིན། ཀླད་པ་ལྕི་སྙམ་པ་ཡོང་ཡང་མི་བདེ་བ་མིན་ཚུལ་ནི། དཔེར་ན་སྐྲ་གསར་དུ་བཞར་བའི་ཚེ་མགོ་བོ་འཁྱགས་པ་ན་ལག་པ་དྲོན་མོ་མགོ་བོར་དལ་གྱིས་བཀབ་པ་ལྟ་བུའི་ཉམས་ཤིག་ཡིན་གསུངས། ཡང་རང་གི་སེམས། ༄༅། །ལ་དམིགས་ནས་ཞི་གནས་བསྒྲུབ་པར་གསུངས་པ་ཡང་མཐར་ཐུག་ན་དྲན་པ་རང་ལ་དམིགས་རྟེན་བྱས་པ་ཡིན། འདིའི་གནད་རྗེ་བློ་བཟང་ཆོས་ཀྱི་རྒྱལ་མཚན་གྱིས་ཕྱག་ཆེན་རྩ་འགྲེལ་ལས་འབྱུང་གསུངས། སེམས་དམིགས་པ་ལ་རྩེ་གཅིག་ཏུ་བཞག་པའི་སྐབས་གྲིམ་ཆ་མེད་པར་སྦྲོ་སང་ངེ་བའི་ཉམས་ཤར་བྱུང་ན་རྒོད་པའི་མགོ་ཚུགས་པ་ཡིན། ཡང་དེའི་ཚེ་གྲིམ་ཆ་མེད་པར་ལྷོད་དེ་བའི་ཉམས་བྱུང་ན་བྱིང་བའི་མགོ་ཚུགས་པ་ཡིན་པས་དེ་གཉིས་བྱུང་མ་ཐག་ནས་ཤེས་བཞིན་གྱིས་ངོས་ཟིན་པར་བྱེད་དགོས་གསུངས། ཡང་དམིགས་པའི་གཞི་གཅིག་རང་ལ་གནས་པས་ཞི་གནས་ཞེས་སྒྲ་བཤད་མཛད་པ་ཞིག་ཀྱང་བྱུང། ཡང་གོང་གི་སྤྱི་བཤད་མཛད་པའི་སྐབས་དགེ་སློང་འདྲེ་སྐྱེས་ཤི་བའི་མཆོད་རྟེན་ལ། བཅུ་གཉིས་སྡེས་སངས་རྒྱས་ཀྱི་དབུ་སྐྲ་དང་སེན་མོའི་མཆོད་རྟེན་དུ་མིང་བཏགས་ནས། གང་ཟག་མང་པོ་ཕྱག་བསྐོར་མཆོད་པ་སོགས་བྱེད་བཅུག་པའི་སྐབས་ཤིག །རབ་འབྱོར་འཁོར་ལྔ་བརྒྱ་དང་བཅས་པ་ཕྱོགས་དེར་ཕེབས་སྐབས་བཅུ་གཉིས་སྡེས་ཕྱག་མཆོད་བྱེད་པར་བསྐུལ་བས་དགྲ་བཅོམ་བ་རྣམས་ཀྱི་མངོན་ཤེས་ཀྱང་མ་བརྟགས་པར་མི་འཇུག་པས་དབུ་སྐྲ་སོགས་ཀྱི་མཆོད་རྟེན་དུ་འཁྲུལ་ནས་ཕྱག་སོགས་མཛད། སླར་འཕགས་པ་ཉེ་བ་འཁོར་གྱིས་རྒྱུ་མཚན་བཤད་བའི་རྐྱེན་གྱིས་ཚེ་དང་ལྡན་པ་ཆུ་འཐུང་ཡང་ཟེར་བ་དོན་འཕགས་པ་རབ་འབྱོར་འཁོར་ལྔ་བརྒྱ་དང་བཅས་བ་ཁྲོས་ནས་མཆོད་རྟེན་བཤིག །དོན་དེ་སངས་རྒྱས་ལ་ཞུས་པས་འཕགས་པ་མ་གཏོགས་སོ་སྐྱེའི་རུས་པ་སོགས་བཅུག་པའི་མཆོད་རྟེན་མི་བྱ་བར་བཅས་པའི་སྔོན་བྱུང་ལས་འཕྲོས་ནས། ཉན་ཐོས་དགྲ་བཅོམ་པ་རྣམས་ཀྱི་ཁོང་ཁྲོ་གཏན་ནས་སྤངས་ཀྱང་ཁྲོ་རྒྱུ་ཞིག་ཡོད་པ་འདྲ་གསུངས། ༈ ཉིན་བཅུ་བཞི་པར། སྤྱི་བཤད་སྐབས་དཔའ་བར་འགྲོ་བའི་ཏིང་ངེ་འཛིན་ལ་དབྱེ་ན། དོན་དམ་ལ་དཔའ་བར་འགྲོ་བའི་ཏིང་ངེ་འཛིན་དང། ཀུན་རྫོབ་ལ་དཔའ་བར་འགྲོ་བའི་ཏིང་ངེ་འཛིན་གཉིས་ཡོད། ཐེག་ཆེན་གྱི་མཐོང་ལམ་བར་ཆད་མེད་ལམ་ལྟ་བུ་རྣམས་དོན་དམ་ལ་དཔའ་བར་འགྲོ་བའི་ཏིང་ངེ་འཛིན་དང། རྣམ་གྲོལ་ལམ་རྡོ་རྗེ་ལྟ་བུའི་ཏིང་ངེ་འཛིན་ཡིན། སྒྱུ་མ་ལྟ་བུའི་ཏིང་ངེ་འཛིན་ནི་བྱང་སེམས་རྣམས་ཀྱིས་ཆོས་ཐམས་ཅད་རང་བཞིན་མེད་པར་སྒྱུ་མ་ལྟ་བུར་བསྒོམ་པའི་མཐུས་རྡུལ་གཅིག་གི་ནང་དུ་རྡུལ་ཐམས་ཅད་ཀྱི་གྲངས་དང་མཉམ་པའི་འཇིག་རྟེན་གྱི་ཁམས་ཤོང་བར་བྱིན་གྱིས་རློབ་ནུས་པའི་ཏིང་ངེ་འཛིན་ལྟ་བུ་ཡིན་གསུང། སྒྱུ་མ་ལྟ་བུའི་ཏིང་ངེ་འཛིན་གྱི་རྫུ་འཕྲུལ་གྱི་བཀོད་པའི་དཔེར་བརྗོད་ལ་སློབ་དཔོན་ཙནྡྲ྄་གོ་མིས་རང་གི་སློབ་མ་པཎྜི་ཏ་ཁ་ཅིག་ལ་མེ་ཏོག་གི་འདབ་མ་ཞིག་ལག་ཏུ་གཏད་ནས། དེའི་ཁ་འབྱེད་བཅུག་པ་ན་དེའི་ནང་དུ་ཐོག་མར་ནགས་ཚལ་གྱིས་གང་བའི་ལུང་པ་ཆེན་པོ་ཞིག་དང། དེ་ནས་ཀྱང་གཙུག་ལག་ཁང་མང་པོས་གང་བའི་ཡུལ་ཁམས་རྒྱ་ཆེན་པོ་བསྟན་པ་ལྟ་བུ་དང། གཞན་ཡང་བཀའ་གདམས་ཐིག་ལེ་བཅུ་དྲུག་གི་བཀོད་པ་ལྟ་བུ་ཡིན་གསུངས། ཡང་མིག་འཁྲུལ་དང་རྫུ་འཕྲུལ་མི་འདྲ་ཀུན་མཁྱེན་དོལ་པོ་བ་དགེ་འདུན་གྱི་ནང་དུ་ཆོས་སྟོན་པ་དང། སྐུ་༄༅། །འབུམ་ཆེན་མོའི་ལས་བཀོད་བྱེད་མཁན་སོགས་ཉིན་གཅིག་ལ་ལུས་ཀྱི་བཀོད་པ་གསུམ་དུས་གཅིག་ཏུ་བསྟན་པའི་རྫུ་འཕྲུལ་མང་བའི་གྲགས་པ་ཡོད་ཀྱང་རྫུ་འཕྲུལ་རང་ཡིན་པའི་ངེས་པ་མེད་པ་འདྲ། གཡུང་སྟོན་རྡོ་རྗེ་དཔལ་གྱིས་ཀུན་མཁྱེན་པར་ཕུལ་བའི་ཞུ་ཡིག་གཅིག་གི་ནང་དུ་མིག་འཁྲུལ་གྱི་རྫས་སྔགས་དགོས་ན་གཡག་སྡེ་པཎ་ཆེན་ལ་བསླབ་ཡོད་ཚུལ་གསུང་འདུག་གསུངས། ཞི་གནས་ཀྱི་ཚོགས་ཀྱི་སྐབས། བྱང་སེམས་ཏིང་ངེ་འཛིན་གྱི་རྩལ་རྫོགས་པ་རྣམས་གྲོང་མཐར་གནས་ཀྱང་ཏིང་ངེ་འཛིན་དང་གཞན་དོན་ལས་གཡོ་བར་མི་ནུས་པས་གང་དུ་གནས་ཀྱང་རྒྱལ་སྲས་ཀྱི་དབེན་པར་གནས་པར་གསུངས་ཀྱང། ལས་དང་པོ་པ་རྣམས་ལ་འདོད་པ་ཆེ་ཞིང་ཆོག་མི་ཤེས་པ་སོགས་ཏིང་ངེ་འཛིན་འགྲུབ་པའི་གེགས་ཡིན་པའི་དཔེར་འཕགས་པ་གནས་བརྟན་གསེར་བེའུ་ལ་གསེར་གྱི་གླང་པོ་ཆེའི་བེའུ་གཅིན་དང་ལྕི་བ་ཐམས་ཅད་གསེར་དུ་འགྱུར་བ་ཞིག་ཡོད་པ་དང། གནས་བརྟན་བྷ་ར་དྷྭ་ཛ་གསེར་ཅན་གྱི་ཕྱག་ལས་ནམ་དགོས་སུ་གསེར་གྱི་དོང་རྩེ་འབྱུང་བ་གཉིས་ཀྱི་རྐྱེན་གྱིས་སོ་མི་གསོད་པའི་གཏམ་རྒྱུད་གསུངས། ཆོས་ཐེག་པ་ཆེན་པོའི་ཆོས་ཡིན་པས་མི་ཆོག་གང་ཟག་ཐེག་པ་ཆེན་པོ་པ་ཡིན་པ་ཞིག་དགོས་པའི་ཤེས་བྱེད་ལ། རྒྱ་གར་དུ་དགྱེས་པ་རྡོ་རྗེའི་རྣལ་འབྱོར་པ་ཞིག་གིས་རྒྱུན་ཞུགས་ཀྱི་འབྲས་བུ་ཐོབ་པའི་གཏམ་རྒྱུད་ཀྱང་གསུངས། སེམས་གནས་དང་པོའི་སྐབས་ཞི་གནས་ཀྱི་གདམས་དག་ལ་བརྟེན་ནས་སེམས་དམིགས་པ་ལ་གསར་དུ་འཇོག་ཅིང་བཞག་པ་དེའི་རྒྱུན་མཐུད་མ་ནུས་པའི་དབང་གིས་ཡང་དང་ཡང་དམིགས་པ་ལ་འཇོག་པ་ཞིག་ཡིན། དེ་ལ་བསླན་ཏེ་འཇོག་པ་ཞེས་པའི་ཐ་སྙད་མི་འོང་སྟེ། རྩེ་གཅིག་པའི་རྒྱུན་མཐུད་མ་ཐུབ་པས་གང་ལ་བསླན་པའི་གཞི་མེད་པའི་ཕྱིར། བསླན་ཏེ་འཇོག་པའི་སྐབས་སེམས་གནས་པའི་རྒྱུ་མཚན་ཅུང་ཟད་མཐུང་ནུས་པས། ད་དུང་རྒྱུན་རིང་དུ་གཏོང་བའི་ཆེད་དུ་བསྐྱར་ནས་འཇོག་པས་ན་བསླན་ཏེ་འཇོག་ཅེས་པའི་ཐ་སྙད་བཏགས་པ་ཡིན་གསུངས། སེམས་གནས་བསྒྲུབ་པའི་སྐབས་བྱིང་བ་བཙན་ཐབས་སུ་གཅོད་པའི་མན་ངག་ཅེས་པ་ནི་ཧ་ཅང་བྱིང་གྲགས་ན། ལྟེ་བའི་ཐད་དུ་རང་སེམས་ཀྱི་ངོ་བོ་འོད་དཀར་པོ་སྲན་མ་ཙམ་གཅིག་བསམས། དེ་ཤུགས་ཆེན་པོས་སྙིང་ཁ་དང་མགྲིན་པ་དང་སྤྱི་གཙུག་གི་བར་ལི་གྱན་ལ་སྤར་ཏེ། སྤྱི་བོ་ནས་འཐོན་ནས་ནམ་མཁའ་དང་རང་སེམས་མཐའ་མེད་དུ་འདྲེས་པར་བསམ་པ་དང། ཡང་རྒོད་པ་བཙན་ཐབས་སུ་གཅོད་ལུགས་ནི། སྤྱི་བོར་རང་སེམས་ཀྱི་ངོ་བོ་འོད་དཀར་པོ་ཤིན་ཏུ་ལྕི་བའི་རྣམ་པ་ཅན་ཅིག་བསམ། དེ་ཤུགས་ཆེན་པོས་མགྲིན་པ་ལྟེ་བ་འོག་སྒོའི་བར་དུ་ཕབ་ཅིང། འོག་སྒོ་ནས་ཐོན་ནས་འོག་དབང་ཆེན་གསེར་གྱིས་གཞིའི་བར་དུ་སོང་བར་བསམ་གསུངས། ཡང་ཞི་གནས་འགྲུབ་ཁའི་ཚེ་ལུས་ཀྱི་ཚོར་བ་བདེ་བའི་རྐྱེན་གྱིས་སེམས་དམིགས་པ་ལ་གནས་མི་ནུས་པ་སྙམ་བྱད་པ་ནི་གནས་ངན་ལེན་བསྐྱེད་པའི་རླུང་རྣམས་རིམ་གྱིས་དག་པའི་མགོ་བརྩམས་པ་ཡིན། གནས་ངན་ལེན་གྱི་རླུང་རྣམས་མཐར་སྤྱི་བོར་འདུས་ནས་ཡལ་འགྲོ་བ་ནི། དང་པོར་མའི་མངལ་དུ་ལུས་ཆགས་༄༅། །པའི་སྐབས། འཕྲང་འགྱུར་གྱི་ཚེ་སྔོན་ལ་ལྟེ་བ་དང་སྙིང་ཁ་དང། སྤྱི་བོ་རྣམས་གང་དུ་ཡོང་བའི་ཐད་དེར་སྲ་བར་འགྱུར་བའི་རྟེན་འབྲེལ་གྱི་གནད་ཀྱིས་ཡིན་གསུངས་པ་འདྲ་ཞིག་གོ་བྱུང། ༈ ཉིན་བཅོ་ལྔ་པར། ཆོས་ཀྱི་རྒྱལ་པོ་ཙོང་ཁ་པ་ཆེན་པོས། འཇིག་རྟེན་རྒུད་པ་ཇི་སྙེད་པ་྾྾་རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བ་གསུངས་ཞེས་དྲངས་ནས་སྤྱི་བཤད་མཛད་སྐབས། མགོན་པོ་ཀླུ་སྒྲུབ་དང་འཕགས་པ་ཐོགས་མེད་གཉིས་ལ་ཤིང་རྟ་ཆེན་པོ་ཞེས་ཟེར་བའི་རྒྱུ་མཚན་ནི། དཔེར་ན་ཤིང་རྟའི་སྟེང་དུ་ནོར་དང་རིན་པོ་ཆེ་སོགས་བཞག་ནས་ཡུལ་གཞན་དུ་དྲངས་ན་མང་དུ་འདྲེན་ནུས་པ་ལྟར། དེ་གཉིས་ཀྱིས་ཀྱང་རྒྱལ་བའི་གསུང་རབ་ཀྱི་དགོངས་པ་ཕྱོགས་རེ་བ་ཙམ་མིན་པར་ཡོངས་སུ་རྫོགས་པར་འགྲེལ་བའི་ཁུར་བཟོད་པས་ན་ཤིང་རྟ་ཆེན་པོ་ཞེས་བྱ། རྣམ་པ་གཅིག་ཏུ་ན་གླང་པོ་ཆེ་སོགས་སྟོབས་དང་ལྡན་པའི་བཞོན་པས་ཤིང་རྟ་དྲངས་ནས། སྔར་འགྲོ་ས་མེད་པའི་གནས་སུ་ཡང་བགྲོད་ནས་ལམ་གསར་དུ་བཏོད་ནུས་པ་ལྟར། དེ་གཉིས་ཀྱིས་ཀྱང་སྔར་སུས་ཀྱང་མ་ནུས་པའི་རྒྱལ་བའི་གསུང་རབ་ཀྱི་དོན་འགྲེལ་པའི་སྲོལ་བཏོད་པས་ན་ཤིང་རྟ་ཆེན་པོ་ཞེས་བྱ་གསུངས། འདི་ལས་འཕྲོས་ནས་མགོན་པོ་ཀླུ་སྒྲུབ་ཀྱིས་ཀླུ་ཡུལ་དུ་ཕེབས་དུས་ཀླུའི་གནས་ཤིག་ན། དེའི་དུས་ཀྱི་མི་རྣམས་ལས་གཟུགས་ཉིས་འགྱུར་སུམ་འགྱུར་གྱིས་ཆེ་བའི་དགྲ་བཅོམ་མང་པོ་བཞུགས་འདུག་པ་ལ། དེ་རྣམས་ཀྱིས་མགོན་པོ་ཀླུ་སྒྲུབ་ལ་འདི་སྐད་དུ། ངེད་རྣམས་ནི་དགྲ་བཅོམ་པ་ཡིན་པས་ཀླུ་རྣམས་ཀྱིས་མཐོང་བ་དང་ཐོས་པ་དང་རེག་པའི་དུག་སོགས་ཀྱིས་མི་ཚུགས་པ་ཡིན། ཁྱོད་སོ་སྐྱེ་ཞིག་ཡིན་འདྲ་བ་ལ་དུག་གིས་མི་ཚུགས་པ་ཅི་ཡིན་གསུང་བ་ན། མགོན་པོ་ཀླུ་སྒྲུབ་ཀྱིས། ང་ལ་རིག་སྔགས་ཀྱི་རྒྱལ་མོ་རྨ་བྱ་ཆེན་མོ་ཡོད་གསུངས་པས། དགྲ་བཅོམ་པ་དེ་རྣམས་ཀྱིས་རིག་སྔགས་རྨ་བྱ་ཆེན་མོ་ཟེར་ལུགས་ཀྱིས་སངས་རྒྱས་ཤཱཀྱ་ཐུབ་པ་ཡང་ཕེབས་ཚར་བ་ཨེ་ཡིན་གསུང་བ་བྱུང་གསུངས། ཡང་ཀླུའི་གནས་གཅིག་ན་སངས་རྒྱས་ཤཱཀྱ་ཐུབ་པའི་བསྟན་པའི་ཉན་ཐོས་དགྲ་བཅོམ་པ་མང་དུ་བཞུགས་འདུག་པ་ལ། དེ་རྣམས་ཀྱིས་ཤེར་ཕྱིན་གྱི་མདོ་ནས་ཇི་ལྟར་གསུངས་པ་ལྟར་གྱི་སྟོང་པ་ཉིད་འདི་སྟོང་ཉིད་ཀྱི་དོན་ཕྱིན་ཅི་མ་ལོག་པ་དང་འདི་ཉིད་ཉན་ཐོས་སོགས་ཀྱི་བསྒོམ་བྱ་ཡིན་གསུང་བ་དང། མགོན་པོ་ཀླུ་སྒྲུབ་ཀྱིས་དེ་ལྟར་ཡིན་ཀྱང་ད་ལྟ་ས་སྟེང་ན་ནི་ལྟ་བ་སྣ་ཚོགས་ཡོད་ཚུལ་ཞུས་གསུངས། ཡང་འཕགས་པ་ཐོགས་མེད་ས་གསུམ་པ་བ་ཡིན་པས། ངེས་པར་སྟོང་པ་ཉིད་མངོན་སུམ་དུ་རྟོགས་པ་ཡིན་པ་ལ་ས་སྡེ་ལྔའི་སྐབས་ཕྱི་རོལ་དོན་གྱིས་སྟོང་བའི་སྟོང་ཉིད་གཏན་ལ་ཕབ་པ་སྤྱིར་གདུལ་བྱའི་དབང་གིས་ཡིན་མོད། འོན་ཀྱང་སྔོན་གྱི་གནས་བརྟན་ཆོས་སྐྱོང་གིས་ཀྱང་སྔགས་ཀྱི་རྟོགས་པ་མཐོན་པོ་ལ་མ་སླེབ་བར་དུ་སྔགས་ཀྱི་ལམ་དེ་སེམས་ཙམ་པའི་ལྟ་བ་དང་སྦྱར་ནས་བསྒོམ་པར་གསུངས་པས་འདི་ལ་དཔྱད་རྒྱུ་ཞིག་ལྟ་འདུག་གསུངས། ༈ ལྷག་མཐོང་གི་ཚོགས་ཀྱི་སྐབས། ལྷ་དང་བླ་མ་དབྱེར་མེད་དུ་བྱས་ནས་གསོལ་བ་འདེབས་པ། ལས་ཀྱི་སྒྲིབ་པ་སྦྱོང་བ། ལྷག་མཐོང་གི་༄༅། །མན་ངག་སྟོན་བའི་གཞུང་ཆེན་མོ་རྣམས་ལ་བལྟ་བ་སྟེ་གསུམ་གསུངས་པའི་དང་པོ་ནི་རང་ཅག་རྣམས་ཀྱིས་ནི་རྟེན་འབྲེལ་གྱི་ཁྱད་ཀྱིས། བླ་མ་རྗེ་བཙུན་འཇམ་དབྱངས་དང་དབྱེར་མེད་དུ་བསྒོམ་དགོས། འཇམ་དབྱངས་ཞི་ཁྲོ་ལ་ཁྱད་མེད་སྟོང་ཉིད་བསྒོམ་པ་སྡིག་པ་དག་པའི་མཆོག་ཡིན་མོད་ཀྱང། སྟོང་ཉིད་དང་པོར་རྟོགས་པ་རང་ལ་ཡང་ལས་སྒྲིབ་ཀྱིས་གེགས་བྱེད་བས་ཡན་ལག་བདུན་པ་སོགས་ཀྱིས་བསག་སྦྱངས་བྱེད། ལྷག་མཐོང་གི་མན་ངག་སྟོན་པའི་གཞུང་ཆེན་མོ་ནི། དབུ་མ་རིགས་ཚོགས་དྲུག །འགྲེལ་པ་བུདྡྷ་པཱ་ལི་ཏ། ཟླ་བའི་དབུ་མ་འཇུག་པ། འགྲེལ་པ་ཚིག་གསལ། རྗེ་རིན་པོ་ཆེའི་རྩ་འཇུག་གི་རྣམ་བཤད་གཉིས། ལམ་རིམ་ཆེ་ཆུང་སོགས་ཡིན། རྩ་འཇུག་གི་རྣམ་བཤད་གཉིས་དང། ལམ་རིམ་རྣམས་ལ་སྟོང་ཉིད་སྟོན་པའི་རིགས་པའི་ཚུལ་ཚིག་ཟིན་ལ་མི་མཐུན་པ་འགའ་རེ་འབྱུང་བ་ནི། གདུལ་བྱ་སེམས་ཙམ་པ་སོགས་ཀྱི་དངོས་སྨྲ་བའི་བག་ཆགས་མཐུག་པའི་རིགས་ཅན་དང། དང་པོ་ཉིད་ནས་དབུ་མ་པའི་རིགས་ཅན་སོགས་ཀྱི་ཁྱད་པར་ཡིན་འདུག་གསུངས། རྗེ་རིན་པོ་ཆེས། འགྲེལ་པ་བུདྡྷ་པཱ་ལི་ཏ་ལ་བལྟས་ན་ལྟ་བའི་གོ་ཚུལ་ཁྱད་པར་ཅན་འབྱུང་བར་གསུངས་ཀྱང། ལས་དང་པོ་པ་ཚོ་ལ་དཔེ་གསལ་པོ་འགའ་ཞིག་ཡོད་པ་མ་གཏོགས་ཧ་ཅང་ཚིག་བརླིང་ཞིང་རིགས་པ་ཟབ་པས་གོ་དཀའ་སེ་བ་ཞིག་ཡིན། འོན་ཀྱང་རྩ་ཤེའི་ཊཱི་ཀ་ཆེན་དུ་དཀའ་བའི་གནད་གང་ཡོད་སར་འགྲེལ་པ་དེའི་ལུང་འདྲེན་རེ་ངེས་པར་མཛད་ཅིང་བཤད་པ་མཛད་ཡོད། ལུང་འདྲེན་དེ་རྣམས་ཀྱང་འགའ་ཞིག་ནི་སྔ་མའི་འཕྲོས་དོན་ལྟ་བུ་དང། འགའ་ཞིག་ཐད་ཀའི་བཤད་པ་དེའི་རྒྱབ་སྐྱོར་ལྟ་བུར་མཛད་ཡོད་གསུངས། ཡང་འཕྲོས་ནས་རྣམ་འགྲེལ་ཊཱིཀ་ཆེན་རིགས་བའི་རྒྱ་མཚོ་དང། ཐར་ལམ་གསལ་བྱེད་གཉིས་སུ་སེམས་ཙམ་པའི་ལྟ་བ་གཏན་ལ་འབེབས་ཚུལ་མི་འདྲ་བ་རེ་བྱུང་ཞིང། གཉིས་ཀ་ཡང་རྗེ་རིན་པོ་ཆེའི་དགོངས་པ་ཡིན་པར་གསུངས་ཡོད། རང་རེ་ཚོས་བརྟགས་ན་གཅིག་ལུང་གི་རྗེས་འབྲངས་དང། གཅིག་རིགས་པའི་རྗེས་འབྲང་གི་སེམས་ཙམ་པའི་དབང་དུ་བྱས་པ་འདྲ་མཐར་གཏུགས་ན་ནི་མའི་འགལ་ཞེས་གསུངས། ཡང་ལམ་རིམ་ཆེན་མོ་ནི་གཙོ་བོར་བཤད་རྒྱུད་དང། ཆུང་བ་ནི་སྙན་བརྒྱུད་གཙོ་ཆེ་བའི་དབང་དུ་མཛད། ལམ་རིམ་ཆེན་མོར་མི་གསལ་པའི་ཞི་ལྷག་གི་གནང་དུ་མྮིག་ལམ་རིམ་ཆུང་བར་ཡོད་གསུངས། གནད་བཞིའི་དཔྱད་པའི་སྐབས་ང་དེ་རང་གི་གདགས་གཞིའི་ཕུང་པོ་ལྔ་དང་གཅིག་ན། ཕུང་པོ་ལྔ་ཡོད་པ་ལྟར་བདག་དེ་ཡང་རྒྱུད་མི་གཅིག་པ་ལྔར་འགྱུར་བའམ། བདག་གཅིག་ཡིན་པ་ལྟར་ཕུང་པོ་ལྔ་ཡང་གཅིག་ཏུ་ཐལ་བའི་སྐྱོན་འབྱུང་སྟེ། ཕུང་པོ་ལྔ་པོ་དེ་ནང་ཕན་ཚུན་འགལ་བ་གང་ཞིག །དེ་རྣམས་བདག་དང་གཅིག་ཡིན་པའི་ཕྱིར་གསུངས། འདིའི་དཔེར་ལྕགས་ཀྱི་ཐུར་མ་ལྔ་དང་གསེར་གྱི་ཐུར་མ་གཅིག་ཡོད་ན་དེ་རྣམས་གཅིག་ན་ནི་ལྕགས་ཐུར་ལྔ་ཡོད་པ་ལྟར་གསེར་ཐུར་དེ་ཡང་ལྔར་འགྱུར་བའམ། ཡང་གསེར་ཐུར་གཅིག་ཡིན་པ་ལྟར་ལྕགས་ཐུར་ལྔ་ཡོད་པ་དེ་ཡང་གཅིག་ཏུ་འགྱུར་བའི་སྐྱོན་ཡོད། ཡང་ང་དེ་རང་གི་གདག་༄༅། །གཞིའི་ཕུང་པོ་ལྔ་དང་ཐ་དད་ན་ནི་ཕུང་པོ་རྣམས་སོ་སོར་བསལ་བའི་འོག་ཏུ་ང་དེ་འདིའོ་ཞེས་རྙེད་རྒྱུ་ཞིག་ཡོད་དགོས་པ་ལས། དེ་འདྲ་མེད་པའི་ཕྱིར་གསུངས་ཡང་དེ་རིང་གི་ཤེས་པ་དེ་བདེན་པར་གྲུབ་ན། ཚུགས་ཐུབ་ཏུ་གྲུབ་དགོས། དེ་ཚུགས་ཐུབ་ཡིན་ན་དེ་རིང་གི་ཤེས་པའི་ངོ་བོར་ཚུགས་ཐུབ་ཡིན་དགོས། དེ་ཡིན་ན་ཁ་སང་གི་ཤེས་པ་ལས་མ་བྱུང་དགོས། བྱུང་ན་ནི་དེ་རིང་གི་ཤེས་པ་དེ་ཆོས་ཅན། ཁྱོད་ཚུགས་ཐུབ་མ་ཡིན་ཏེ་ཁ་སང་གི་ཤེས་པ་ལ་རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བའི་ཕྱིར་གསུངས། ༈ ཉིན་བཅུ་དྲུག་པར། བསྡུས་དོན་གྱི་སྐབས་སངས་རྒྱ་བ་ལ་བཞི་བའི་སྒོ་གཉིས་སུ་མེད་པ་ལྟ་བུ་ནི། རྒྱུད་སྡེ་བཞིའི་ནང་ནས་རྣལ་འབྱོར་བླ་ན་མེད་པའི་རྒྱུད་འདི་ཉིད་ཡིན། དེས་ན་རྡོ་རྗེ་སློབ་དཔོན་མཚན་ཉིད་དང་ལྡན་པའི་དྲུང་དུ་དབང་བཞི་རྣམ་པར་དག་པ་ཐོབ་པ་ཞིག་བྱས། དམ་ཚིག་དང་སྡོམ་པ་མིག་གི་འབྲས་བུ་བཞིན་དུ་བསྲུངས་ནས། ཐུན་མོང་བའི་ལམ་རིམ་འདི་རྣམས་ཀྱང་རྣལ་འབྱོར་བླ་ན་མེད་བའི་ལམ་རིམ་དང་སྦྲེལ་ནས། མགོ་བཤེས་གཉེན་བསྟེན་ཚུལ་ནས་བཟུང། མཇུག་མི་སློབ་པའི་ཟུང་འཇུག་གི་བར་སྟན་ཐོག་གཅིག་ལ་ཉམས་སུ་བླངས་ནས། དེ་ཐམས་ཅད་ཀྱི་སྐབས་སུ་ཡང་བླ་མའི་རྣལ་འབྱོར་ལམ་གྱི་སྲོག་ཤིང་དུ་བྱས་པའི་སྒོ་ནས་སྙིགས་དུས་ཀྱི་ཚེ་ཐུང་གཅིག་རང་ལ་རྡོ་རྗེ་འཆང་གི་གོ་འཕང་མངོན་དུ་བྱས་ནས། སེམས་ཅན་ཐམས་ཅད་ཀྱི་རེ་བ་སྐོང་བའི་ཡིད་བཞིན་གྱི་ནོར་བུ་རིན་པོ་ཆེར་འགྱུར་བའི་འདུན་པ་མཛོད་གསུངས། ༧རིགས་དང་དཀྱིལ་འཁོར་ཀུན་གྱི་ཁྱབ་བདག་རྡོ་རྗེ་འཆང་ཀུན་གཟིགས ༧པཎ་ཆེན་ཆོས་ཀྱི་རྒྱལ་པོས་བདེ་ལམ་གྱི་ཟབ་ཁྲིད་སྐུ་དྲིན་བསྐྱངས་སྐབས། དངོས་གཞིའི་དམིགས་སྐོར་རྣམས་ཚིག་སྣ་རིང་ཐུང་གི་ཁྱད་ཙམ་མ་གཏོགས་བདེ་ལམ་རྩ་བ་ནང་བཞིན་སྩལ་བའི་ཟིན་བྲིས་མ་དགོས་ཤིང། མཐའ་དཔྱོད་ཀྱི་གསུང་བཤད་ཤིན་ཏུ་རྒྱས་པ་ལུང་རིགས་ཀྱིས་བརྒྱན་ཏེ་མཛད་པ་རྣམས་ལས་རགས་པ་ཐོ་རེ་ལོང་དེ་ཞིག་རང་བློར་གསལ་པ་ལས་ཕལ་ཆེར་གསུང་ཇི་བཞིན་དང་འགའ་ཞིག་ཆིག་སྣ་ཅུང་ཟད་བསྡུས། དེ་དག་གི་ཤེས་བྱེད་ལ་སྔོན་གྱི་སྐྱེས་རབས་དང་རྟོགས་བརྗོད་དང་གཏམ་རྒྱུད་སྦྱར་བ་རྣམས་ལས་ཉུང་བ་རྣམས་གང་གསུང་ལྟར་དང། མང་བ་རྣམས་འགའ་ཞིག་ཕྱོགས་ཙམ་དང་འགའ་ཞིག་མིང་སྨོས་ཙམ་བྱས། གཞན་མ་འདིའི་ཤེས་བྱེད་ལ་འདི་ལྟར་གསུངས་ཟེར་བ་ལྟ་བུའི་མཚམས་སྦྱོར་གྱི་ཚིག་རྣམས་མ་གཏོགས་བསམ་བཞིན་དུ་རང་བཟོ་བྱས་པ་ནི་གཏན་ནས་མེད། མ་རྟོགས་ལོག་རྟོག་ཐེ་ཚོམ་གྱི་དྲི་མའི་སྐྱོན་གྱིས་གོས་པ་ནི་ལོས་ཡོང། གང་ལྟར་རང་གི་བརྗེད་ཐོ་དང་ཞར་ལ་བླ་མའི་གསུང་ལ་གཅེས་སྤྲས་བྱེད་པ་འགའ་རེ་ལ་ཕན་བསམས་ནས ༧རྒྱལ་དབང་མཆོག་གི ༧བཀའ་འབངས་ཀྱི་ཐ་ཤལ་འདུ་གསུམ་པ་མི་ཐུང་སྐུ་སྐྱེའི་མིད་ཅན་གྱིས་ཟིན་བྲིས་བྱས་པ་ཡིན། ལོ་འདོད་དང་གྲགས་འདོད་བྱེད་དགོས་མེད། ཤེས་རབ་ཡོད་ན་ནི་གསུང་རབ་ཀྱི་ཚིག་དོན་མང་པོ་ཞིག་ལ་རྟོག་དབྱོད་འཇུག་པའི་ཐབས་སུ་ཡོད་ལ། སྭསྟི། ཕན་བདེའི་འབྱུང་གནས་རྒྱལ་བསྟན་ཡོངས་འདུའི་ལྗོན། །རབ་རྒྱས་འགྲོ་ཀུན་ཐར་མཆོག་འབྲས་བཟང་ལ། །སྤྱོད་ཕྱིར་བཀྲ་ཤིས་ལྷུན་གྲུབ་ཆོས་གྲྭ་ཆེར། །ཆོས་སྦྱིན་འཛད་མེད་དལ་འགྲོའི་རྒྱུན་འདི་སྤེལ། །མངྒ་ལཾ༎ ༎
